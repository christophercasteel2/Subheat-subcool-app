// ================= number formatting =================

// whole number as text, independent of the calculator's number format
istr(v)
BEGIN
  LOCAL n:=ROUND(ABS(v),0);
  LOCAL s:="";
  IF n<1 THEN RETURN "0"; END;
  WHILE n>=1 DO
    s:=CHAR(48+(n MOD 10))+s;
    n:=IP(n/10);
  END;
  IF v<0 THEN s:="-"+s; END;
  RETURN s;
END;

// one decimal place
fmt1(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*10,0);
  LOCAL s:=istr(IP(n/10))+"."+CHAR(48+(n MOD 10));
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// v with d decimal places (d = 0 to 5)
fmtN(v,d)
BEGIN
  LOCAL m:=10^d;
  LOCAL n:=ROUND(ABS(v)*m,0);
  LOCAL s:=istr(IP(n/m));
  LOCAL f;
  IF d>0 THEN
    f:=istr(n MOD m);
    WHILE DIM(f)<d DO f:="0"+f; END;
    s:=s+"."+f;
  END;
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// up to d decimals, trailing zeros dropped
fmtTr(v,d)
BEGIN
  LOCAL s:=fmtN(v,d);
  IF d>0 THEN
    WHILE RIGHT(s,1)=="0" DO s:=LEFT(s,DIM(s)-1); END;
    IF RIGHT(s,1)=="." THEN s:=LEFT(s,DIM(s)-1); END;
  END;
  RETURN s;
END;

// any size of number; very large or small ones as 1.23E-6
fmtE(v)
BEGIN
  LOCAL a:=ABS(v);
  LOCAL ex, m;
  IF a<0.000000000001 THEN RETURN "0"; END;
  IF a>=0.000001 AND a<10000000 THEN
    // six significant figures
    ex:=5-FLOOR(LOG(a)+0.0000000001);
    IF ex>5 THEN RETURN fmtTr(v,MIN(11,ex)); END;
    ex:=5;
    IF a>=10 THEN ex:=4; END;
    IF a>=100 THEN ex:=3; END;
    IF a>=1000 THEN ex:=2; END;
    IF a>=10000 THEN ex:=1; END;
    IF a>=100000 THEN ex:=0; END;
    RETURN fmtTr(v,ex);
  END;
  ex:=FLOOR(LN(a)/LN(10));
  m:=a/10^ex;
  IF m>=9.9995 THEN
    m:=m/10;
    ex:=ex+1;
  END;
  IF m<1 THEN
    m:=m*10;
    ex:=ex-1;
  END;
  RETURN IFTE(v<0,"-","")+fmtTr(m,5)+"E"+istr(ex);
END;

// sine, cosine and arctangent in degrees, whatever the angle setting

BEGIN

END;

BEGIN
  IF HAngle==0 THEN RETURN COS(a*3.14159265358979/180); END;
  RETURN COS(a);
END;

atanD(x)
BEGIN
  IF HAngle==0 THEN RETURN ATAN(x)*180/3.14159265358979; END;
  RETURN ATAN(x);
END;

// angle (degrees) whose cosine is x
acosD(x)
BEGIN
  LOCAL c:=MAX(-1,MIN(1,x));
  LOCAL a;
  IF ABS(c)<0.000000000001 THEN RETURN 90; END;
  a:=atanD(sqroot(1-c*c)/c);
  IF c<0 THEN a:=a+180; END;
  RETURN a;
END;

// the value of π the problem asks for: 1 exact, 2 3.14, 3 3.1416, 4 22/7
piOf(c)
BEGIN
  IF c==2 THEN RETURN 3.14; END;
  IF c==3 THEN RETURN 3.1416; END;
  IF c==4 THEN RETURN 22/7; END;
  RETURN 3.14159265358979;
END;

// degrees as degrees, minutes and seconds, to a tenth of a second;
// 0 minutes and seconds are left off: 135°15', 29°5'36''
dmsTxt(v)
BEGIN
  LOCAL t:=ROUND(ABS(v)*36000,0);
  LOCAL dg:=IP(t/36000);
  LOCAL mn:=IP((t-dg*36000)/600);
  LOCAL sc:=(t-dg*36000-mn*600)/10;
  LOCAL s:=IFTE(v<0 AND t>0,"-","")+istr(dg)+"°";
  IF mn>0 OR sc>0 THEN s:=s+istr(mn)+"'"; END;
  IF sc>0 THEN s:=s+fmtTr(sc,1)+"''"; END;
  RETURN s;
END;

// feet as feet and inches to the nearest 1/16 in.: 6 ft 7 5/8 in
ftIn(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*192,0);
  LOCAL ft:=IP(n/192);
  LOCAL r:=n-ft*192;
  LOCAL wi:=IP(r/16);
  LOCAL f:=r-wi*16;
  LOCAL dn:=16;
  LOCAL s;
  WHILE f>0 AND (f MOD 2)==0 DO
    f:=f/2;
    dn:=dn/2;
  END;
  s:=IFTE(v<0 AND n>0,"-","")+istr(ft)+" ft "+istr(wi);
  IF f>0 THEN s:=s+" "+istr(f)+"/"+istr(dn); END;
  RETURN s+" in";
END;

// kind of angle
angType(a)
BEGIN
  IF a<=0 THEN RETURN "none"; END;
  IF ABS(a-90)<0.0000001 THEN RETURN "right"; END;
  IF a<90 THEN RETURN "acute"; END;
  IF ABS(a-180)<0.0000001 THEN RETURN "straight"; END;
  IF a<180 THEN RETURN "obtuse"; END;
  IF a<360 THEN RETURN "reflex"; END;
  RETURN "full turn or more";
END;

  IF gSD(s)<0 THEN t:=fmtTr(v,4); ELSE t:=fmtN(v,gSD(s)); END;
  IF u=="" THEN RETURN t; END;
  IF LEFT(u,1)=="°" OR LEFT(u,1)=="'" OR u=="%" THEN RETURN t+u; END;
  RETURN t+" "+u;
END;

setSlot(s,v)
BEGIN
  gV(s):=MAX(gSLo(s),MIN(gSHi(s),v));
  IF gSC(s)>0 THEN gV(s):=ROUND(gV(s),0); END;
END;

// Up/Down: step a number, or move through a choice list
bumpSlot(s,d)
BEGIN
  LOCAL n;
  IF gSC(s)>0 THEN
    n:=SIZE(choiceList(gSC(s)));
    gV(s):=gV(s)+IFTE(d>0,1,-1);
    IF gV(s)>n THEN gV(s):=1; END;
    IF gV(s)<1 THEN gV(s):=n; END;
  ELSE
    setSlot(s,gV(s)+d*gSS(s));
  END;
END;

// a two-way choice flips, a longer list opens, a number opens a dialog
editSlot(s)
BEGIN
  LOCAL v:=gV(s);
  LOCAL lst;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    IF SIZE(lst)==2 THEN
      setSlot(s,IFTE(v==1,2,1));
    ELSE
      IF CHOOSE(v,slotName(s),lst) THEN
        IF v>=1 THEN setSlot(s,v); END;
      END;
    END;
  ELSE
    IF INPUT(v,slotName(s),slotName(s)+":",gSH(s)) THEN
      IF TYPE(v)==0 THEN setSlot(s,v); END;
    END;
  END;
  gDown:=0;
END;

  LOCAL kc:={47,42,43,44,37,38,39,32,33,34,48};

entVal()
BEGIN
  LOCAL v:=0;
  LOCAL sc:=0;
  LOCAL k, d;
  FOR k FROM 1 TO SIZE(gEK) DO

      sc:=1;
    ELSE
      IF sc>0 THEN
        sc:=sc/10;
        v:=v+d*sc;
      ELSE
        v:=v*10+d;
      END;
    END;
  END;
  RETURN IFTE(gENeg,-v,v);
END;

// a length in other units (in, ft, yd or mm, cm, m) and, in feet and

  IF u==2 THEN addRow("In sq yd",vtxt(ar/9,"yd²"),"",cDim); END;

  IF u==5 THEN addRow("In sq m",vtxt(ar/10000,"m²"),"",cDim); END;

// 1 angle arithmetic in degrees, minutes and seconds
calcAng()
BEGIN
  LOCAL a:=dmsIn(gV(1),gV(2),gV(3));
  LOCAL b:=dmsIn(gV(5),gV(6),gV(7));
  LOCAL op:=gV(4);
  LOCAL n:=gV(8);
  LOCAL r:=a;
  LOCAL fm:={"A + B","A - B","A x n","A ÷ n","180 - (A + B)","90 - A","180 - A","A"};
  IF op==1 THEN r:=a+b; END;
  IF op==2 THEN r:=a-b; END;
  IF op==3 THEN r:=a*n; END;
  IF op==4 THEN
    IF n==0 THEN
      addRow("n","cannot be 0","",cWarn);
      RETURN 0;
    END;
    r:=a/n;
  END;
  IF op==5 THEN r:=180-a-b; END;
  IF op==6 THEN r:=90-a; END;
  IF op==7 THEN r:=180-a; END;
  addRow(fm(op),dmsTxt(r),"",cSelc);
  addRow("Decimal",fmtE(r)+"°","",cSelc);
  addRow("Type",angType(r),"",cText);
  addRow("A",dmsTxt(a)+" = "+fmtE(a)+"°","",cText);
  IF op<=2 OR op==5 THEN addRow("B",dmsTxt(b)+" = "+fmtE(b)+"°","",cText); END;
  IF op==5 AND r<=0 THEN addRow("","A + B must be under 180°","",cWarn); END;
  RETURN 1;
END;

// 4 the third angle of a triangle
calcTri()
BEGIN
  LOCAL a:=dmsIn(gV(14),gV(15),0);
  LOCAL b:=dmsIn(gV(16),gV(17),0);
  LOCAL c, big;
  IF gV(13)==2 THEN
    IF a>=180 THEN
      addRow("Apex A","must be under 180°","",cWarn);
      RETURN 0;
    END;
    b:=(180-a)/2;
    c:=b;
    addRow("Apex A",dmsTxt(a),"",cText);
    addRow("Base angles",dmsTxt(b),"each",cSelc);
    addRow("Decimal",fmtE(b)+"°","",cSelc);
  ELSE
    c:=180-a-b;
    IF c<=0 THEN
      addRow("C","A + B must be under 180°","",cWarn);
      RETURN 0;
    END;
    addRow("Angle C",dmsTxt(c),"",cSelc);
    addRow("Decimal",fmtE(c)+"°","",cSelc);
    addRow("A + B",dmsTxt(a+b),"",cText);
  END;
  big:=MAX(a,MAX(b,c));
  addRow("Triangle",IFTE(ABS(big-90)<0.0000001,"right",IFTE(big>90,"obtuse","acute")),"",cText);
  IF ABS(a-60)<0.0000001 AND ABS(b-60)<0.0000001 THEN addRow("","equilateral (all 60°)","",cDim); END;
  RETURN 1;
END;

// 6 rectangle
calcRect()
BEGIN
  LOCAL u:=gV(27);
  LOCAL l:=gV(28);
  LOCAL w:=gV(29);
  LOCAL ar:=gV(30);
  LOCAL fd:=0;
  IF l>0 AND w>0 THEN
    ar:=l*w;
  ELSE
    IF l>0 AND ar>0 THEN
      w:=ar/l;
      fd:=2;
    ELSE
      IF w>0 AND ar>0 THEN
        l:=ar/w;
        fd:=1;
      ELSE
        addRow("Enter","L and W, or A","",cWarn);
        RETURN 0;
      END;
    END;
  END;
  IF fd==1 THEN addRow("Length L",vtxt(l,lenU(u)),"",cSelc); END;
  IF fd==2 THEN addRow("Width W",vtxt(w,lenU(u)),"",cSelc); END;
  addRow("Area",vtxt(ar,lenU(u)+"²"),"",IFTE(fd==0,cSelc,cText));
  addRow("Perimeter",vtxt(2*l+2*w,lenU(u)),"",cSelc);
  addRow("Diagonal",vtxt(sqroot(l*l+w*w),lenU(u)),"",cText);
  areaConv(ar,u);
  RETURN 1;
END;

// 10 cost of material: the amount in the unit of the price, waste,

// 11 Pythagorean theorem: the side left at 0
calcPyth()
BEGIN
  LOCAL u:=gV(51);
  LOCAL a:=gV(52);
  LOCAL b:=gV(53);
  LOCAL c:=gV(54);
  LOCAL fd:=0;
  LOCAL x;
  IF c==0 AND a>0 AND b>0 THEN
    c:=sqroot(a*a+b*b);
    fd:=3;
  END;
  IF a==0 AND b>0 AND c>b THEN
    a:=sqroot(c*c-b*b);
    fd:=1;
  END;
  IF b==0 AND a>0 AND c>a THEN
    b:=sqroot(c*c-a*a);
    fd:=2;
  END;
  IF fd==0 THEN
    IF a>0 AND b>0 AND c>0 THEN
      x:=ABS(a*a+b*b-c*c)<0.000001*c*c;
      addRow("Right angle?",IFTE(x,"yes: a² + b² = c²","no"),"",IFTE(x,cGood,cWarn));
      addRow("a² + b²",fmtE(a*a+b*b),"",cText);
      addRow("c²",fmtE(c*c),"",cText);
      RETURN 1;
    END;
    addRow("Enter","2 sides; 0 = unknown","",cWarn);
    addRow("","The hypotenuse is longest","",cWarn);
    RETURN 0;
  END;
  x:=IFTE(fd==1,a,IFTE(fd==2,b,c));
  addRow(IFTE(fd==3,"Hyp. c",IFTE(fd==1,"Leg a","Leg b")),vtxt(x,lenU(u)),"",cSelc);
  IF u<=2 THEN addRow("Ft-in",ftIn(x*inPer(u)/12),"",cSelc); END;
  addRow("Area",vtxt(a*b/2,lenU(u)+"²"),"",cText);
  addRow("Perimeter",vtxt(a+b+c,lenU(u)),"",cText);
  RETURN 1;
END;

// 12 triangle area from base and height
calcTriA()
BEGIN
  LOCAL u:=gV(55);
  LOCAL b:=gV(56);
  LOCAL h:=gV(57);
  LOCAL ar:=gV(58);
  IF b>0 AND h>0 THEN
    ar:=b*h/2;
    addRow("Area",vtxt(ar,lenU(u)+"²"),"(1/2) b h",cSelc);
  ELSE
    IF b>0 AND ar>0 THEN
      addRow("Height h",vtxt(2*ar/b,lenU(u)),"2A / b",cSelc);
    ELSE
      IF h>0 AND ar>0 THEN
        addRow("Base b",vtxt(2*ar/h,lenU(u)),"2A / h",cSelc);
      ELSE
        addRow("Enter","b and h, or A","",cWarn);
        RETURN 0;
      END;
    END;
  END;
  areaConv(ar,u);
  RETURN 1;
END;

// 13 triangle from its three sides (Heron; isosceles height)
calcTri3()
BEGIN
  LOCAL u:=gV(59);
  LOCAL a:=gV(60);
  LOCAL b:=gV(61);
  LOCAL c:=gV(62);
  LOCAL s:=(a+b+c)/2;
  LOCAL ar, ga, gb, gc;
  IF a<=0 OR b<=0 OR c<=0 OR a+b<=c OR a+c<=b OR b+c<=a THEN
    addRow("Not a triangle","","",cWarn);
    addRow("","Each side < the other two","",cWarn);
    RETURN 0;
  END;
  ar:=sqroot(s*(s-a)*(s-b)*(s-c));
  gc:=acosD((a*a+b*b-c*c)/(2*a*b));
  ga:=acosD((b*b+c*c-a*a)/(2*b*c));
  gb:=180-ga-gc;
  addRow("Area",vtxt(ar,lenU(u)+"²"),"",cSelc);
  addRow("Perimeter",vtxt(a+b+c,lenU(u)),"",cSelc);
  addRow("Sides",IFTE(a==b AND b==c,"equilateral",IFTE(a==b OR b==c OR a==c,"isosceles","scalene")),"",cText);
  addRow("Angles",IFTE(MAX(ga,MAX(gb,gc))>90.0000001,"obtuse",IFTE(MAX(ga,MAX(gb,gc))>89.9999999,"right","acute")),"",cText);
  addRow("Height to c",vtxt(2*ar/c,lenU(u)),"",cText);
  addRow("A, B, C",fmtN(ga,1)+", "+fmtN(gb,1)+", "+fmtN(gc,1)+"°","angles across from a, b, c",cText);
  RETURN 1;
END;

// box s as a label: its name and value, or ? when it is 0
dl(nm,s)
BEGIN
  IF gV(s)==0 THEN RETURN nm+"=?"; END;

dash(x1,y1,x2,y2,c)
BEGIN
  LOCAL n:=MAX(2,IP(MAX(ABS(x2-x1),ABS(y2-y1))/3));
  LOCAL k;
  FOR k FROM 0 TO n-1 STEP 2 DO
    LINE_P(G1,ROUND(x1+(x2-x1)*k/n,0),ROUND(y1+(y2-y1)*k/n,0),ROUND(x1+(x2-x1)*(k+1)/n,0),ROUND(y1+(y2-y1)*(k+1)/n,0),c);
  END;
END;

// a line two pixels wide (the side being typed)
thick(x1,y1,x2,y2,c)
BEGIN
  LINE_P(G1,x1,y1,x2,y2,c);
  LINE_P(G1,x1+1,y1,x2+1,y2,c);
  LINE_P(G1,x1,y1+1,x2,y2+1,c);
END;

// arc around (cx,cy), radius r, from angle a1 to a2 (degrees, counter-
// clockwise from the right)
arcP(cx,cy,r,a1,a2,c)
BEGIN
  LOCAL n:=MAX(2,CEILING(ABS(a2-a1)/8));
  LOCAL k, t1, t2;
  FOR k FROM 0 TO n-1 DO
    t1:=a1+(a2-a1)*k/n;
    t2:=a1+(a2-a1)*(k+1)/n;
    LINE_P(G1,ROUND(cx+r*cosD(t1),0),ROUND(cy-r*sinD(t1),0),ROUND(cx+r*cosD(t2),0),ROUND(cy-r*sinD(t2),0),c);
  END;
END;

// a ray from (x,y) at angle a, length n
ray(x,y,a,n,c)
BEGIN
  LINE_P(G1,x,y,ROUND(x+n*cosD(a),0),ROUND(y-n*sinD(a),0),c);
END;

// right-angle mark at corner (x,y) opening right (dx 1) or left (-1), up
rmark(x,y,dx)
BEGIN
  LINE_P(G1,x+6*dx,y,x+6*dx,y-6,cLn);
  LINE_P(G1,x,y-6,x+6*dx,y-6,cLn);
END;

BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL k;
  FOR k FROM 1 TO SIZE(sl) DO
    gV(sl(k)):=gDef(sl(k));
  END;
  gEnt:=0;
END;

// open tile k: a tool, or the Learn pages
openTile(k)
BEGIN
  gHome:=k;
  IF k>gNT THEN
    learnMenu();
  ELSE
    openTool(k);
  END;
END;

// ================= input handling =================

// returns 0 to quit
onHome(ev)
BEGIN
  LOCAL mx, my, k;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    IF my<21 THEN
      IF mx>=272 THEN RETURN 0; END;
      IF mx>=224 THEN showHelp(); END;
      RETURN 1;
    END;
    k:=IP((my-23)/43)*4+MIN(3,IP(mx/80))+1;
    IF my>=23 AND k>=1 AND k<=gNT+1 THEN openTile(k); END;
    RETURN 1;
  END;
  IF ev==7 THEN gHome:=IFTE(gHome==1,gNT+1,gHome-1); END;
  IF ev==8 THEN gHome:=IFTE(gHome==gNT+1,1,gHome+1); END;
  IF ev==2 THEN
    gHome:=gHome-4;
    WHILE gHome<1 DO gHome:=gHome+4*IP((gNT+4)/4); END;
    WHILE gHome>gNT+1 DO gHome:=gHome-4; END;
  END;
  IF ev==12 THEN
    gHome:=gHome+4;
    IF gHome>gNT+1 THEN gHome:=((gHome-1) MOD 4)+1; END;
  END;
  IF ev==30 THEN openTile(gHome); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN RETURN 0; END;
  RETURN 1;
END;

onTool(ev)
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, mx, my, rw, idx, dg;
  IF gTSel>n THEN gTSel:=1; END;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    commitEnt();
    IF my>=220 THEN
      k:=IP(mx/64)+1;
      IF k==1 THEN editSlot(sl(gTSel)); END;
      IF k==2 THEN resetTool(); END;
      IF k==3 THEN openTool(IFTE(gTool==1,gNT,gTool-1)); END;
      IF k==4 THEN openTool(IFTE(gTool==gNT,1,gTool+1)); END;
      IF k==5 THEN goHome(); END;
      RETURN 1;
    END;
    IF my<21 THEN
      IF mx>=272 THEN toolHelp(gTool); END;
      RETURN 1;
    END;
    // tap an input box to change it
    rw:=IP((my-23)/19);
    idx:=rw+1;
    IF my>=23 AND mx<160 AND idx>=1 AND idx<=n THEN
      gTSel:=idx;
      editSlot(sl(idx));
    END;
    RETURN 1;
  END;
  // digits and the decimal point type straight into a number box
  dg:=keyDigit(ev);
  IF dg>=0 THEN
    IF gSC(sl(gTSel))==0 THEN typeKey(dg); END;
    RETURN 1;
  END;
  IF gEnt THEN
    IF ev==45 OR ev==27 THEN
      gENeg:=1-gENeg;
      RETURN 1;
    END;
    IF ev==19 THEN
      backKey();
      RETURN 1;
    END;
    IF ev==4 THEN
      gEnt:=0;
      RETURN 1;
    END;
    // any other key keeps the typed number first
    commitEnt();
    IF ev==30 THEN RETURN 1; END;
  ELSE
    IF (ev==45 OR ev==27) AND gSC(sl(gTSel))==0 THEN
      gEnt:=1;
      gEK:={};
      gENeg:=1;
      RETURN 1;
    END;
  END;
  IF ev==7 THEN gTSel:=IFTE(gTSel==1,n,gTSel-1); END;
  IF ev==8 THEN gTSel:=IFTE(gTSel==n,1,gTSel+1); END;
  IF ev==2 THEN bumpSlot(sl(gTSel),1); END;
  IF ev==12 THEN bumpSlot(sl(gTSel),-1); END;
  IF ev==30 THEN editSlot(sl(gTSel)); END;
  IF ev==3 THEN toolHelp(gTool); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN goHome(); END;
  RETURN 1;
END;

EXPORT GEOMETRY()
BEGIN
  LOCAL run:=1;
  LOCAL ha:=HAngle;
  LOCAL ev;
  HAngle:=1;      // degrees while the program runs
  DIMGROB_P(G1,320,240);
  DIMGROB_P(G9,320,30);
  gEnt:=0;
  WHILE run DO
    IF gScr==1 THEN drawHome(); ELSE drawTool(); END;
    ev:=WAIT(-1);   // key code, touch event list, or -1 after a 1 minute timeout
    IF gScr==1 THEN run:=onHome(ev); ELSE run:=onTool(ev); END;
  END;
  HAngle:=ha;
END;
