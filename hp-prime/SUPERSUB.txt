// ================= number formatting =================

// whole number as text, independent of the calculator's number format
istr(v)
BEGIN
  LOCAL n:=ROUND(ABS(v),0);
  LOCAL s:="";
  IF n<1 THEN RETURN "0"; END;
  WHILE n>=1 DO
    s:=CHAR(48+(n MOD 10))+s;
    n:=IP(n/10);
  END;
  IF v<0 THEN s:="-"+s; END;
  RETURN s;
END;

// one decimal place
fmt1(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*10,0);
  LOCAL s:=istr(IP(n/10))+"."+CHAR(48+(n MOD 10));
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// gauge pressure: psig, or inches of mercury below atmospheric
fmtP(p)
BEGIN
  IF p<0 THEN RETURN fmt1(-p*2.03602)+" inHg"; END;
  RETURN fmt1(p)+" psig";
END;

// v with d decimal places (d = 0 to 4)
fmtN(v,d)
BEGIN
  LOCAL m:=10^d;
  LOCAL n:=ROUND(ABS(v)*m,0);
  LOCAL s:=istr(IP(n/m));
  LOCAL f;
  IF d>0 THEN
    f:=istr(n MOD m);
    WHILE DIM(f)<d DO f:="0"+f; END;
    s:=s+"."+f;
  END;
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// a sensible number of decimals for the size of v
fmtSig(v)
BEGIN
  LOCAL a:=ABS(v);
  IF a<0.00005 THEN RETURN "0"; END;
  IF a>=1000 THEN RETURN fmtN(v,0); END;
  IF a>=100 THEN RETURN fmtN(v,1); END;
  IF a>=1 THEN RETURN fmtN(v,2); END;
  RETURN fmtN(v,4);
END;

// pounds as "1 lb 4.5 oz"
fmtLbOz(lb)
BEGIN
  LOCAL a:=ABS(lb);
  LOCAL w:=IP(a);
  LOCAL oz:=ROUND((a-w)*16,1);
  IF oz>=16 THEN
    w:=w+1;
    oz:=0;
  END;
  RETURN IFTE(lb<0,"-","")+istr(w)+" lb "+fmt1(oz)+" oz";
END;

// ================= saturation tables =================

tabv(r,k,bub)
BEGIN
  IF bub THEN RETURN gBub(r,k)/100; END;
  RETURN gDew(r,k)/100;
END;

// saturation pressure (psig) at temperature t (F); bub=1 bubble, 0 dew
psat(r,t,bub)
BEGIN
  LOCAL x:=(t-gTmin)/gTstep;
  LOCAL k:=MAX(0,MIN(gNpt-2,FLOOR(x)));
  LOCAL l1:=LN(tabv(r,k+1,bub));
  LOCAL l2:=LN(tabv(r,k+2,bub));
  RETURN EXP(l1+(l2-l1)*(x-k))-gAtm;
END;

// saturation temperature (F) at pressure p (psig); bub=1 bubble, 0 dew
tsat(r,p,bub)
BEGIN
  LOCAL la:=LN(MAX(p+gAtm,0.1));
  LOCAL k:=1;
  LOCAL l1, l2;
  WHILE k<gNpt-1 AND LN(tabv(r,k+1,bub))<la DO
    k:=k+1;
  END;
  l1:=LN(tabv(r,k,bub));
  l2:=LN(tabv(r,k+1,bub));
  RETURN gTmin+gTstep*(k-1+(la-l1)/(l2-l1));
END;

// 1 if a saturation temperature lies inside the chart data (with a small
// margin), so readings far off the end of the table are not shown
inRange(t)
BEGIN
  RETURN t>=gTmin-10 AND t<=gTmin+gTstep*(gNpt-1)+5;
END;

// saturated liquid density, lb/ft3, at t F (table at 0, 40, 80, 120 F)
dens(r,t)
BEGIN
  LOCAL x:=MAX(0,MIN(3,t/40));
  LOCAL k:=MIN(2,FLOOR(x));
  RETURN (gDen(r,k+1)+(gDen(r,k+2)-gDen(r,k+1))*(x-k))/100;
END;

// ================= psychrometrics =================
// ASHRAE moist-air relations in I-P units at the local pressure gAtm

// saturation pressure of water vapour, psia, over water or (below 32 F) ice
pws(t)
BEGIN
  LOCAL c:=(t-32)/1.8;
  IF c>=0 THEN RETURN 0.088610*EXP(17.625*c/(c+243.04)); END;
  RETURN 0.088649*EXP(22.587*c/(c+273.86));
END;

// temperature (F) at which water boils at p psia
tboil(p)
BEGIN
  LOCAL l;
  IF p>=0.08865 THEN
    l:=LN(p*51.7149)/LN(10);
    RETURN (1730.63/(8.07131-l)-233.426)*1.8+32;
  END;
  l:=LN(p/0.088649);
  RETURN 273.86*l/(22.587-l)*1.8+32;
END;

// humidity ratio of saturated air at t F, lb/lb
wsat(t)
BEGIN
  LOCAL p:=pws(t);
  RETURN 0.621945*p/(gAtm-p);
END;

// humidity ratio from dry bulb and wet bulb
wdbwb(db,wb)
BEGIN
  LOCAL ws:=wsat(wb);
  IF wb>=32 THEN RETURN ((1093-0.556*wb)*ws-0.240*(db-wb))/(1093+0.444*db-wb); END;
  RETURN ((1220-0.04*wb)*ws-0.240*(db-wb))/(1220+0.444*db-0.48*wb);
END;

// enthalpy, Btu per lb of dry air
enth(db,w)
BEGIN
  RETURN 0.240*db+w*(1061+0.444*db);
END;

// relative humidity, %
rhum(db,w)
BEGIN
  RETURN MIN(100,100*gAtm*w/(0.621945+w)/pws(db));
END;

// ================= gauge geometry =================
// side 1 = low gauge, 2 = high gauge.  A gauge position f runs 0..1
// over a 270 degree sweep, clockwise from lower left.

gcx(side)
BEGIN
  RETURN IFTE(side==1,80,240);
END;

f2a(f)
BEGIN
  RETURN 225-270*f;
END;

// the trig functions follow the calculator's (or app's) angle setting, so
// all dial angles, which are worked out in degrees, are scaled by gK
angInit()
BEGIN
  gK:=ATAN(1)/45;
END;

// sweep position f as an angle in the calculator's unit, for SIN/COS
f2u(f)
BEGIN
  RETURN f2a(f)*gK;
END;

p2f(side,p)
BEGIN
  IF side==2 THEN RETURN MAX(0,MIN(1,p/gHighMax)); END;
  IF p<0 THEN RETURN MAX(0,(p+14.696)/14.696)*gVac; END;
  RETURN MIN(1,gVac+(1-gVac)*p/gLowMax);
END;

f2p(side,f)
BEGIN
  IF side==2 THEN RETURN f*gHighMax; END;
  IF f<gVac THEN RETURN f/gVac*14.696-14.696; END;
  RETURN (f-gVac)/(1-gVac)*gLowMax;
END;

// angle in degrees of the vector (x,y), -180..180
atan2d(y,x)
BEGIN
  IF x>0 THEN RETURN ATAN(y/x)/gK; END;
  IF x<0 THEN
    IF y>=0 THEN RETURN ATAN(y/x)/gK+180; END;
    RETURN ATAN(y/x)/gK-180;
  END;
  IF y>0 THEN RETURN 90; END;
  IF y<0 THEN RETURN -90; END;
  RETURN 0;
END;

// pressure under a finger at screen point (mx,my) on a gauge
touchP(side,mx,my)
BEGIN
  LOCAL s:=225-atan2d(gCY-my,mx-gcx(side));
  LOCAL p;
  WHILE s<0 DO s:=s+360; END;
  WHILE s>=360 DO s:=s-360; END;
  IF s>270 THEN s:=IFTE(s>315,0,270); END;
  p:=f2p(side,s/270);
  IF p<0 THEN RETURN ROUND(p,1); END;
  RETURN ROUND(p,0);
END;

// filled circle, drawn as a triangle fan
fcirc(cx,cy,r,c)
BEGIN
  LOCAL n:=MAX(12,MIN(48,CEILING(r/1.5)));
  LOCAL k, a1, a2;
  FOR k FROM 0 TO n-1 DO
    a1:=k*360/n*gK;
    a2:=(k+1)*360/n*gK;
    TRIANGLE_P(G1,cx,cy,cx+r*COS(a1),cy-r*SIN(a1),cx+r*COS(a2),cy-r*SIN(a2),c);
  END;
END;

// arc of radius rr on a gauge between sweep positions f1 and f2
garc(cx,rr,f1,f2,c)
BEGIN
  LOCAL n:=MAX(1,CEILING(ABS(f2-f1)*64));
  LOCAL a:=f2u(f1);
  LOCAL xa:=cx+rr*COS(a);
  LOCAL ya:=gCY-rr*SIN(a);
  LOCAL k, xb, yb;
  FOR k FROM 1 TO n DO
    a:=f2u(f1+(f2-f1)*k/n);
    xb:=cx+rr*COS(a);
    yb:=gCY-rr*SIN(a);
    LINE_P(G1,xa,ya,xb,yb,c);
    xa:=xb;
    ya:=yb;
  END;
END;

// radial tick from radius r1 to r2 at sweep position f
gtick(cx,f,r1,r2,c)
BEGIN
  LOCAL a:=f2u(f);
  LINE_P(G1,cx+r1*COS(a),gCY-r1*SIN(a),cx+r2*COS(a),gCY-r2*SIN(a),c);
END;

// label centred at radius rr, sweep position f
glabel(cx,f,rr,s,fnt,c)
BEGIN
  LOCAL a:=f2u(f);
  ctext(s,cx+rr*COS(a),gCY-rr*SIN(a)-(4+fnt),fnt,c);
END;

  // saturation temperature ring: dew on the low side, bubble on the high
  garc(cx,gR-25,p2f(side,psat(gRef,gTmin,bub)),p2f(side,psat(gRef,tmax,bub)),tc);
  garc(cx,gR-26,p2f(side,psat(gRef,gTmin,bub)),p2f(side,psat(gRef,tmax,bub)),tc);
  FOR t FROM gTmin TO tmax STEP 10 DO
    p:=psat(gRef,t,bub);
    IF p<=mx AND p>=IFTE(side==1,-14.69,0) THEN
      f:=p2f(side,p);
      gtick(cx,f,gR-25,gR-30,tc);
      IF (t MOD 20)==0 AND f*270-lastA>=24 THEN
        glabel(cx,f,gR-38,istr(t),1,tc);
        lastA:=f*270;
      END;
    END;
  END;

  // face printing
  ctext("psig",cx,gCY+13,1,cDim);
  ctext(IFTE(side==1,"LOW","HIGH"),cx,gCY+34,2,rc);
  ctext(gRName(gRef),cx,gCY+49,1,tc);
END;

needle(side,p)
BEGIN
  LOCAL cx:=gcx(side);
  LOCAL a:=f2u(p2f(side,p));
  LOCAL dx:=COS(a);
  LOCAL dy:=-SIN(a);
  LOCAL lng:=gR-7;
  LOCAL w:=3;
  TRIANGLE_P(G1,cx+lng*dx,gCY+lng*dy,cx-w*dy,gCY+w*dx,cx+w*dy,gCY-w*dx,cInk);
  TRIANGLE_P(G1,cx-14*dx,gCY-14*dy,cx-w*dy,gCY+w*dx,cx+w*dy,gCY-w*dx,cInk);
  fcirc(cx,gCY,6,cInk);
  fcirc(cx,gCY,3,IFTE(side==1,cLow,cHigh));
END;

// colour for a superheat (side 1) or subcooling (side 2) reading
rateC(side,d)
BEGIN
  IF side==1 THEN
    IF d<2 OR d>30 THEN RETURN cBad; END;
    IF d<5 OR d>20 THEN RETURN cWarn; END;
    RETURN cGood;
  END;
  IF d<2 OR d>25 THEN RETURN cBad; END;
  IF d<5 OR d>15 THEN RETURN cWarn; END;
  RETURN cGood;
END;

setSlot(s,v)
BEGIN
  gV(s):=MAX(gSLo(s),MIN(gSHi(s),v));
  IF gSC(s)>0 THEN gV(s):=ROUND(gV(s),0); END;
  IF s==26 THEN gV(28):=1; END;
  IF s==28 THEN gV(28):=MIN(gV(28),SIZE(convUnits(gV(26)))); END;
  IF s==29 THEN setAtm(); END;
  // the gauge readings: write back to the gauge screen
  IF s==32 THEN setVal(1,gV(32)); END;
  IF s==33 THEN setVal(3,gV(33)); END;
  IF s==34 THEN setVal(2,gV(34)); END;
  IF s==35 THEN setVal(4,gV(35)); END;

// Up/Down: step a number, or move through a choice list
bumpSlot(s,d)
BEGIN
  LOCAL n;
  IF gSC(s)>0 THEN
    n:=SIZE(choiceList(gSC(s)));
    gV(s):=gV(s)+IFTE(d>0,1,-1);
    IF gV(s)>n THEN gV(s):=1; END;
    IF gV(s)<1 THEN gV(s):=n; END;
    IF s==26 THEN gV(28):=1; END;
  ELSE
    setSlot(s,gV(s)+d*gSS(s));
  END;
END;

editSlot(s)
BEGIN
  LOCAL v:=gV(s);
  IF gSC(s)>0 THEN
    IF CHOOSE(v,gSL(s),choiceList(gSC(s))) THEN
      IF v>=1 THEN setSlot(s,v); END;
    END;
  ELSE
    IF INPUT(v,gSL(s),gSL(s)+":",slotHelp(s)) THEN
      IF TYPE(v)==0 THEN setSlot(s,v); END;
    END;
  END;
END;

  LOCAL hi:=gTmin+gTstep*(gNpt-1)-(gRows-1)*gPTStep;
  gPTTop:=gTmin+ROUND((gPTTop-gTmin)/gPTStep,0)*gPTStep;

ptFind()
BEGIN
  LOCAL ch:=1;
  LOCAL v, tb, td;
  LOCAL nl:=CHAR(10);
  IF CHOOSE(ch,"Look up","By temperature (°F)","By pressure (psig)")==0 THEN RETURN 0; END;
  IF ch==1 THEN
    v:=40;
    IF INPUT(v,"Look up temperature","°F:","Saturation temperature in °F") THEN
      IF TYPE(v)==0 THEN
        gFindT:=v;
        gFindB:=v;
        MSGBOX(gRName(gRef)+" at "+fmt1(v)+"°F"+nl+"Liquid (bubble): "+fmtP(psat(gRef,v,1))+nl+"Vapor (dew): "+fmtP(psat(gRef,v,0)));
      END;
    END;
  ELSE
    v:=100;
    IF INPUT(v,"Look up pressure","psig:","Gauge pressure. Vacuum: negative psig") THEN
      IF TYPE(v)==0 THEN
        tb:=tsat(gRef,v,1);
        td:=tsat(gRef,v,0);
        IF inRange(tb)==0 OR inRange(td)==0 THEN
          MSGBOX(fmtP(v)+" is off the "+gRName(gRef)+" chart ("+istr(gTmin)+" to "+istr(gTmin+gTstep*(gNpt-1))+"°F)");
          RETURN 0;
        END;
        gFindT:=td;
        gFindB:=tb;
        MSGBOX(gRName(gRef)+" at "+fmtP(v)+nl+"Liquid (bubble): "+fmt1(tb)+"°F"+nl+"Vapor (dew): "+fmt1(td)+"°F"+nl+"Glide: "+fmt1(td-tb)+"°F");
      END;
    END;
  END;
  gPTTop:=gFindT-5*gPTStep;
  ptClamp();
  RETURN 1;
END;

  IF a==-1 THEN RETURN "Overfeeding or low airflow"; END;
  IF b==-1 THEN RETURN "Possibly low charge"; END;
  IF b==1 THEN RETURN "Possibly overcharged"; END;
  RETURN "Charge looks normal";
END;

gaugesOK()
BEGIN
  RETURN inRange(tsat(gRef,gLP,0)) AND inRange(tsat(gRef,gHP,1));
END;

// full check diagnosis from suction (s), head (h), superheat (a) and
// subcooling (b) status, each -1 low, 0 normal, 1 high; ht = heating mode
diagFull(s,h,a,b,ht)
BEGIN
  IF s==0 AND h==0 AND a==0 AND b==0 THEN RETURN "Readings look normal"; END;
  IF s==1 AND h==-1 THEN RETURN "Weak compressor (valves)"; END;
  IF a==1 AND b==-1 THEN RETURN "Low charge: leak check"; END;
  IF a==-1 AND b==1 THEN RETURN "Overcharged"; END;
  IF a==1 AND b==1 THEN RETURN "Restriction: drier/meter"; END;
  IF s==-1 AND a<=0 THEN RETURN IFTE(ht,"Low outdoor air / iced coil","Low indoor airflow"); END;
  IF h==1 AND b<=0 THEN RETURN IFTE(ht,"Low indoor airflow (heat)","Dirty condenser / low air"); END;
  IF h==1 AND b==1 THEN RETURN "Overcharge / noncondensables"; END;
  IF a==1 THEN RETURN "Underfeeding: charge/meter"; END;
  IF a==-1 THEN RETURN "Overfeeding: TXV/charge"; END;
  IF b==-1 THEN RETURN "Possibly low charge"; END;
  IF b==1 THEN RETURN "Possibly overcharged"; END;
  IF h==-1 THEN RETURN "Low head: charge/compressor"; END;
  IF h==1 THEN RETURN "High head: condenser/charge"; END;
  IF s==-1 THEN RETURN "Low suction: airflow/charge"; END;
  IF s==1 THEN RETURN "High suction: comp/charge"; END;
  RETURN "Check pressures vs chart";
END;

// every reading on one screen; heating (heat pump) swaps which air
// stream each coil works against
calcFull()
BEGIN
  LOCAL ht:=gV(31)==2;
  LOCAL te:=tsat(gRef,gLP,0);
  LOCAL tc:=tsat(gRef,gHP,1);
  LOCAL td:=tsat(gRef,gHP,0);
  LOCAL sh:=gST-te;
  LOCAL sc:=tc-gLT;
  LOCAL ds:=gV(7)-td;
  LOCAL dt:=IFTE(gV(31)==2,gV(5)-gV(4),gV(4)-gV(5));
  LOCAL et:=IFTE(gV(31)==2,gV(3),gV(4))-te;
  LOCAL cs:=tc-IFTE(gV(31)==2,gV(4),gV(3));
  LOCAL ap:=gLT-IFTE(gV(31)==2,gV(4),gV(3));
  LOCAL cr:=(gHP+gAtm)/(gLP+gAtm);
  LOCAL ss, hs, s3, s4;
  IF gaugesOK()==0 THEN
    addRow("Gauges","off the chart","check pressures",cBad);
    RETURN 0;
  END;
  // suction status from the evaporator TD, head status from the split
  IF ht THEN
    ss:=-hilo(et,15,30);
    hs:=hilo(cs,20,50);
  ELSE
    ss:=-hilo(et,30,40);
    hs:=hilo(cs,10,30);
  END;
  s3:=hilo(sh,5,20);
  s4:=hilo(sc,5,15);
  addRow("Evap sat",fmt1(te)+"°F","DEW at "+fmtP(gLP),cText);
  addRow("Cond sat",fmt1(tc)+"°F","BUBBLE at "+fmtP(gHP),cText);
  addRow("Superheat",fmt1(sh)+"°F","SLT - dew",rateC(1,sh));
  addRow("Subcool",fmt1(sc)+"°F","bubble - LLT",rateC(2,sc));
  addRow("Disch. SH",fmt1(ds)+"°F","DLT - dew at head",IFTE(gV(7)>225,cBad,IFTE(ds<20,cWarn,cText)));
  addRow("Air ΔT",fmt1(dt)+"°F",IFTE(ht,"supply - return","return - supply"),IFTE(ht,band(dt,15,30),band(dt,16,22)));
  addRow("Evap TD",fmt1(et)+"°F",IFTE(ht,"outdoor - evap sat","return - evap sat"),IFTE(ht,band(et,15,30),band(et,30,40)));
  addRow("Cond split",fmt1(cs)+"°F",IFTE(ht,"cond sat - return","cond sat - outdoor"),IFTE(ht,band(cs,20,50),band(cs,10,30)));
  addRow("Approach",fmt1(ap)+"°F",IFTE(ht,"LLT - return","LLT - outdoor"),cText);
  addRow("Comp. ratio",fmtN(cr,2)+" : 1","psia / psia",IFTE(cr<=4,cGood,IFTE(cr<=6,cWarn,cBad)));
  addRow("Likely","","",cDim);
  addRow("",diagFull(ss,hs,s3,s4,ht),"",IFTE(ss==0 AND hs==0 AND s3==0 AND s4==0,cGood,cWarn));
  RETURN 1;
END;

calcCharge()
BEGIN
  LOCAL fixed:=gV(1)==1;
  LOCAL tol:=gV(30);
  LOCAL te:=tsat(gRef,gLP,0);
  LOCAL tc:=tsat(gRef,gHP,1);
  LOCAL sh:=gST-te;
  LOCAL sc:=tc-gLT;
  LOCAL tsh:=(3*gV(2)-80-gV(3))/2;
  LOCAL ok:=gV(2)>=50 AND gV(2)<=76 AND gV(3)>=55 AND gV(3)<=115 AND tsh>=5;
  LOCAL d, shs, scs;
  IF gaugesOK()==0 THEN
    addRow("Gauges","off the chart","check pressures",cBad);
    RETURN 0;
  END;
  IF fixed THEN
    addRow("Superheat",fmt1(sh)+"°F",IFTE(ok,"target "+fmt1(tsh),"no chart target"),cText);
    addRow("Subcooling",fmt1(sc)+"°F","typ 5-15",cText);
    IF ok THEN
      d:=sh-tsh;
      IF d>tol THEN addRow("Charge","ADD refrigerant","SH "+fmt1(d)+" high",cWarn); END;
      IF d<-tol THEN addRow("Charge","REMOVE refrig.","SH "+fmt1(-d)+" low",cWarn); END;
      IF ABS(d)<=tol THEN addRow("Charge","OK","within ±"+istr(tol)+"°F",cGood); END;
      shs:=hilo(sh,tsh-tol,tsh+tol);
    ELSE
      addRow("Charge","Weigh in charge",IFTE(tsh<5,"target below 5","outside the chart"),cWarn);
      shs:=hilo(sh,5,15);
    END;
    scs:=hilo(sc,5,15);
  ELSE
    addRow("Superheat",fmt1(sh)+"°F","TXV typ 5-15",cText);
    addRow("Subcooling",fmt1(sc)+"°F","target "+fmt1(gV(6)),cText);
    d:=sc-gV(6);
    IF d<-tol THEN addRow("Charge","ADD refrigerant","SC "+fmt1(-d)+" low",cWarn); END;
    IF d>tol THEN addRow("Charge","REMOVE refrig.","SC "+fmt1(d)+" high",cWarn); END;
    IF ABS(d)<=tol THEN addRow("Charge","OK","within ±"+istr(tol)+"°F",cGood); END;
    shs:=hilo(sh,5,15);
    scs:=hilo(sc,gV(6)-tol,gV(6)+tol);
  END;
  addRow("Likely",diagText(shs,scs),"",IFTE(shs==0 AND scs==0,cGood,cWarn));
  addRow("Evap sat",fmt1(te)+"°F","dew at "+fmtP(gLP),cText);
  addRow("Cond sat",fmt1(tc)+"°F","bubble at "+fmtP(gHP),cText);
  RETURN 1;
END;

calcCap()
BEGIN
  LOCAL cfm:=gV(9);
  LOCAL k:=gAtm/14.696;
  LOCAL wr, ws, hr, hs, tot, sen;
  IF gV(2)>gV(4) OR gV(8)>gV(5) THEN
    addRow("Check","wet bulb above dry bulb","",cBad);
    RETURN 0;
  END;
  wr:=wdbwb(gV(4),gV(2));
  ws:=wdbwb(gV(5),gV(8));
  hr:=enth(gV(4),wr);
  hs:=enth(gV(5),ws);
  tot:=4.5*k*cfm*(hr-hs);
  sen:=1.08*k*cfm*(gV(4)-gV(5));
  addRow("Return air",fmtN(rhum(gV(4),wr),0)+"% RH","h "+fmt1(hr)+" BTU/lb",cText);
  addRow("Supply air",fmtN(rhum(gV(5),ws),0)+"% RH","h "+fmt1(hs)+" BTU/lb",cText);
  addRow("Total",fmtN(tot,0)+" BTU/h",fmtN(tot/12000,2)+" tons",cText);
  addRow("Sensible",fmtN(sen,0)+" BTU/h","",cText);
  addRow("Latent",fmtN(tot-sen,0)+" BTU/h","",cText);
  IF tot>0 THEN addRow("SHR",fmtN(sen/tot,2),"typ 0.70-0.80",cText); END;
  addRow("CFM per ton",fmtN(cfm/gV(10),0),"typ 350-450",band(cfm/gV(10),350,450));
  addRow("Of rated",fmtN(tot/120/gV(10),0)+"%","field estimate",cText);
  RETURN 1;
END;

// pressure unit u <-> psia
psiaFrom(u,v)
BEGIN
  IF u==1 THEN RETURN v+gAtm; END;
  IF u==2 THEN RETURN v; END;
  IF u==3 THEN RETURN gAtm-v/2.03602; END;
  IF u==4 THEN RETURN v/6.894757+gAtm; END;
  IF u==5 THEN RETURN v/6.894757; END;
  IF u==6 THEN RETURN v*14.50377+gAtm; END;
  RETURN v/51714.9;
END;

psiaTo(u,p)
BEGIN
  IF u==1 THEN RETURN p-gAtm; END;
  IF u==2 THEN RETURN p; END;
  IF u==3 THEN RETURN (gAtm-p)*2.03602; END;
  IF u==4 THEN RETURN (p-gAtm)*6.894757; END;
  IF u==5 THEN RETURN p*6.894757; END;
  IF u==6 THEN RETURN (p-gAtm)/14.50377; END;
  RETURN p*51714.9;
END;

// temperature unit u <-> F
degFrom(u,v)
BEGIN
  IF u==1 THEN RETURN v; END;
  IF u==2 THEN RETURN v*1.8+32; END;
  IF u==3 THEN RETURN (v-273.15)*1.8+32; END;
  RETURN v-459.67;
END;

degTo(u,f)
BEGIN
  IF u==1 THEN RETURN f; END;
  IF u==2 THEN RETURN (f-32)/1.8; END;
  IF u==3 THEN RETURN (f-32)/1.8+273.15; END;
  RETURN f+459.67;
END;

// size of each unit in the first unit of its list (linear units)
convFactor(ct)
BEGIN
  IF ct==3 THEN RETURN {1,12000,1000,3412.14,3.41214}; END;
  IF ct==4 THEN RETURN {1,2.11888,0.588578,2118.88}; END;
  IF ct==5 THEN RETURN {1,0.0833333,3.28084,0.0328084,0.00328084}; END;
  RETURN {1,0.0625,2.20462,0.00220462};
END;

calcConv()
BEGIN
  LOCAL ct:=gV(26);
  LOCAL u:=convUnits(ct);
  LOCAL n:=SIZE(u);
  LOCAL fr:=MAX(1,MIN(n,gV(28)));
  LOCAL v:=gV(27);
  LOCAL f:=convFactor(ct);
  LOCAL bv, k, ov, txt;
  IF ct==1 THEN bv:=psiaFrom(fr,v); END;
  IF ct==2 THEN bv:=degFrom(fr,v); END;
  IF ct>2 THEN bv:=v*f(fr); END;
  FOR k FROM 1 TO n DO
    IF ct==1 THEN ov:=psiaTo(k,bv); END;
    IF ct==2 THEN ov:=degTo(k,bv); END;
    IF ct>2 THEN ov:=bv/f(k); END;
    txt:=fmtSig(ov);
    IF ct==1 AND k==3 AND ov<0 THEN txt:="above atmosphere"; END;
    addRow(u(k),txt,IFTE(k==fr,"entered",""),IFTE(k==fr,cSelc,cText));
  END;
  RETURN 1;
END;

// returns 0 to quit
onMain(ev)
BEGIN
  LOCAL k, mx, my, side, dx, dy;
  IF TYPE(ev)==6 THEN
    k:=ev(1);
    IF k==2 THEN gDrag:=0; END;
    IF SIZE(ev)<3 THEN RETURN 1; END;
    mx:=ev(2);
    my:=ev(3);
    IF k==0 THEN
      gDrag:=0;
      FOR side FROM 1 TO 2 DO
        dx:=mx-gcx(side);
        dy:=gCY-my;
        IF my>=21 AND my<170 AND dx*dx+dy*dy<=(gR+10)^2 THEN gDrag:=2*side-1; END;
      END;
      IF my>=186 AND my<=201 THEN
        side:=IFTE(mx<160,1,2);
        gWasSel:=IFTE(gSel==2*side AND mx-(side-1)*160<88,1,0);
        gSel:=2*side;
        IF mx-(side-1)*160>=88 THEN gDrag:=2*side; END;
      END;
      IF my>=170 AND my<186 THEN
        gWasSel:=IFTE(gSel==IFTE(mx<160,1,3),1,0);
        gSel:=IFTE(mx<160,1,3);
      END;
    END;
    IF (k==0 OR k==1) AND gDrag>0 THEN
      gSel:=gDrag;
      IF gDrag==1 OR gDrag==3 THEN
        setVal(gDrag,touchP((gDrag+1)/2,mx,my));
      ELSE
        side:=gDrag/2;
        setVal(gDrag,ROUND(slLo(side)+(slHi(side)-slLo(side))*MAX(0,MIN(1,(mx-(side-1)*160-96)/58)),0));
      END;
    END;
    IF k==3 THEN
      IF my>=220 THEN
        k:=IP(mx/64)+1;
        IF k==1 THEN pickRef(); END;
        IF k==2 THEN gScr:=2; END;
        IF k==3 THEN toolsMenu(); END;
        IF k==4 THEN editVal(); END;
        IF k==5 THEN RETURN 0; END;
      END;
      IF my<21 THEN pickRef(); END;
      // a second tap on a reading that was already selected types a value
      IF my>=170 AND my<=201 AND gWasSel==1 THEN editVal(); END;
      gWasSel:=0;
    END;
    RETURN 1;
  END;
  IF ev==2 THEN setVal(gSel,getVal(gSel)+1); END;
  IF ev==12 THEN setVal(gSel,getVal(gSel)-1); END;
  IF ev==50 THEN setVal(gSel,getVal(gSel)+10); END;
  IF ev==45 THEN setVal(gSel,getVal(gSel)-10); END;
  IF ev==7 THEN gSel:=IFTE(gSel==1,4,gSel-1); END;
  IF ev==8 THEN gSel:=IFTE(gSel==4,1,gSel+1); END;
  IF ev==30 THEN editVal(); END;
  IF ev==1 THEN pickRef(); END;
  IF ev==11 THEN gScr:=2; END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==4 THEN RETURN 0; END;
  RETURN 1;
END;

onPT(ev)
BEGIN
  LOCAL k, mx, my, n;
  IF TYPE(ev)==6 THEN
    IF SIZE(ev)<3 THEN RETURN 1; END;
    k:=ev(1);
    mx:=ev(2);
    my:=ev(3);
    IF k==0 THEN gDragY:=my; END;
    IF k==1 AND my>=38 AND my<220 THEN
      n:=IP((gDragY-my)/16);
      IF n<>0 THEN
        ptScroll(n);
        gDragY:=gDragY-n*16;
      END;
    END;
    IF k==3 THEN
      IF my>=220 THEN
        n:=IP(mx/64)+1;
        IF n==1 THEN pickRef(); END;
        IF n==2 THEN
          gPTStep:=IFTE(gPTStep==1,5,1);
          ptClamp();
        END;
        IF n==3 THEN ptFind(); END;
        IF n==4 THEN gScr:=1; END;
        IF n==5 THEN toolsMenu(); END;
      END;
      IF my<21 THEN nextRef(1); END;
    END;
    RETURN 1;
  END;
  IF ev==2 THEN ptScroll(-1); END;
  IF ev==12 THEN ptScroll(1); END;
  IF ev==45 THEN ptScroll(-(gRows-1)); END;
  IF ev==50 THEN ptScroll(gRows-1); END;
  IF ev==7 THEN nextRef(-1); END;
  IF ev==8 THEN nextRef(1); END;
  IF ev==30 THEN ptFind(); END;
  IF ev==1 THEN pickRef(); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 OR ev==6 OR ev==11 THEN gScr:=1; END;
  RETURN 1;
END;

// returns 0 to quit (never: Esc goes back to the gauges)
onTool(ev)
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, mx, my, rw, idx;
  IF TYPE(ev)==6 THEN
    IF SIZE(ev)<3 THEN RETURN 1; END;
    k:=ev(1);
    mx:=ev(2);
    my:=ev(3);
    IF k==3 THEN
      IF my>=220 THEN
        k:=IP(mx/64)+1;
        IF k==1 THEN pickRef(); END;
        IF k==2 AND n>0 THEN editSlot(sl(gTSel)); END;
        IF k==3 THEN toolsMenu(); END;
        IF k==4 THEN gScr:=1; END;
        IF k==5 THEN toolHelp(gTool); END;
        RETURN 1;
      END;
      IF my<21 THEN
        IF mx>=206 AND toolRef(gTool) THEN pickRef(); END;
        RETURN 1;
      END;
      rw:=IP((my-24)/20);
      idx:=rw*2+IP(mx/160)+1;
      IF gTool==1 THEN
        rw:=IP((my-24)/21);
        idx:=IFTE(mx<160,rw+1,0);
      END;
      IF my>=24 AND idx>=1 AND idx<=n AND (gTool==1 OR rw<IP((n+1)/2)) THEN
        gTSel:=idx;
        editSlot(sl(idx));
      END;
    END;
    RETURN 1;
  END;
  IF n>0 THEN
    IF gTSel>n THEN gTSel:=1; END;
    IF ev==7 THEN gTSel:=IFTE(gTSel==1,n,gTSel-1); END;
    IF ev==8 THEN gTSel:=IFTE(gTSel==n,1,gTSel+1); END;
    IF ev==2 THEN bumpSlot(sl(gTSel),1); END;
    IF ev==12 THEN bumpSlot(sl(gTSel),-1); END;
    IF ev==50 THEN bumpSlot(sl(gTSel),10); END;
    IF ev==45 THEN bumpSlot(sl(gTSel),-10); END;
    IF ev==30 THEN editSlot(sl(gTSel)); END;
  END;
  IF ev==1 THEN pickRef(); END;
  IF ev==11 THEN gScr:=2; END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==3 THEN toolHelp(gTool); END;
  IF ev==4 OR ev==6 THEN gScr:=1; END;
  RETURN 1;
END;

// ================= main =================

EXPORT SUPERSUB()
BEGIN
  LOCAL run:=1;
  LOCAL ev;
  angInit();
  setAtm();
  DIMGROB_P(G1,320,240);
  DIMGROB_P(G2,320,240);
  DIMGROB_P(G9,320,30);
  gBgRef:=0;
  gDrag:=0;
  WHILE run DO
    IF gScr==1 THEN drawMain(); END;
    IF gScr==2 THEN drawPT(); END;
    IF gScr==3 THEN drawTool(); END;
    ev:=WAIT(-1);   // key code, touch event list, or -1 after a 1 minute timeout
    IF gScr==1 THEN
      run:=onMain(ev);
    ELSE
      IF gScr==2 THEN run:=onPT(ev); ELSE run:=onTool(ev); END;
    END;
  END;
END;
