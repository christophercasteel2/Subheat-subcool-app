// ================= number formatting =================

// whole number as text, independent of the calculator's number format
istr(v)
BEGIN
  LOCAL n:=ROUND(ABS(v),0);
  LOCAL s:="";
  IF n<1 THEN RETURN "0"; END;
  WHILE n>=1 DO
    s:=CHAR(48+(n MOD 10))+s;
    n:=IP(n/10);
  END;
  IF v<0 THEN s:="-"+s; END;
  RETURN s;
END;

// one decimal place
fmt1(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*10,0);
  LOCAL s:=istr(IP(n/10))+"."+CHAR(48+(n MOD 10));
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// v with d decimal places (d = 0 to 5)
fmtN(v,d)
BEGIN
  LOCAL m:=10^d;
  LOCAL n:=ROUND(ABS(v)*m,0);
  LOCAL s:=istr(IP(n/m));
  LOCAL f;
  IF d>0 THEN
    f:=istr(n MOD m);
    WHILE DIM(f)<d DO f:="0"+f; END;
    s:=s+"."+f;
  END;
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// up to d decimals, trailing zeros dropped
fmtTr(v,d)
BEGIN
  LOCAL s:=fmtN(v,d);
  IF d>0 THEN
    WHILE RIGHT(s,1)=="0" DO s:=LEFT(s,DIM(s)-1); END;
    IF RIGHT(s,1)=="." THEN s:=LEFT(s,DIM(s)-1); END;
  END;
  RETURN s;
END;

// any size of number; very large or small ones as 1.23E-6
fmtE(v)
BEGIN
  LOCAL a:=ABS(v);
  LOCAL ex, m;
  IF a<0.000000000001 THEN RETURN "0"; END;
  IF a>=0.001 AND a<10000000 THEN
    // six significant figures
    ex:=5;
    IF a>=10 THEN ex:=4; END;
    IF a>=100 THEN ex:=3; END;
    IF a>=1000 THEN ex:=2; END;
    IF a>=10000 THEN ex:=1; END;
    IF a>=100000 THEN ex:=0; END;
    RETURN fmtTr(v,ex);
  END;
  ex:=FLOOR(LN(a)/LN(10));
  m:=a/10^ex;
  IF m>=9.9995 THEN
    m:=m/10;
    ex:=ex+1;
  END;
  IF m<1 THEN
    m:=m*10;
    ex:=ex-1;
  END;
  RETURN IFTE(v<0,"-","")+fmtTr(m,3)+"E"+istr(ex);
END;

// most wires of area a1 that fit in raceway area ar (Chapter 9 Table 1
// and Note 7), with n2 more wires of area a2 already in it
maxFit(ar,a1,a2,n2)
BEGIN
  LOCAL x;
  IF n2>0 THEN RETURN MAX(0,IP((0.4*ar-n2*a2)/a1+0.0000001)); END;
  x:=0.4*ar/a1;
  IF x>=2.8 THEN
    IF x-IP(x)>=0.8 THEN RETURN IP(x)+1; END;
    RETURN IP(x);
  END;
  IF 2*a1<=0.31*ar THEN RETURN 2; END;
  IF a1<=0.53*ar THEN RETURN 1; END;
  RETURN 0;
END;

// smallest size of material m with a Table 310.16 column c ampacity of at
// least a that a breaker of b amps may protect (240.4(D)); 0 = none
wireFor(a,b,m,c)
BEGIN
  LOCAL k, so;
  FOR k FROM IFTE(m==2,4,3) TO SIZE(gCM) DO
    so:=smallOC(k,m);
    IF ampT(1,k,m,c)>=a-0.0001 AND (so==0 OR so>=b-0.0001) THEN RETURN k; END;
  END;
  RETURN 0;
END;

// wireFor as text, with parallel sets (1/0 and up, 310.10(G)) when one
// conductor is not enough

BEGIN
  LOCAL k:=wireFor(a,b,m,c);

  IF k>0 THEN RETURN sizeName(k)+" ("+istr(ampT(1,k,m,c))+" A)"; END;
  FOR n FROM 2 TO 8 DO
    FOR j FROM 12 TO SIZE(gCM) DO
      IF ampT(1,j,m,c)*n>=a-0.0001 THEN RETURN istr(n)+" sets of "+sizeName(j); END;
    END;
  END;
  RETURN "-";
END;

BEGIN

setSlot(s,v)
BEGIN
  gV(s):=MAX(gSLo(s),MIN(gSHi(s),v));
  IF gSC(s)>0 THEN gV(s):=ROUND(gV(s),0); END;
END;

// Up/Down: step a number, or move through a choice list
bumpSlot(s,d)
BEGIN
  LOCAL n;
  IF gSC(s)>0 THEN
    n:=SIZE(choiceList(gSC(s)));
    gV(s):=gV(s)+IFTE(d>0,1,-1);
    IF gV(s)>n THEN gV(s):=1; END;
    IF gV(s)<1 THEN gV(s):=n; END;
  ELSE
    setSlot(s,gV(s)+d*gSS(s));
  END;
END;

// a two-way choice flips, a longer list opens, a number opens a dialog
editSlot(s)
BEGIN
  LOCAL v:=gV(s);
  LOCAL lst;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    IF SIZE(lst)==2 THEN
      setSlot(s,IFTE(v==1,2,1));
    ELSE
      IF CHOOSE(v,slotName(s),lst) THEN
        IF v>=1 THEN setSlot(s,v); END;
      END;
    END;
  ELSE
    IF INPUT(v,slotName(s),slotName(s)+":",gSH(s)) THEN
      IF TYPE(v)==0 THEN setSlot(s,v); END;
    END;
  END;
  gDown:=0;
END;

    gEK:={};
    gENeg:=0;
  END;
  FOR k FROM 1 TO SIZE(gEK) DO
    IF gEK(k)==10 THEN hasdot:=1; END;
  END;
  IF (d<10 OR hasdot==0) AND SIZE(gEK)<12 THEN gEK(SIZE(gEK)+1):=d; END;
END;

backKey()
BEGIN
  LOCAL nl:={};
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK)-1 DO
    nl(k):=gEK(k);
  END;
  gEK:=nl;
  IF SIZE(gEK)==0 THEN gEnt:=0; END;
END;

calcVD()
BEGIN
  LOCAL a:=gV(2);
  LOCAL lf:=gV(3);
  LOCAL k:=gV(4);
  LOCAL m:=gV(5);
  LOCAL mul:=IFTE(gV(6)==2,gRt3,2);
  LOCAL vs:=gV(7);
  LOCAL mx:=gV(9);
  LOCAL kk:=IFTE(m==2,21.2,12.9);
  LOCAL r, vd, j, best;
  IF m==2 AND k<=3 THEN
    addRow("Aluminum","starts at 12 AWG","pick a larger size",cWarn);
    RETURN 0;
  END;
  r:=IFTE(gV(8)==1,kk/gCM(k),gR8(m,k)/1000);
  vd:=mul*a*lf*r;
  addRow("Voltage drop",fmtN(vd,2)+" V",IFTE(mul>2,"1.732","2")+" x I x L x R",cSelc);
  addRow("Drop",fmtN(100*vd/vs,2)+"%",IFTE(100*vd/vs<=mx,"OK, under ","over ")+fmtTr(mx,1)+"%",IFTE(100*vd/vs<=mx,cGood,cBad));
  addRow("At the load",fmtN(vs-vd,1)+" V","of "+fmtTr(vs,1)+" V",cText);
  addRow("R per 1000 ft",fmtE(1000*r)+" Ω",IFTE(gV(8)==1,"K / CM = "+fmt1(kk)+" / "+istr(gCM(k)),"Table 8, 75°C"),cText);
  best:=0;
  FOR j FROM IFTE(m==2,4,1) TO SIZE(gCM) DO
    IF best==0 AND 100*mul*a*lf*IFTE(gV(8)==1,kk/gCM(j),gR8(m,j)/1000)/vs<=mx THEN best:=j; END;
  END;
  addRow("Smallest OK",sizeName(best),"for "+fmtTr(mx,1)+"% drop",cText);
  IF a>0 THEN addRow("Max length",fmtN(mx/100*vs/(mul*a*r),0)+" ft","one way, this wire",cText); END;
  addRow("Circular mils",istr(gCM(k))+" cmil",sizeName(k),cDim);
  RETURN 1;
END;

calcWire()
BEGIN
  LOCAL ld:=gV(10)*IFTE(gV(11)==2,1.25,1);
  LOCAL m:=gV(12);
  LOCAL wc:=gV(13);
  LOCAL tcol:=MIN(gV(14),gV(13));
  LOCAL cf:=corrF(ambC(gV(15)),wc);
  LOCAL lf:=gV(16);
  LOCAL sy:=gV(17);
  LOCAL ki:=0;
  LOCAL kt:=0;
  LOCAL k, sz, so, k3, k5, nt;
  addRow("Must carry",fmtN(ld,1)+" A",IFTE(gV(11)==2,"125% of "+fmtTr(gV(10),1)+" A","the load"),cText);
  IF cf<=0 THEN
    addRow("Ambient","too hot","for "+colName(wc)+" wire",cBad);
    RETURN 0;
  END;
  FOR k FROM IFTE(m==2,4,3) TO SIZE(gCM) DO
    IF ki==0 AND ampT(1,k,m,wc)>0 AND ampT(1,k,m,wc)*cf>=ld THEN ki:=k; END;
    IF kt==0 AND ampT(1,k,m,tcol)>0 AND ampT(1,k,m,tcol)>=ld THEN kt:=k; END;
  END;
  IF ki==0 OR kt==0 THEN
    addRow("Wire","over 2000 kcmil","use parallel sets",cBad);
    RETURN 0;
  END;
  addRow("By insulation",sizeName(ki),istr(ampT(1,ki,m,wc))+" x "+fmtN(cf,2)+" = "+fmt1(ampT(1,ki,m,wc)*cf)+" A",IFTE(cf<>1,cWarn,cText));
  addRow("By terminals",sizeName(kt),istr(ampT(1,kt,m,tcol))+" A, "+colName(tcol)+" column",cText);
  sz:=MAX(ki,kt);
  nt:="Table 310.16";
  IF lf>0 THEN
    k3:=vdSize(m,gV(10),lf,gSysV(sy),sysMul(sy),3);
    k5:=vdSize(m,gV(10),lf,gSysV(sy),sysMul(sy),5);
    IF k3==0 THEN
      addRow("3% drop","over 2000 kcmil",istr(lf)+" ft is too long",cBad);
      nt:="drop still over 3%";
    ELSE
      addRow("3% drop",sizeName(k3),fmtN(vdPct(k3,m,gV(10),lf,gSysV(sy),sysMul(sy)),2)+"% at "+istr(lf)+" ft",cText);
      IF k3>sz THEN
        sz:=k3;
        nt:="sized up for 3% drop";
      END;
    END;
    IF k5>0 THEN addRow("5% drop",sizeName(k5),fmtN(vdPct(k5,m,gV(10),lf,gSysV(sy),sysMul(sy)),2)+"%, feeder + branch",cText); END;
  END;
  // the breaker for this load must be allowed on the wire (240.4(D))
  so:=smallOC(sz,m);
  WHILE so>0 AND so<stdUp(ld) DO
    sz:=sz+1;
    so:=smallOC(sz,m);
    nt:="240.4(D): "+istr(stdUp(ld))+" A breaker";
  END;
  addRow("Use",sizeName(sz)+IFTE(m==2," Al"," Cu"),nt,cSelc);
  IF so>0 THEN addRow("Breaker max",istr(so)+" A","240.4(D) small wire",cWarn); END;
  RETURN 1;
END;

calcAmp()
BEGIN
  LOCAL k:=gV(18);
  LOCAL m:=gV(19);
  LOCAL it:=gV(20);
  LOCAL tcl:=gInsCol(it);
  LOCAL cu:=MIN(3,tcl);
  LOCAL fa:=gV(21)==2;
  LOCAL tb:=ampT(IFTE(fa,2,1),k,m,cu);
  LOCAL inl:=choiceList(11);
  LOCAL cf, af, am, tl, so;
  addRow("Insulation",inl(it)+": "+IFTE(tcl==4,"60°C",colName(cu)),IFTE(tcl==4,"90°C wire, 334.80","column used"),cText);
  IF tb<=0 THEN
    addRow("Table","not listed",sizeName(k)+" "+IFTE(m==2,"aluminum","copper"),cWarn);
    RETURN 0;
  END;
  cf:=corrF(ambC(gV(22)),cu);
  af:=IFTE(fa,1,adjF(gV(23)));
  am:=tb*cf*af;
  IF tcl==4 THEN am:=MIN(am,ampT(1,k,m,1)); END;
  addRow(IFTE(fa,"Table 310.17","Table 310.16"),istr(tb)+" A",sizeName(k)+" "+IFTE(m==2,"Al","Cu")+IFTE(fa,", free air",""),cSelc);
  addRow("Temp factor",fmtN(cf,2)+" x",ambTxt(gV(22)),IFTE(cf<>1,cWarn,cText));
  addRow("Bundling",fmtN(af,2)+" x",IFTE(fa,"free air: none",istr(gV(23))+" current-carrying"),IFTE(af<1,cWarn,cText));
  IF gV(24)>1 THEN
    tl:=ampT(IFTE(fa,2,1),k,m,gV(24)-1);
    addRow("Terminals",istr(tl)+" A",colName(gV(24)-1)+" column limit",cText);
    am:=MIN(am,tl);
  END;
  addRow("Ampacity",fmtTr(am,1)+" A","allowed",cSelc);
  so:=smallOC(k,m);
  IF so>0 THEN addRow("Breaker max",istr(so)+" A","240.4(D)",cText); END;
  RETURN 1;
END;

calcBrk()
BEGIN
  LOCAL c:=gV(25);
  LOCAL nc:=gV(26);
  LOCAL need:=1.25*c+nc;
  LOCAL k:=gV(27);
  LOCAL m:=gV(28);
  LOCAL am:=ampT(1,k,m,gV(29));
  LOCAL so:=smallOC(k,m);
  LOCAL mn:=stdUp(need);
  LOCAL mxb;

calcPow()
BEGIN
  LOCAL v:=gV(65);
  LOCAL a:=gV(66);
  LOCAL r:=gV(67);
  LOCAL p:=gV(68);
  LOCAL pf:=gV(69);
  LOCAL kk:=IFTE(gV(70)==2,gRt3,1);
  LOCAL n:=(v>0)+(a>0)+(r>0)+(p>0);
  LOCAL md:=0;
  LOCAL pr:={"V and I","V and R","V and P","I and R","I and P","R and P"};
  LOCAL va, vr, gv, ga, gr;
  IF n<2 THEN
    addRow("Enter","any two values","leave the rest 0",cWarn);
    RETURN 0;
  END;
  IF v>0 AND a>0 THEN md:=1; END;
  IF md==0 AND v>0 AND r>0 THEN md:=2; END;
  IF md==0 AND v>0 AND p>0 THEN md:=3; END;
  IF md==0 AND a>0 AND r>0 THEN md:=4; END;
  IF md==0 AND a>0 AND p>0 THEN md:=5; END;
  IF md==0 THEN md:=6; END;
  IF md==2 THEN a:=v/r; END;
  IF md==3 THEN a:=p/(kk*v*pf); END;
  IF md==4 THEN v:=a*r; END;
  IF md==5 THEN v:=p/(kk*a*pf); END;
  IF md==6 THEN
    a:=sqroot(p/(kk*r*pf));
    v:=a*r;
  END;
  r:=v/a;
  p:=kk*v*a*pf;
  va:=kk*v*a;
  vr:=sqroot(va*va-p*p);
  gv:=md<=3;
  ga:=md==1 OR md==4 OR md==5;
  gr:=md==2 OR md==4 OR md==6;
  addRow("Volts",fmtE(v)+" V",IFTE(gv,"entered","worked out"),IFTE(gv,cText,cSelc));
  addRow("Amps",fmtE(a)+" A",IFTE(ga,"entered","worked out"),IFTE(ga,cText,cSelc));
  addRow(IFTE(pf<1,"Impedance","Ohms"),fmtE(r)+" Ω",IFTE(gr,"entered","V / I"),IFTE(gr,cText,cSelc));
  addRow("Real power",fmtE(p)+" W",fmtE(p/1000)+" kW",IFTE(md>=5 OR md==3,cText,cSelc));
  addRow("Apparent",fmtE(va)+" VA",fmtE(va/1000)+" kVA",cSelc);
  addRow("Reactive",fmtE(vr)+" VAR","√(VA² - W²)",cText);
  addRow("Power factor",fmtN(pf,2),IFTE(kk>1,"3-phase: x 1.732","W / VA"),cText);
  IF n>2 THEN addRow("Note","used "+pr(md),"set the others to 0",cWarn); END;
  RETURN 1;
END;

calcOhm()
BEGIN
  LOCAL v:=gV(71);
  LOCAL a:=gV(72);
  LOCAL r:=gV(73);
  LOCAL p:=gV(74);
  LOCAL n:=(v>0)+(a>0)+(r>0)+(p>0);
  LOCAL md:=0;
  LOCAL fm:={"R = E/I   P = E x I","I = E/R   P = E²/R","I = P/E   R = E²/P",
    "E = I x R   P = I² x R","E = P/I   R = P/I²","E = √(P x R)   I = √(P/R)"};
  LOCAL pr:={"E and I","E and R","E and P","I and R","I and P","R and P"};
  LOCAL gv, ga, gr, gp;
  IF n<2 THEN
    addRow("Enter","any two values","leave the rest 0",cWarn);
    RETURN 0;
  END;
  IF v>0 AND a>0 THEN md:=1; END;
  IF md==0 AND v>0 AND r>0 THEN md:=2; END;
  IF md==0 AND v>0 AND p>0 THEN md:=3; END;
  IF md==0 AND a>0 AND r>0 THEN md:=4; END;
  IF md==0 AND a>0 AND p>0 THEN md:=5; END;
  IF md==0 THEN md:=6; END;
  IF md==1 THEN
    r:=v/a;
    p:=v*a;
  END;
  IF md==2 THEN
    a:=v/r;
    p:=v*v/r;
  END;
  IF md==3 THEN
    a:=p/v;
    r:=v*v/p;
  END;
  IF md==4 THEN
    v:=a*r;
    p:=a*a*r;
  END;
  IF md==5 THEN
    v:=p/a;
    r:=p/(a*a);
  END;
  IF md==6 THEN
    v:=sqroot(p*r);
    a:=sqroot(p/r);
  END;
  gv:=md<=3;
  ga:=md==1 OR md==4 OR md==5;
  gr:=md==2 OR md==4 OR md==6;
  gp:=md==3 OR md==5 OR md==6;
  addRow("Volts (E)",fmtE(v)+" V",IFTE(gv,"entered","worked out"),IFTE(gv,cText,cSelc));
  addRow("Amps (I)",fmtE(a)+" A",IFTE(ga,"entered","worked out"),IFTE(ga,cText,cSelc));
  addRow("Ohms (R)",fmtE(r)+" Ω",IFTE(gr,"entered","worked out"),IFTE(gr,cText,cSelc));
  addRow("Watts (P)",fmtE(p)+" W",IFTE(gp,"entered","worked out"),IFTE(gp,cText,cSelc));
  addRow("Formula",fm(md),"from "+pr(md),cDim);
  IF n>2 THEN addRow("Note","used "+pr(md),"set the others to 0",cWarn); END;
  RETURN 1;
END;

drawTool()
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, s, x0, y, txt, big, fv;
  toolCalc(gTool);
  RECT_P(G1,0,0,319,239,cBg);
  RECT_P(G1,0,0,319,20,gCatC(gTCat(gTool)));
  TEXTOUT_P(gTName(gTool),G1,5,3,3,cWhite);
  chip("HELP",272);
  // input boxes, two per row
  FOR k FROM 1 TO n DO
    s:=sl(k);
    x0:=((k-1) MOD 2)*160;
    y:=24+IP((k-1)/2)*20;
    TEXTOUT_P(slotName(s),G1,x0+4,y+4,1,cDim);
    IF k==gTSel AND gEnt THEN
      RECT_P(G1,x0+74,y,x0+157,y+17,cSelc,cHit);
      txt:=entText()+"_";
    ELSE
      RECT_P(G1,x0+74,y,x0+157,y+17,IFTE(k==gTSel,cSelc,cEdge),cPanel);
      txt:=slotText(s);
    END;
    big:=DIM(txt)<=11;
    TEXTOUT_P(txt,G1,x0+78,y+IFTE(big,2,4),IFTE(big,2,1),cText);
  END;
  // results
  y:=24+IP((n+1)/2)*20+3;
  LINE_P(G1,0,y-2,319,y-2,cEdge);

    IF (k MOD 2)==0 THEN RECT_P(G1,0,y,319,y+14,cRow); END;
    TEXTOUT_P(gRL(k),G1,5,y+2,1,cDim);

    y:=y+15;

  drawMenu({"EDIT","RESET","< PREV","NEXT >","HOME"});
  BLIT_P(G0,G1);
END;

openTool(t)
BEGIN
  gTool:=t;
  gTSel:=1;
  gEnt:=0;
  gScr:=2;
END;

goHome()
BEGIN
  gHome:=gTool;
  gEnt:=0;
  gScr:=1;
END;

toolsMenu()
BEGIN
  LOCAL t:=gTool;
  IF CHOOSE(t,"Tools",gTName) THEN
    IF t>=1 THEN openTool(t); END;
  END;
  gDown:=0;
END;

// RESET: the tool's example values
resetTool()
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL k;
  FOR k FROM 1 TO SIZE(sl) DO
    gV(sl(k)):=gDef(sl(k));
  END;
  gEnt:=0;
END;

// open tile k: a tool, or the Learn pages
openTile(k)
BEGIN
  gHome:=k;
  IF k>gNT THEN
    learnMenu();
  ELSE
    openTool(k);
  END;
END;

// ================= input handling =================

// returns 0 to quit
onHome(ev)
BEGIN
  LOCAL mx, my, k;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    IF my<21 THEN
      IF mx>=272 THEN RETURN 0; END;
      IF mx>=224 THEN showHelp(); END;
      RETURN 1;
    END;
    k:=IP((my-23)/43)*4+MIN(3,IP(mx/80))+1;
    IF my>=23 AND k>=1 AND k<=20 THEN openTile(k); END;
    RETURN 1;
  END;
  IF ev==7 THEN gHome:=IFTE(gHome==1,20,gHome-1); END;
  IF ev==8 THEN gHome:=IFTE(gHome==20,1,gHome+1); END;
  IF ev==2 THEN gHome:=IFTE(gHome<=4,gHome+16,gHome-4); END;
  IF ev==12 THEN gHome:=IFTE(gHome>16,gHome-16,gHome+4); END;
  IF ev==30 THEN openTile(gHome); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN RETURN 0; END;
  RETURN 1;
END;

onTool(ev)
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, mx, my, rw, idx, dg;
  IF gTSel>n THEN gTSel:=1; END;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    commitEnt();
    IF my>=220 THEN
      k:=IP(mx/64)+1;
      IF k==1 THEN editSlot(sl(gTSel)); END;
      IF k==2 THEN resetTool(); END;
      IF k==3 THEN openTool(IFTE(gTool==1,gNT,gTool-1)); END;
      IF k==4 THEN openTool(IFTE(gTool==gNT,1,gTool+1)); END;
      IF k==5 THEN goHome(); END;
      RETURN 1;
    END;
    IF my<21 THEN
      IF mx>=272 THEN toolHelp(gTool); END;
      RETURN 1;
    END;
    // tap an input box to change it
    rw:=IP((my-24)/20);
    idx:=rw*2+IP(mx/160)+1;
    IF my>=24 AND rw<IP((n+1)/2) AND idx<=n THEN
      gTSel:=idx;
      editSlot(sl(idx));
    END;
    RETURN 1;
  END;
  // digits and the decimal point type straight into a number box
  dg:=keyDigit(ev);
  IF dg>=0 THEN
    IF gSC(sl(gTSel))==0 THEN typeKey(dg); END;
    RETURN 1;
  END;
  IF gEnt THEN
    IF ev==45 OR ev==27 THEN
      gENeg:=1-gENeg;
      RETURN 1;
    END;
    IF ev==19 THEN
      backKey();
      RETURN 1;
    END;
    IF ev==4 THEN
      gEnt:=0;
      RETURN 1;
    END;
    // any other key keeps the typed number first
    commitEnt();
    IF ev==30 THEN RETURN 1; END;
  ELSE
    IF (ev==45 OR ev==27) AND gSC(sl(gTSel))==0 THEN
      gEnt:=1;
      gEK:={};
      gENeg:=1;
      RETURN 1;
    END;
  END;
  IF ev==7 THEN gTSel:=IFTE(gTSel==1,n,gTSel-1); END;
  IF ev==8 THEN gTSel:=IFTE(gTSel==n,1,gTSel+1); END;
  IF ev==2 THEN bumpSlot(sl(gTSel),1); END;
  IF ev==12 THEN bumpSlot(sl(gTSel),-1); END;
  IF ev==30 THEN editSlot(sl(gTSel)); END;
  IF ev==3 THEN toolHelp(gTool); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN goHome(); END;
  RETURN 1;
END;

EXPORT NECCALC()
BEGIN
  LOCAL run:=1;
  LOCAL ev;
  DIMGROB_P(G1,320,240);
  DIMGROB_P(G9,320,30);
  gEnt:=0;
  WHILE run DO
    IF gScr==1 THEN drawHome(); ELSE drawTool(); END;
    ev:=WAIT(-1);   // key code, touch event list, or -1 after a 1 minute timeout
    IF gScr==1 THEN run:=onHome(ev); ELSE run:=onTool(ev); END;
  END;
END;
