// ================= number formatting =================

// whole number as text, independent of the calculator's number format
istr(v)
BEGIN
  LOCAL n:=ROUND(ABS(v),0);
  LOCAL s:="";
  IF n<1 THEN RETURN "0"; END;
  WHILE n>=1 DO
    s:=CHAR(48+(n MOD 10))+s;
    n:=IP(n/10);
  END;
  IF v<0 THEN s:="-"+s; END;
  RETURN s;
END;

// one decimal place
fmt1(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*10,0);
  LOCAL s:=istr(IP(n/10))+"."+CHAR(48+(n MOD 10));
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// v with d decimal places (d = 0 to 5)
fmtN(v,d)
BEGIN
  LOCAL m:=10^d;
  LOCAL n:=ROUND(ABS(v)*m,0);
  LOCAL s:=istr(IP(n/m));
  LOCAL f;
  IF d>0 THEN
    f:=istr(n MOD m);
    WHILE DIM(f)<d DO f:="0"+f; END;
    s:=s+"."+f;
  END;
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// up to d decimals, trailing zeros dropped
fmtTr(v,d)
BEGIN
  LOCAL s:=fmtN(v,d);
  IF d>0 THEN
    WHILE RIGHT(s,1)=="0" DO s:=LEFT(s,DIM(s)-1); END;
    IF RIGHT(s,1)=="." THEN s:=LEFT(s,DIM(s)-1); END;
  END;
  RETURN s;
END;

// any size of number; very large or small ones as 1.23E-6
fmtE(v)
BEGIN
  LOCAL a:=ABS(v);
  LOCAL ex, m;
  IF a<0.000000000001 THEN RETURN "0"; END;
  IF a>=0.001 AND a<10000000 THEN
    // six significant figures
    ex:=5;
    IF a>=10 THEN ex:=4; END;
    IF a>=100 THEN ex:=3; END;
    IF a>=1000 THEN ex:=2; END;
    IF a>=10000 THEN ex:=1; END;
    IF a>=100000 THEN ex:=0; END;
    RETURN fmtTr(v,ex);
  END;
  ex:=FLOOR(LN(a)/LN(10));
  m:=a/10^ex;
  IF m>=9.9995 THEN
    m:=m/10;
    ex:=ex+1;
  END;
  IF m<1 THEN
    m:=m*10;
    ex:=ex-1;
  END;
  RETURN IFTE(v<0,"-","")+fmtTr(m,3)+"E"+istr(ex);
END;

// ampacity from Table 310.16 (tb 1) or 310.17 (tb 2): size k, material m
// (1 Cu, 2 Al), column c (1 60C, 2 75C, 3 90C); 0 = not listed
ampT(tb,k,m,c)
BEGIN
  IF tb==2 THEN RETURN gA17((m-1)*3+c,k); END;
  RETURN gA16((m-1)*3+c,k);
END;

BEGIN
  LOCAL t:=ambC(x);

END;

// Table 310.15(B)(1) correction factor at t C for column c (0 = too hot)
corrF(t,c)
BEGIN
  LOCAL k;
  FOR k FROM 1 TO SIZE(gAmbTop) DO
    IF t<=gAmbTop(k)+0.0001 THEN RETURN gCorr(c,k); END;
  END;
  RETURN 0;
END;

// Table 310.15(C)(1) adjustment for n current-carrying conductors
adjF(n)
BEGIN
  IF n<=3 THEN RETURN 1; END;
  IF n<=6 THEN RETURN 0.8; END;
  IF n<=9 THEN RETURN 0.7; END;
  IF n<=20 THEN RETURN 0.5; END;
  IF n<=30 THEN RETURN 0.45; END;
  IF n<=40 THEN RETURN 0.4; END;
  RETURN 0.35;
END;

END;

// most wires of area a1 that fit in raceway area ar (Chapter 9 Table 1
// and Note 7), with n2 more wires of area a2 already in it
maxFit(ar,a1,a2,n2)
BEGIN
  LOCAL x;
  IF n2>0 THEN RETURN MAX(0,IP((0.4*ar-n2*a2)/a1+0.0000001)); END;
  x:=0.4*ar/a1;
  IF x>=2.8 THEN
    IF x-IP(x)>=0.8 THEN RETURN IP(x)+1; END;
    RETURN IP(x);
  END;
  IF 2*a1<=0.31*ar THEN RETURN 2; END;
  IF a1<=0.53*ar THEN RETURN 1; END;
  RETURN 0;
END;

// ================= input boxes =================

// the conduit trade size box is only used when finding wires that fit
slotName(s)
BEGIN
  RETURN gSL(s);
END;

slotUnit(s)
BEGIN
  IF s==15 OR s==21 THEN RETURN IFTE(gV(1)==2,"°F","°C"); END;
  RETURN gSU(s);
END;

// text shown in an input box
slotText(s)
BEGIN
  LOCAL u:=slotUnit(s);
  LOCAL v:=gV(s);
  LOCAL lst, t;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    RETURN lst(MAX(1,MIN(SIZE(lst),v)));
  END;
  IF v==0 AND gSZ(s)==1 THEN RETURN "--"; END;
  IF v==0 AND gSZ(s)==2 THEN RETURN "?"; END;
  IF gSD(s)<0 THEN t:=fmtTr(v,4); ELSE t:=fmtN(v,gSD(s)); END;
  IF u=="" THEN RETURN t; END;
  IF LEFT(u,1)=="°" THEN RETURN t+u; END;
  RETURN t+" "+u;
END;

setSlot(s,v)
BEGIN
  gV(s):=MAX(gSLo(s),MIN(gSHi(s),v));
  IF gSC(s)>0 THEN gV(s):=ROUND(gV(s),0); END;
END;

// Up/Down: step a number, or move through a choice list
bumpSlot(s,d)
BEGIN
  LOCAL n;
  IF gSC(s)>0 THEN
    n:=SIZE(choiceList(gSC(s)));
    gV(s):=gV(s)+IFTE(d>0,1,-1);
    IF gV(s)>n THEN gV(s):=1; END;
    IF gV(s)<1 THEN gV(s):=n; END;
  ELSE
    setSlot(s,gV(s)+d*gSS(s));
  END;
END;

// a two-way choice flips, a longer list opens, a number opens a dialog
editSlot(s)
BEGIN
  LOCAL v:=gV(s);
  LOCAL lst;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    IF SIZE(lst)==2 THEN
      setSlot(s,IFTE(v==1,2,1));
    ELSE
      IF CHOOSE(v,slotName(s),lst) THEN
        IF v>=1 THEN setSlot(s,v); END;
      END;
    END;
  ELSE
    IF INPUT(v,slotName(s),slotName(s)+":",gSH(s)) THEN
      IF TYPE(v)==0 THEN setSlot(s,v); END;
    END;
  END;
  gDown:=0;
END;

// ---- typing a number straight into the selected box ----

// key code -> digit 0-9, 10 for the decimal point, -1 for other keys
keyDigit(ev)
BEGIN
  LOCAL kc:={47,42,43,44,37,38,39,32,33,34,48};
  LOCAL k;
  FOR k FROM 1 TO 11 DO
    IF ev==kc(k) THEN RETURN k-1; END;
  END;
  RETURN -1;
END;

entText()
BEGIN
  LOCAL s:=IFTE(gENeg,"-","");
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK) DO
    s:=s+IFTE(gEK(k)==10,".",CHAR(48+gEK(k)));
  END;
  RETURN s;
END;

entVal()
BEGIN
  LOCAL v:=0;
  LOCAL sc:=0;
  LOCAL k, d;
  FOR k FROM 1 TO SIZE(gEK) DO
    d:=gEK(k);
    IF d==10 THEN
      sc:=1;
    ELSE
      IF sc>0 THEN
        sc:=sc/10;
        v:=v+d*sc;
      ELSE
        v:=v*10+d;
      END;
    END;
  END;
  RETURN IFTE(gENeg,-v,v);
END;

typeKey(d)
BEGIN
  LOCAL hasdot:=0;
  LOCAL k;
  IF gEnt==0 THEN
    gEnt:=1;
    gEK:={};
    gENeg:=0;
  END;
  FOR k FROM 1 TO SIZE(gEK) DO
    IF gEK(k)==10 THEN hasdot:=1; END;
  END;
  IF (d<10 OR hasdot==0) AND SIZE(gEK)<12 THEN gEK(SIZE(gEK)+1):=d; END;
END;

backKey()
BEGIN
  LOCAL nl:={};
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK)-1 DO
    nl(k):=gEK(k);
  END;
  gEK:=nl;
  IF SIZE(gEK)==0 THEN gEnt:=0; END;
END;

// store the typed number (if any) in the selected box
commitEnt()
BEGIN
  LOCAL sl:=toolIns(gTool);
  IF gEnt AND SIZE(gEK)>0 THEN setSlot(sl(gTSel),entVal()); END;
  gEnt:=0;
END;

BEGIN

BEGIN

calcVD()
BEGIN
  LOCAL a:=gV(2);
  LOCAL lf:=gV(3);
  LOCAL k:=gV(4);
  LOCAL m:=gV(5);
  LOCAL mul:=IFTE(gV(6)==2,gRt3,2);
  LOCAL vs:=gV(7);
  LOCAL mx:=gV(9);
  LOCAL kk:=IFTE(m==2,21.2,12.9);
  LOCAL r, vd, j, best;
  IF m==2 AND k<=3 THEN
    addRow("Aluminum","starts at 12 AWG","pick a larger size",cWarn);
    RETURN 0;
  END;
  r:=IFTE(gV(8)==1,kk/gCM(k),gR8(m,k)/1000);
  vd:=mul*a*lf*r;
  addRow("Voltage drop",fmtN(vd,2)+" V",IFTE(mul>2,"1.732","2")+" x I x L x R",cSelc);
  addRow("Drop",fmtN(100*vd/vs,2)+"%",IFTE(100*vd/vs<=mx,"OK, under ","over ")+fmtTr(mx,1)+"%",IFTE(100*vd/vs<=mx,cGood,cBad));
  addRow("At the load",fmtN(vs-vd,1)+" V","of "+fmtTr(vs,1)+" V",cText);
  addRow("R per 1000 ft",fmtE(1000*r)+" Ω",IFTE(gV(8)==1,"K / CM = "+fmt1(kk)+" / "+istr(gCM(k)),"Table 8, 75°C"),cText);
  best:=0;
  FOR j FROM IFTE(m==2,4,1) TO SIZE(gCM) DO
    IF best==0 AND 100*mul*a*lf*IFTE(gV(8)==1,kk/gCM(j),gR8(m,j)/1000)/vs<=mx THEN best:=j; END;
  END;
  addRow("Smallest OK",sizeName(best),"for "+fmtTr(mx,1)+"% drop",cText);
  IF a>0 THEN addRow("Max length",fmtN(mx/100*vs/(mul*a*r),0)+" ft","one way, this wire",cText); END;
  addRow("Circular mils",istr(gCM(k))+" cmil",sizeName(k),cDim);
  RETURN 1;
END;

calcWire()
BEGIN
  LOCAL ld:=gV(10)*IFTE(gV(11)==2,1.25,1);
  LOCAL m:=gV(12);
  LOCAL wc:=gV(13);
  LOCAL tcol:=MIN(gV(14),gV(13));
  LOCAL cf:=corrF(ambC(gV(15)),wc);
  LOCAL af:=adjF(gV(16));
  LOCAL ki:=0;
  LOCAL kt:=0;
  LOCAL k, sz, so;
  addRow("Must carry",fmtN(ld,1)+" A",IFTE(gV(11)==2,"125% of "+fmtTr(gV(10),1)+" A","the load"),cText);
  IF cf<=0 THEN
    addRow("Ambient","too hot","for "+colName(wc)+" wire",cBad);
    RETURN 0;
  END;
  addRow("Factors","x "+fmtN(cf,2)+" x "+fmtN(af,2),"temp "+ambTxt(gV(15))+", "+istr(gV(16))+" wires",cText);
  FOR k FROM IFTE(m==2,4,3) TO SIZE(gCM) DO
    IF ki==0 AND ampT(1,k,m,wc)>0 AND ampT(1,k,m,wc)*cf*af>=ld THEN ki:=k; END;
    IF kt==0 AND ampT(1,k,m,tcol)>0 AND ampT(1,k,m,tcol)>=ld THEN kt:=k; END;
  END;
  IF ki==0 OR kt==0 THEN
    addRow("Wire","over 2000 kcmil","use parallel sets",cBad);
    RETURN 0;
  END;
  addRow("By insulation",sizeName(ki),fmt1(ampT(1,ki,m,wc)*cf*af)+" A after factors",cText);
  addRow("By terminals",sizeName(kt),istr(ampT(1,kt,m,tcol))+" A, "+colName(tcol)+" column",cText);
  sz:=MAX(ki,kt);
  addRow("Use",sizeName(sz)+" "+IFTE(m==2,"Al","Cu"),"Table 310.16",cSelc);
  so:=smallOC(sz,m);
  IF so>0 THEN addRow("Breaker max",istr(so)+" A","240.4(D) small wire",cWarn); END;
  RETURN 1;
END;

calcAmp()
BEGIN
  LOCAL k:=gV(17);
  LOCAL m:=gV(18);
  LOCAL it:=gV(19);
  LOCAL tcl:=gInsCol(it);
  LOCAL cu:=MIN(3,tcl);
  LOCAL fa:=gV(20)==2;
  LOCAL tb:=ampT(IFTE(fa,2,1),k,m,cu);
  LOCAL inl:=choiceList(11);
  LOCAL cf, af, am, tl, so;
  addRow("Insulation",inl(it)+": "+IFTE(tcl==4,"60°C",colName(cu)),IFTE(tcl==4,"90°C wire, 334.80","column used"),cText);
  IF tb<=0 THEN
    addRow("Table","not listed",sizeName(k)+" "+IFTE(m==2,"aluminum","copper"),cWarn);
    RETURN 0;
  END;
  cf:=corrF(ambC(gV(21)),cu);
  af:=IFTE(fa,1,adjF(gV(22)));
  am:=tb*cf*af;
  IF tcl==4 THEN am:=MIN(am,ampT(1,k,m,1)); END;
  addRow(IFTE(fa,"Table 310.17","Table 310.16"),istr(tb)+" A",sizeName(k)+" "+IFTE(m==2,"Al","Cu")+IFTE(fa,", free air",""),cSelc);
  addRow("Temp factor",fmtN(cf,2)+" x",ambTxt(gV(21)),IFTE(cf<>1,cWarn,cText));
  addRow("Bundling",fmtN(af,2)+" x",IFTE(fa,"free air: none",istr(gV(22))+" current-carrying"),IFTE(af<1,cWarn,cText));
  IF gV(23)>1 THEN
    tl:=ampT(IFTE(fa,2,1),k,m,gV(23)-1);
    addRow("Terminals",istr(tl)+" A",colName(gV(23)-1)+" column limit",cText);
    am:=MIN(am,tl);
  END;
  addRow("Ampacity",fmtTr(am,1)+" A","allowed",cSelc);
  so:=smallOC(k,m);
  IF so>0 THEN addRow("Breaker max",istr(so)+" A","240.4(D)",cText); END;
  RETURN 1;
END;

calcBrk()
BEGIN
  LOCAL c:=gV(24);
  LOCAL nc:=gV(25);
  LOCAL need:=1.25*c+nc;
  LOCAL k:=gV(26);
  LOCAL m:=gV(27);
  LOCAL am:=ampT(1,k,m,gV(28));
  LOCAL so:=smallOC(k,m);
  LOCAL mn:=stdUp(need);
  LOCAL mxb;
  addRow("Load",fmtTr(c+nc,1)+" A","cont. "+fmtTr(c,1)+" + other "+fmtTr(nc,1),cText);
  addRow("125% rule",fmtTr(need,2)+" A","1.25 x cont. + other",cText);
  addRow("Min breaker",istr(mn)+" A","210.20(A), 240.6(A)",cSelc);
  IF am<=0 THEN
    addRow("Wire","not listed",sizeName(k)+" "+colName(gV(28)),cWarn);
    RETURN 0;
  END;
  IF so>0 THEN
    mxb:=MIN(so,stdUp(am));
  ELSE
    mxb:=IFTE(am<=800,stdUp(am),stdDn(am));
  END;
  addRow("Wire ampacity",istr(am)+" A",sizeName(k)+", "+colName(gV(28)),IFTE(am>=need,cGood,cBad));
  addRow("Max for wire",istr(mxb)+" A",IFTE(so>0,"240.4(D)","next size up 240.4(B)"),cText);
  IF am>=need AND mn<=mxb THEN
    addRow("Result",istr(mn)+" A breaker","wire is big enough",cGood);
  ELSE
    addRow("Result","larger wire","wire under "+fmtTr(need,1)+" A",cBad);
  END;
  RETURN 1;
END;

calcFill()
BEGIN
  LOCAL ty:=gV(30);
  LOCAL g:=gV(32);
  LOCAL n1:=gV(34);
  LOCAL n2:=gV(36);
  LOCAL a1:=gFill(g,gV(33));
  LOCAL a2:=gFill(g,gV(35));
  LOCAL tot, pc, ar, j, mf, cn;
  LOCAL fs:=choiceList(16);
  LOCAL ts:=choiceList(14);
  LOCAL ct:=choiceList(13);
  IF a1<=0 OR (n2>0 AND a2<=0) THEN
    addRow("Table 5","size not listed","for this insulation",cWarn);
    RETURN 0;
  END;
  tot:=n1*a1+n2*a2;
  pc:=IFTE(n1+n2<=1,53,IFTE(n1+n2==2,31,40));
  addRow("Wire area",fmtN(a1,4)+" in²",fs(gV(33))+" each, Table 5",cText);
  IF n2>0 THEN addRow("2nd wires",fmtN(a2,4)+" in²",istr(n2)+" x "+fs(gV(35)),cText); END;
  addRow("Total",fmtN(tot,4)+" in²",istr(n1+n2)+" wires: "+istr(pc)+"% allowed",cText);
  IF gV(29)==1 THEN
    ar:=gCond(ty,gV(31));
    IF ar<=0 THEN
      addRow("Conduit","not made","in that trade size",cWarn);
      RETURN 0;
    END;
    mf:=maxFit(ar,a1,a2,n2);
    addRow("Conduit",fmtN(ar,3)+" in²",ts(gV(31))+" "+ct(ty)+", Table 4",cText);
    addRow("Max wires",istr(mf),fs(gV(33))+IFTE(n2>0," with the 2nd group",""),cSelc);
    addRow("Your fill",fmt1(100*tot/ar)+"%",IFTE(100*tot/ar<=pc,"OK","over "+istr(pc)+"%"),IFTE(100*tot/ar<=pc,cGood,cBad));
  ELSE
    cn:=0;
    FOR j FROM 1 TO SIZE(ts) DO
      ar:=gCond(ty,j);
      IF cn==0 AND ar>0 THEN
        IF n2==0 THEN
          IF maxFit(ar,a1,0,0)>=n1 THEN cn:=j; END;
        ELSE
          IF tot<=pc/100*ar THEN cn:=j; END;
        END;
      END;
    END;
    IF cn==0 THEN
      addRow("Conduit","none big enough",ct(ty)+" stops here",cBad);
      RETURN 0;
    END;
    ar:=gCond(ty,cn);
    addRow("Use",ts(cn)+" "+ct(ty),"smallest that fits",cSelc);
    addRow("Fill",fmt1(100*tot/ar)+"%","of "+fmtN(ar,3)+" in²",cText);
  END;
  RETURN 1;
END;

calcAwg()
BEGIN
  LOCAL k:=gV(37);
  LOCAL m:=gV(38);
  LOCAL it:=gV(39);
  LOCAL tcl:=MIN(3,gInsCol(it));
  LOCAL cm:=gCM(k);
  LOCAL inl:=choiceList(11);

  addRow("Table 8 R",IFTE(r>0,fmtE(r)+" Ω","-"),"per 1000 ft, 75°C",cText);
  addRow("Raceway",ampTxt(ampT(1,k,m,1))+" / "+ampTxt(ampT(1,k,m,2))+" / "+ampTxt(ampT(1,k,m,3))+" A","60/75/90°C, 310.16",cText);
  addRow("Free air",ampTxt(ampT(2,k,m,1))+" / "+ampTxt(ampT(2,k,m,2))+" / "+ampTxt(ampT(2,k,m,3))+" A","60/75/90°C, 310.17",cText);

calcBox()
BEGIN
  LOCAL vs:=gBoxV(gV(40));
  LOCAL cl:=IFTE(gV(42)==2,1,0);
  LOCAL gd:=IFTE(gV(44)==2,1,0);
  LOCAL cnt:=gV(41)+cl+2*gV(43)+gd+gV(45);
  LOCAL need:=cnt*vs;
  LOCAL bx:=gV(46);
  LOCAL sz:=choiceList(17);
  addRow("Per wire",fmtN(vs,2)+" in³",sz(gV(40))+", 314.16(B)",cText);
  addRow("Count",istr(cnt),"wires "+istr(gV(41))+" + clamps "+istr(cl)+" + yokes "+istr(2*gV(43))+" + gnd "+istr(gd)+" + fit "+istr(gV(45)),cText);
  addRow("Box needs",fmtN(need,2)+" in³","count x volume",cSelc);
  IF bx>0 THEN
    addRow("Your box",IFTE(bx>=need,"big enough","TOO SMALL"),fmtTr(bx,1)+" in³",IFTE(bx>=need,cGood,cBad));
    addRow("Wires allowed",istr(IP(bx/vs+0.0000001)),"of this size in it",cText);
  END;
  RETURN 1;
END;

calcOhm()
BEGIN
  LOCAL v:=gV(47);
  LOCAL a:=gV(48);
  LOCAL r:=gV(49);
  LOCAL p:=gV(50);
  LOCAL n:=(v>0)+(a>0)+(r>0)+(p>0);
  LOCAL md:=0;
  LOCAL fm:={"R = E/I   P = E x I","I = E/R   P = E²/R","I = P/E   R = E²/P",
    "E = I x R   P = I² x R","E = P/I   R = P/I²","E = √(P x R)   I = √(P/R)"};
  LOCAL pr:={"E and I","E and R","E and P","I and R","I and P","R and P"};
  LOCAL gv, ga, gr, gp;
  IF n<2 THEN
    addRow("Enter","any two values","leave the rest 0",cWarn);
    RETURN 0;
  END;
  IF v>0 AND a>0 THEN md:=1; END;
  IF md==0 AND v>0 AND r>0 THEN md:=2; END;
  IF md==0 AND v>0 AND p>0 THEN md:=3; END;
  IF md==0 AND a>0 AND r>0 THEN md:=4; END;
  IF md==0 AND a>0 AND p>0 THEN md:=5; END;
  IF md==0 THEN md:=6; END;
  IF md==1 THEN
    r:=v/a;
    p:=v*a;
  END;
  IF md==2 THEN
    a:=v/r;
    p:=v*v/r;
  END;
  IF md==3 THEN
    a:=p/v;
    r:=v*v/p;
  END;
  IF md==4 THEN
    v:=a*r;
    p:=a*a*r;
  END;
  IF md==5 THEN
    v:=p/a;
    r:=p/(a*a);
  END;
  IF md==6 THEN
    v:=sqroot(p*r);
    a:=sqroot(p/r);
  END;
  gv:=md<=3;
  ga:=md==1 OR md==4 OR md==5;
  gr:=md==2 OR md==4 OR md==6;
  gp:=md==3 OR md==5 OR md==6;
  addRow("Volts (E)",fmtE(v)+" V",IFTE(gv,"entered","worked out"),IFTE(gv,cText,cSelc));
  addRow("Amps (I)",fmtE(a)+" A",IFTE(ga,"entered","worked out"),IFTE(ga,cText,cSelc));
  addRow("Ohms (R)",fmtE(r)+" Ω",IFTE(gr,"entered","worked out"),IFTE(gr,cText,cSelc));
  addRow("Watts (P)",fmtE(p)+" W",IFTE(gp,"entered","worked out"),IFTE(gp,cText,cSelc));
  addRow("Formula",fm(md),"from "+pr(md),cDim);
  IF n>2 THEN addRow("Note","used "+pr(md),"set the others to 0",cWarn); END;
  RETURN 1;
END;

toolCalc(t)
BEGIN
  clearRows();
  IF t==1 THEN calcVD(); END;
  IF t==2 THEN calcWire(); END;
  IF t==3 THEN calcAmp(); END;
  IF t==4 THEN calcBrk(); END;
  IF t==5 THEN calcFill(); END;
  IF t==6 THEN calcAwg(); END;
  IF t==7 THEN calcBox(); END;
  IF t==8 THEN calcOhm(); END;
END;

// ================= input handling =================

// returns 0 to quit
onHome(ev)
BEGIN
  LOCAL mx, my, k;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    IF my<21 THEN
      IF mx>=272 THEN RETURN 0; END;
      IF mx>=224 THEN showHelp(); END;
      RETURN 1;
    END;
    k:=IP((my-23)/72)*3+MIN(2,IP(mx/107))+1;
    IF my>=23 AND k>=1 AND k<=9 THEN openTile(k); END;
    RETURN 1;
  END;
  IF ev==7 THEN gHome:=IFTE(gHome==1,9,gHome-1); END;
  IF ev==8 THEN gHome:=IFTE(gHome==9,1,gHome+1); END;
  IF ev==2 THEN gHome:=IFTE(gHome<=3,gHome+6,gHome-3); END;
  IF ev==12 THEN gHome:=IFTE(gHome>6,gHome-6,gHome+3); END;
  IF ev==30 THEN openTile(gHome); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN RETURN 0; END;
  RETURN 1;
END;

onTool(ev)
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, mx, my, rw, idx, dg;
  IF gTSel>n THEN gTSel:=1; END;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    commitEnt();
    IF my>=220 THEN
      k:=IP(mx/64)+1;
      IF k==1 THEN editSlot(sl(gTSel)); END;
      IF k==2 THEN resetTool(); END;
      IF k==3 THEN openTool(IFTE(gTool==1,gNT,gTool-1)); END;
      IF k==4 THEN openTool(IFTE(gTool==gNT,1,gTool+1)); END;
      IF k==5 THEN goHome(); END;
      RETURN 1;
    END;
    IF my<21 THEN
      IF mx>=272 THEN toolHelp(gTool); END;
      RETURN 1;
    END;
    // tap an input box to change it
    rw:=IP((my-24)/20);
    idx:=rw*2+IP(mx/160)+1;
    IF my>=24 AND rw<IP((n+1)/2) AND idx<=n THEN
      gTSel:=idx;
      editSlot(sl(idx));
    END;
    RETURN 1;
  END;
  // digits and the decimal point type straight into a number box
  dg:=keyDigit(ev);
  IF dg>=0 THEN
    IF gSC(sl(gTSel))==0 THEN typeKey(dg); END;
    RETURN 1;
  END;
  IF gEnt THEN
    IF ev==45 OR ev==27 THEN
      gENeg:=1-gENeg;
      RETURN 1;
    END;
    IF ev==19 THEN
      backKey();
      RETURN 1;
    END;
    IF ev==4 THEN
      gEnt:=0;
      RETURN 1;
    END;
    // any other key keeps the typed number first
    commitEnt();
    IF ev==30 THEN RETURN 1; END;
  ELSE
    IF (ev==45 OR ev==27) AND gSC(sl(gTSel))==0 THEN
      gEnt:=1;
      gEK:={};
      gENeg:=1;
      RETURN 1;
    END;
  END;
  IF ev==7 THEN gTSel:=IFTE(gTSel==1,n,gTSel-1); END;
  IF ev==8 THEN gTSel:=IFTE(gTSel==n,1,gTSel+1); END;
  IF ev==2 THEN bumpSlot(sl(gTSel),1); END;
  IF ev==12 THEN bumpSlot(sl(gTSel),-1); END;
  IF ev==30 THEN editSlot(sl(gTSel)); END;
  IF ev==3 THEN toolHelp(gTool); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN goHome(); END;
  RETURN 1;
END;

EXPORT NECCALC()
BEGIN
  LOCAL run:=1;
  LOCAL ev;
  DIMGROB_P(G1,320,240);
  DIMGROB_P(G9,320,30);
  gEnt:=0;
  WHILE run DO
    IF gScr==1 THEN drawHome(); ELSE drawTool(); END;
    ev:=WAIT(-1);   // key code, touch event list, or -1 after a 1 minute timeout
    IF gScr==1 THEN run:=onHome(ev); ELSE run:=onTool(ev); END;
  END;
END;
