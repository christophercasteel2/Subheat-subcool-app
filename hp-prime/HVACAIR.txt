// ================= number formatting =================

// whole number as text, independent of the calculator's number format
istr(v)
BEGIN
  LOCAL n:=ROUND(ABS(v),0);
  LOCAL s:="";
  IF n<1 THEN RETURN "0"; END;
  WHILE n>=1 DO
    s:=CHAR(48+(n MOD 10))+s;
    n:=IP(n/10);
  END;
  IF v<0 THEN s:="-"+s; END;
  RETURN s;
END;

// one decimal place
fmt1(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*10,0);
  LOCAL s:=istr(IP(n/10))+"."+CHAR(48+(n MOD 10));
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// v with d decimal places (d = 0 to 5)
fmtN(v,d)
BEGIN
  LOCAL m:=10^d;
  LOCAL n:=ROUND(ABS(v)*m,0);
  LOCAL s:=istr(IP(n/m));
  LOCAL f;
  IF d>0 THEN
    f:=istr(n MOD m);
    WHILE DIM(f)<d DO f:="0"+f; END;
    s:=s+"."+f;
  END;
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// up to d decimals, trailing zeros dropped
fmtTr(v,d)
BEGIN
  LOCAL s:=fmtN(v,d);
  IF d>0 THEN
    WHILE RIGHT(s,1)=="0" DO s:=LEFT(s,DIM(s)-1); END;
    IF RIGHT(s,1)=="." THEN s:=LEFT(s,DIM(s)-1); END;
  END;
  RETURN s;
END;

// any size of number; very large or small ones as 1.23E-6
fmtE(v)
BEGIN
  LOCAL a:=ABS(v);
  LOCAL ex, m;
  IF a<0.000000000001 THEN RETURN "0"; END;
  IF a>=0.001 AND a<10000000 THEN
    // six significant figures
    ex:=5;
    IF a>=10 THEN ex:=4; END;
    IF a>=100 THEN ex:=3; END;
    IF a>=1000 THEN ex:=2; END;
    IF a>=10000 THEN ex:=1; END;
    IF a>=100000 THEN ex:=0; END;
    RETURN fmtTr(v,ex);
  END;
  ex:=FLOOR(LN(a)/LN(10));
  m:=a/10^ex;
  IF m>=9.9995 THEN
    m:=m/10;
    ex:=ex+1;
  END;
  IF m<1 THEN
    m:=m*10;
    ex:=ex-1;
  END;
  RETURN IFTE(v<0,"-","")+fmtTr(m,3)+"E"+istr(ex);
END;

// ================= moist air (ASHRAE Fundamentals, I-P) =================

sqroot(x)
BEGIN
  RETURN MAX(0,x)^0.5;
END;

// barometric pressure, psia, at the elevation in the Elevation box
patm()
BEGIN
  RETURN 14.696*(1-0.0000068754*gV(1))^5.2559;
END;

// saturation pressure of water vapour, psia, over ice below 32.018 F
// (Hyland-Wexler)
pws(t)
BEGIN
  LOCAL r:=MAX(150,t+459.67);
  IF t<32.018 THEN
    RETURN EXP(-10214.165/r-4.8932428-0.0053765794*r+0.00000019202377*r^2+0.00000000035575832*r^3-0.000000000000090344688*r^4+4.1635019*LN(r));
  END;
  RETURN EXP(-10440.397/r-11.29465-0.027022355*r+0.00001289036*r^2-0.0000000024780681*r^3+6.5459673*LN(r));
END;

// humidity ratio, lb per lb of dry air, for vapour pressure pw at pressure p
wOf(pw,p)
BEGIN
  RETURN 0.621945*pw/MAX(0.0001,p-pw);
END;

// vapour pressure for humidity ratio w at pressure p
pwOf(w,p)
BEGIN
  RETURN p*w/(0.621945+w);
END;

// humidity ratio from dry bulb and wet bulb
wDbWb(db,wb,p)
BEGIN
  LOCAL ws:=wOf(pws(wb),p);
  IF wb>=32 THEN RETURN ((1093-0.556*wb)*ws-0.240*(db-wb))/(1093+0.444*db-wb); END;
  RETURN ((1220-0.04*wb)*ws-0.240*(db-wb))/(1220+0.444*db-0.48*wb);
END;

// dew point, F, for vapour pressure pw
dewOf(pw)
BEGIN
  LOCAL lo:=-100;
  LOCAL hi:=300;
  LOCAL k, md;
  IF pw<=0.000001 THEN RETURN -100; END;
  FOR k FROM 1 TO 36 DO
    md:=(lo+hi)/2;
    IF pws(md)>pw THEN hi:=md; ELSE lo:=md; END;
  END;
  RETURN (lo+hi)/2;
END;

// wet bulb, F, for dry bulb db and humidity ratio w
wbOf(db,w,p)
BEGIN
  LOCAL lo:=MIN(db,dewOf(pwOf(w,p)));
  LOCAL hi:=db;
  LOCAL k, md;
  FOR k FROM 1 TO 30 DO
    md:=(lo+hi)/2;
    IF wDbWb(db,md,p)>w THEN hi:=md; ELSE lo:=md; END;
  END;
  RETURN (lo+hi)/2;
END;

// enthalpy, BTU per lb of dry air
enth(db,w)
BEGIN
  RETURN 0.240*db+w*(1061+0.444*db);
END;

// specific volume, cubic feet per lb of dry air
spvol(db,w,p)
BEGIN
  RETURN 0.370486*(db+459.67)*(1+1.607858*w)/p;
END;

// humidity ratio from the dry bulb and one more reading x:
// kind 1 relative humidity %, 2 wet bulb, 3 dew point
wFrom(db,kind,x,p)
BEGIN
  IF kind==1 THEN RETURN wOf(MAX(0,MIN(100,x))/100*pws(db),p); END;
  IF kind==2 THEN RETURN wDbWb(db,MIN(x,db),p); END;
  RETURN wOf(pws(MIN(x,db)),p);
END;

// relative humidity, %
rhOf(db,w,p)
BEGIN
  RETURN 100*pwOf(w,p)/pws(db);
END;

// ================= ducts =================

// round duct with the same friction as an a x b rectangular duct (ASHRAE)
deq(a,b)
BEGIN
  RETURN 1.30*(a*b)^0.625/(a+b)^0.25;
END;

// friction loss, in.wc per 100 ft, for q CFM in a round duct d inches
// across, roughness eps ft; standard air, Colebrook friction factor
fric(q,d,eps)
BEGIN
  LOCAL df:=d/12;
  LOCAL v:=q/(gPi*df*df/4);
  LOCAL rey:=8.56*d*v;
  LOCAL f, k;
  IF q<=0 THEN RETURN 0; END;
  IF rey<2300 THEN
    f:=64/rey;
  ELSE
    f:=0.25/LOG(eps/(3.7*df)+5.74/rey^0.9)^2;
    FOR k FROM 1 TO 8 DO
      f:=1/(2*LOG(eps/(3.7*df)+2.51/(rey*f^0.5)))^2;
    END;
  END;
  RETURN 100*f/df*(v/4005)^2;
END;

// round duct diameter, in, giving friction fr for q CFM
dFor(q,fr,eps)
BEGIN
  LOCAL lo:=0;
  LOCAL hi:=6;
  LOCAL k, md;
  FOR k FROM 1 TO 30 DO
    md:=(lo+hi)/2;
    IF fric(q,EXP(md),eps)>fr THEN lo:=md; ELSE hi:=md; END;
  END;
  RETURN EXP((lo+hi)/2);
END;

// width, in, of a rectangular duct h inches high equal to round duct d
widthFor(d,h)
BEGIN
  LOCAL lo:=0.5;
  LOCAL hi:=1000;
  LOCAL k, md;
  FOR k FROM 1 TO 40 DO
    md:=(lo+hi)/2;
    IF deq(md,h)<d THEN lo:=md; ELSE hi:=md; END;
  END;
  RETURN (lo+hi)/2;
END;

// smallest standard round duct of at least d inches
stdRound(d)
BEGIN
  LOCAL k;
  FOR k FROM 1 TO SIZE(gRound) DO
    IF gRound(k)>=d-0.05 THEN RETURN gRound(k); END;
  END;
  RETURN CEILING(d);
END;

// colour for a duct velocity (ACCA Manual D: 900 fpm max supply, 700 return)
velC(v)
BEGIN
  IF v>900 THEN RETURN cBad; END;
  IF v>700 THEN RETURN cWarn; END;
  RETURN cGood;
END;

// boxes whose name follows another box's choice
slotName(s)
BEGIN
  LOCAL lst;
  IF s==4 THEN
    lst:={"RH","Wet bulb","Dew point"};
    RETURN lst(gV(3));
  END;
  IF s==19 THEN RETURN IFTE(gV(14)==1,"% outside","Mixed temp"); END;
  IF s==46 THEN RETURN IFTE(gV(45)==1,"Diameter","Width"); END;
  IF s==69 THEN RETURN IFTE(gV(64)==1,"New RPM","New CFM"); END;
  RETURN gSL(s);
END;

slotUnit(s)
BEGIN
  IF s==4 THEN RETURN IFTE(gV(3)==1,"%","°F"); END;
  IF s==19 THEN RETURN IFTE(gV(14)==1,"%","°F"); END;
  IF s==69 THEN RETURN IFTE(gV(64)==1,"RPM","CFM"); END;
  RETURN gSU(s);
END;

// the box a heat-formula tool is solving for shows ?
isFound(s)
BEGIN
  RETURN (s==28 AND gV(27)==1) OR (s==29 AND gV(27)==2) OR (s==30 AND gV(27)==3);
END;

// text shown in an input box
slotText(s)
BEGIN
  LOCAL u:=slotUnit(s);
  LOCAL v:=gV(s);
  LOCAL lst, t;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    RETURN lst(MAX(1,MIN(SIZE(lst),v)));
  END;
  IF isFound(s) THEN RETURN "?"; END;
  IF s==47 AND gV(45)==1 THEN RETURN "--"; END;
  IF v==0 AND gSZ(s)==1 THEN RETURN "--"; END;
  IF v==0 AND gSZ(s)==2 THEN RETURN "?"; END;
  IF gSD(s)<0 THEN t:=fmtTr(v,4); ELSE t:=fmtN(v,gSD(s)); END;
  IF s==82 THEN
    lst:=convUnits(gV(80));
    RETURN t+" "+lst(MAX(1,MIN(SIZE(lst),gV(81))));
  END;
  IF u=="" THEN RETURN t; END;
  IF LEFT(u,1)=="°" THEN RETURN t+u; END;
  RETURN t+" "+u;
END;

setSlot(s,v)
BEGIN
  gV(s):=MAX(gSLo(s),MIN(gSHi(s),v));
  IF gSC(s)>0 THEN gV(s):=ROUND(gV(s),0); END;
  IF s==80 THEN gV(81):=1; END;
  IF s==81 THEN gV(81):=MIN(gV(81),SIZE(convUnits(gV(80)))); END;
END;

// Up/Down: step a number, or move through a choice list
bumpSlot(s,d)
BEGIN
  LOCAL n;
  IF gSC(s)>0 THEN
    n:=SIZE(choiceList(gSC(s)));
    gV(s):=gV(s)+IFTE(d>0,1,-1);
    IF gV(s)>n THEN gV(s):=1; END;
    IF gV(s)<1 THEN gV(s):=n; END;
    IF s==80 THEN gV(81):=1; END;
  ELSE
    setSlot(s,gV(s)+d*gSS(s));
  END;
END;

// a two-way choice flips, a longer list opens, a number opens a dialog
editSlot(s)
BEGIN
  LOCAL v:=gV(s);
  LOCAL lst;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    IF SIZE(lst)==2 THEN
      setSlot(s,IFTE(v==1,2,1));
    ELSE
      IF CHOOSE(v,slotName(s),lst) THEN
        IF v>=1 THEN setSlot(s,v); END;
      END;
    END;
  ELSE
    IF INPUT(v,slotName(s),slotName(s)+":",gSH(s)) THEN
      IF TYPE(v)==0 THEN setSlot(s,v); END;
    END;
  END;
  gDown:=0;
END;

// key code -> digit 0-9, 10 for the decimal point, -1 for other keys
keyDigit(ev)
BEGIN
  LOCAL kc:={47,42,43,44,37,38,39,32,33,34,48};
  LOCAL k;
  FOR k FROM 1 TO 11 DO
    IF ev==kc(k) THEN RETURN k-1; END;
  END;
  RETURN -1;
END;

entText()
BEGIN
  LOCAL s:=IFTE(gENeg,"-","");
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK) DO
    s:=s+IFTE(gEK(k)==10,".",CHAR(48+gEK(k)));
  END;
  RETURN s;
END;

entVal()
BEGIN
  LOCAL v:=0;
  LOCAL sc:=0;
  LOCAL k, d;
  FOR k FROM 1 TO SIZE(gEK) DO
    d:=gEK(k);
    IF d==10 THEN
      sc:=1;
    ELSE
      IF sc>0 THEN
        sc:=sc/10;
        v:=v+d*sc;
      ELSE
        v:=v*10+d;
      END;
    END;
  END;
  RETURN IFTE(gENeg,-v,v);
END;

typeKey(d)
BEGIN
  LOCAL hasdot:=0;
  LOCAL k;
  IF gEnt==0 THEN
    gEnt:=1;
    gEK:={};
    gENeg:=0;
  END;
  FOR k FROM 1 TO SIZE(gEK) DO
    IF gEK(k)==10 THEN hasdot:=1; END;
  END;
  IF (d<10 OR hasdot==0) AND SIZE(gEK)<12 THEN gEK(SIZE(gEK)+1):=d; END;
END;

backKey()
BEGIN
  LOCAL nl:={};
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK)-1 DO
    nl(k):=gEK(k);
  END;
  gEK:=nl;
  IF SIZE(gEK)==0 THEN gEnt:=0; END;
END;

// store the typed number (if any) in the selected box
commitEnt()
BEGIN
  LOCAL sl:=toolIns(gTool);
  IF gEnt AND SIZE(gEK)>0 THEN setSlot(sl(gTSel),entVal()); END;
  gEnt:=0;
END;

calcProps()
BEGIN
  LOCAL p:=patm();
  LOCAL db:=gV(2);
  LOCAL kind:=gV(3);
  LOCAL x:=gV(4);
  LOCAL w, pw, v;
  IF kind==1 AND (x<=0 OR x>100) THEN
    addRow("Check","RH must be 1 to 100%","",cBad);
    RETURN 0;
  END;
  IF kind>1 AND x>db THEN
    addRow("Check",IFTE(kind==2,"wet bulb","dew point")+" above dry bulb","",cBad);
    RETURN 0;
  END;
  w:=wFrom(db,kind,x,p);
  IF w<=0 THEN
    addRow("Check","wet bulb too low","no moisture left",cBad);
    RETURN 0;
  END;
  pw:=pwOf(w,p);
  v:=spvol(db,w,p);
  addRow("Rel. humidity",fmt1(100*pw/pws(db))+"%",IFTE(kind==1,"entered",""),IFTE(kind==1,cText,cSelc));
  addRow("Wet bulb",fmt1(wbOf(db,w,p))+"°F",IFTE(kind==2,"entered",""),IFTE(kind==2,cText,cSelc));
  addRow("Dew point",fmt1(dewOf(pw))+"°F",IFTE(kind==3,"entered",""),IFTE(kind==3,cText,cSelc));
  addRow("Moisture",fmt1(w*7000)+" gr/lb",fmtN(w*1000,2)+" g/kg",cSelc);
  addRow("Enthalpy",fmtN(enth(db,w),2)+" BTU/lb","per lb of dry air",cSelc);
  addRow("Volume",fmtN(v,2)+" ft³/lb","density "+fmtN((1+w)/v,4)+" lb/ft³",cText);
  addRow("Air pressure",fmtN(p,3)+" psia",fmtN(p*2.03602,2)+" inHg",cText);
  addRow("Heat factors","1.08 = "+fmtN(1.08*p/14.696,3),"4.5 = "+fmtN(4.5*p/14.696,2)+" here",cText);
  RETURN 1;
END;

calcDew()
BEGIN
  LOCAL t:=gV(5);
  LOCAL ts:=gV(7);
  LOCAL dp:=dewOf(gV(6)/100*pws(t));
  LOCAL mg:=ts-dp;
  LOCAL c:=cGood;
  LOCAL vd:="no sweating";
  IF mg<=3 THEN
    c:=cWarn;
    vd:="close: may sweat";
  END;
  IF mg<=0 THEN
    c:=cBad;
    vd:="WILL SWEAT";
  END;
  addRow("Dew point",fmt1(dp)+"°F","of the air",cSelc);
  addRow("Surface",vd,fmtTr(ts,1)+"°F",c);
  addRow("Margin",IFTE(mg>=0,"+","")+fmt1(mg)+"°F","surface - dew point",c);
  IF ts>=t THEN
    addRow("Sweats above","never","surface warmer than air",cText);
  ELSE
    addRow("Sweats above",fmt1(100*pws(ts)/pws(t))+"% RH","at "+fmtTr(t,1)+"°F air",cText);
  END;
  RETURN 1;
END;

calcCoil()
BEGIN
  LOCAL p:=patm();
  LOCAL k:=p/14.696;
  LOCAL q:=gV(8);
  LOCAL wr, ws, tot, sen, cpt;
  IF gV(11)>gV(10) OR gV(13)>gV(12) THEN
    addRow("Check","wet bulb above dry bulb","",cBad);
    RETURN 0;
  END;
  wr:=wDbWb(gV(10),gV(11),p);
  ws:=wDbWb(gV(12),gV(13),p);
  tot:=4.5*k*q*(enth(gV(10),wr)-enth(gV(12),ws));
  sen:=1.08*k*q*(gV(10)-gV(12));
  cpt:=q/gV(9);
  addRow("Total",fmtN(tot,0)+" BTU/h",fmtN(tot/12000,2)+" tons, "+fmtN(tot/(120*gV(9)),0)+"% of rated",cSelc);
  addRow("Sensible",fmtN(sen,0)+" BTU/h","1.08 x CFM x split",cText);
  addRow("Latent",fmtN(tot-sen,0)+" BTU/h","moisture removed",cText);
  IF tot>0 THEN addRow("SHR",fmtN(sen/tot,2),"typ 0.70-0.80",cText); END;
  addRow("Split",fmt1(gV(10)-gV(12))+"°F","return - supply",cText);
  addRow("CFM per ton",fmtN(cpt,0),"typ 350-450",IFTE(cpt>=350 AND cpt<=450,cGood,cWarn));
  addRow("Condensate",fmtN(4.5*k*q*(wr-ws)/8.34,2)+" gal/h",fmtN(4.5*k*q*(wr-ws),1)+" lb/h",cText);
  RETURN 1;
END;

calcMix()
BEGIN
  LOCAL p:=patm();
  LOCAL od:=gV(15);
  LOCAL rd:=gV(17);
  LOCAL wo:=wFrom(od,1,gV(16),p);
  LOCAL wr:=wFrom(rd,1,gV(18),p);
  LOCAL ho:=enth(od,wo);
  LOCAL hr:=enth(rd,wr);
  LOCAL x, md, wm;
  IF gV(14)==1 THEN
    x:=MAX(0,MIN(100,gV(19)))/100;
    md:=x*od+(1-x)*rd;
  ELSE
    md:=gV(19);
    IF ABS(od-rd)<0.5 THEN
      addRow("Check","outside = return temp","can't tell the %",cWarn);
      RETURN 0;
    END;
    x:=(md-rd)/(od-rd);
  END;
  wm:=wo*x+wr*(1-x);
  IF gV(14)==1 THEN
    addRow("Mixed air",fmt1(md)+"°F",fmtTr(100*x,1)+"% outside air",cSelc);
  ELSE
    addRow("Outside air",fmt1(100*x)+"%","of the total airflow",IFTE(x<0 OR x>1,cBad,cSelc));
    IF x<0 OR x>1 THEN addRow("Check","mixed not between","outside and return",cBad); END;
  END;
  addRow("Outside CFM",fmtN(x*gV(20),0)+" CFM","of "+fmtN(gV(20),0),cText);
  addRow("Mixed RH",fmt1(rhOf(md,wm,p))+"%","WB "+fmt1(wbOf(md,wm,p))+"°F",cText);
  addRow("Mixed",fmt1(wm*7000)+" gr/lb","h "+fmtN(enth(md,wm),1)+" BTU/lb",cText);
  addRow("Enthalpy",fmtN(ho,1)+" outside",fmtN(hr,1)+" return, BTU/lb",cText);
  IF ho<hr THEN
    addRow("Economizer","outside air helps","lower enthalpy",cGood);
  ELSE
    addRow("Economizer","minimum OA only","outside air holds more heat",cWarn);
  END;
  RETURN 1;
END;

calcFurn()

calcHeatF()
BEGIN
  LOCAL k:=patm()/14.696;
  LOCAL fnd:=gV(27);
  LOCAL q:=gV(28);
  LOCAL cfm:=gV(29);
  LOCAL dt:=gV(30);
  LOCAL gr:=gV(31);
  LOCAL lt;
  IF fnd==1 THEN q:=1.08*k*cfm*dt; END;
  IF fnd==2 THEN
    IF dt<=0 THEN
      addRow("Enter","a temperature difference","",cWarn);
      RETURN 0;
    END;
    cfm:=q/(1.08*k*dt);
  END;
  IF fnd==3 THEN
    IF cfm<=0 THEN
      addRow("Enter","the airflow","",cWarn);
      RETURN 0;
    END;
    dt:=q/(1.08*k*cfm);
  END;
  addRow("Sensible",fmtN(q,0)+" BTU/h",fmtN(1.08*k,3)+" x CFM x ΔT",IFTE(fnd==1,cSelc,cText));
  addRow("Airflow",fmtN(cfm,0)+" CFM",IFTE(fnd==2,"BTU/h / ("+fmtN(1.08*k,3)+" x ΔT)","entered"),IFTE(fnd==2,cSelc,cText));
  addRow("Temp diff",fmt1(dt)+"°F",IFTE(fnd==3,"BTU/h / ("+fmtN(1.08*k,3)+" x CFM)","entered"),IFTE(fnd==3,cSelc,cText));
  IF gr>0 THEN
    lt:=0.68*k*cfm*gr;
    addRow("Latent",fmtN(lt,0)+" BTU/h",fmtN(0.68*k,3)+" x CFM x grains",cText);
    addRow("Total",fmtN(q+lt,0)+" BTU/h",fmtN((q+lt)/12000,2)+" tons",cSelc);
  ELSE
    addRow("In tons",fmtN(q/12000,2),"12,000 BTU/h each",cText);
  END;
  RETURN 1;
END;

calcDuctSz()
BEGIN
  LOCAL q:=gV(41);

  LOCAL wi:=CEILING(wr-0.05);
  LOCAL v, ar;
  addRow("Round",fmt1(d)+" in","exact, at "+fmtN(fr,2)+" in.wc",cText);

  addRow("Use round",istr(ds)+" in",fmtN(v,0)+" fpm, "+fmtN(fric(q,ds,eps),3)+" /100 ft",velC(v));
  IF wr>999 THEN
    addRow("Rectangular","too flat","make it taller",cWarn);
  ELSE

END;

convFactor(ct)
BEGIN
  IF ct==1 THEN RETURN {1,2.11888,0.58857778,35.314667}; END;
  IF ct==2 THEN RETURN {1,0.0040146308,4.0146308,0.039370079,27.679905,13.595098}; END;
  IF ct==3 THEN RETURN {1,196.85039,60,88}; END;
  IF ct==4 THEN RETURN {1,3.4121416,3412.1416,12000,1000}; END;
  IF ct==6 THEN RETURN {1,7,7000}; END;
  RETURN {1,0.0069444444,10.76391,0.001076391};
END;

// temperature in unit u (1 F, 2 C, 3 K) to kelvin, and back
degK(u,v)
BEGIN
  IF u==1 THEN RETURN (v-32)/1.8+273.15; END;
  IF u==2 THEN RETURN v+273.15; END;
  RETURN v;
END;

degOf(u,k)
BEGIN
  IF u==1 THEN RETURN (k-273.15)*1.8+32; END;
  IF u==2 THEN RETURN k-273.15; END;
  RETURN k;
END;

calcConv()
BEGIN
  LOCAL ct:=gV(80);
  LOCAL u:=convUnits(ct);
  LOCAL n:=SIZE(u);
  LOCAL fr:=MAX(1,MIN(n,gV(81)));
  LOCAL v:=gV(82);
  LOCAL f:=convFactor(ct);
  LOCAL bv, k, ov;
  IF ct==5 THEN bv:=degK(fr,v); ELSE bv:=v*f(fr); END;
  FOR k FROM 1 TO n DO
    IF ct==5 THEN ov:=degOf(k,bv); ELSE ov:=bv/f(k); END;
    addRow(u(k),fmtE(ov),IFTE(k==fr,"entered",""),IFTE(k==fr,cSelc,cText));
  END;
  IF ct==5 AND bv<0 THEN addRow("Note","below absolute zero","",cBad); END;
  RETURN 1;
END;

calcLoad()

END;

  TEXTOUT_P(gTName(gTool),G1,5,3,3,cWhite);
  chip("HELP",272);
  // input boxes, two per row
  FOR k FROM 1 TO n DO
    s:=sl(k);
    x0:=((k-1) MOD 2)*160;
    y:=24+IP((k-1)/2)*20;
    TEXTOUT_P(slotName(s),G1,x0+4,y+4,1,cDim);
    IF k==gTSel AND gEnt THEN
      RECT_P(G1,x0+74,y,x0+157,y+17,cSelc,cHit);
      txt:=entText()+"_";
    ELSE
      RECT_P(G1,x0+74,y,x0+157,y+17,IFTE(k==gTSel,cSelc,cEdge),cPanel);
      txt:=slotText(s);
    END;
    big:=DIM(txt)<=11;
    TEXTOUT_P(txt,G1,x0+78,y+IFTE(big,2,4),IFTE(big,2,1),cText);
  END;
  // results
  y:=24+IP((n+1)/2)*20+3;
  LINE_P(G1,0,y-2,319,y-2,cEdge);
  FOR k FROM 1 TO SIZE(gRL) DO
    IF (k MOD 2)==0 THEN RECT_P(G1,0,y,319,y+14,cRow); END;
    TEXTOUT_P(gRL(k),G1,5,y+2,1,cDim);
    TEXTOUT_P(gRV(k),G1,90,y+1,2,gRC(k));
    rtext(gRN(k),316,y+2,1,cDim);
    y:=y+15;
  END;
  drawMenu({"EDIT","RESET","< PREV","NEXT >","HOME"});
  BLIT_P(G0,G1);
END;

openTool(t)
BEGIN
  gTool:=t;
  gTSel:=1;
  gEnt:=0;
  gScr:=2;
END;

goHome()
BEGIN
  gHome:=gTool;
  gEnt:=0;
  gScr:=1;
END;

toolsMenu()
BEGIN
  LOCAL t:=gTool;
  IF CHOOSE(t,"Tools",gTName) THEN
    IF t>=1 THEN openTool(t); END;
  END;
  gDown:=0;
END;

// RESET: the tool's example values
resetTool()
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL k;
  FOR k FROM 1 TO SIZE(sl) DO
    gV(sl(k)):=gDef(sl(k));
  END;
  gEnt:=0;
END;

// ================= input handling =================

// returns 0 to quit
onHome(ev)
BEGIN
  LOCAL mx, my, k;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    IF my<21 THEN
      IF mx>=272 THEN RETURN 0; END;
      IF mx>=224 THEN showHelp(); END;
      RETURN 1;
    END;
    k:=IP((my-23)/43)*4+IP(mx/80)+1;
    IF my>=23 AND k>=1 AND k<=gNT THEN openTool(k); END;
    RETURN 1;
  END;
  IF ev==7 THEN gHome:=IFTE(gHome==1,gNT,gHome-1); END;
  IF ev==8 THEN gHome:=IFTE(gHome==gNT,1,gHome+1); END;
  IF ev==2 THEN gHome:=IFTE(gHome<=4,gHome+gNT-4,gHome-4); END;
  IF ev==12 THEN gHome:=IFTE(gHome>gNT-4,gHome-gNT+4,gHome+4); END;
  IF ev==30 THEN openTool(gHome); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN RETURN 0; END;
  RETURN 1;
END;

onTool(ev)
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, mx, my, rw, idx, dg;
  IF gTSel>n THEN gTSel:=1; END;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    commitEnt();
    IF my>=220 THEN
      k:=IP(mx/64)+1;
      IF k==1 THEN editSlot(sl(gTSel)); END;
      IF k==2 THEN resetTool(); END;
      IF k==3 THEN openTool(IFTE(gTool==1,gNT,gTool-1)); END;
      IF k==4 THEN openTool(IFTE(gTool==gNT,1,gTool+1)); END;
      IF k==5 THEN goHome(); END;
      RETURN 1;
    END;
    IF my<21 THEN
      IF mx>=272 THEN toolHelp(gTool); END;
      RETURN 1;
    END;
    // tap an input box to change it
    rw:=IP((my-24)/20);
    idx:=rw*2+IP(mx/160)+1;
    IF my>=24 AND rw<IP((n+1)/2) AND idx<=n THEN
      gTSel:=idx;
      editSlot(sl(idx));
    END;
    RETURN 1;
  END;
  // digits and the decimal point type straight into a number box
  dg:=keyDigit(ev);
  IF dg>=0 THEN
    IF gSC(sl(gTSel))==0 THEN typeKey(dg); END;
    RETURN 1;
  END;
  IF gEnt THEN
    IF ev==45 OR ev==27 THEN
      gENeg:=1-gENeg;
      RETURN 1;
    END;
    IF ev==19 THEN
      backKey();
      RETURN 1;
    END;
    IF ev==4 THEN
      gEnt:=0;
      RETURN 1;
    END;
    // any other key keeps the typed number first
    commitEnt();
    IF ev==30 THEN RETURN 1; END;
  ELSE
    IF (ev==45 OR ev==27) AND gSC(sl(gTSel))==0 THEN
      gEnt:=1;
      gEK:={};
      gENeg:=1;
      RETURN 1;
    END;
  END;
  IF ev==7 THEN gTSel:=IFTE(gTSel==1,n,gTSel-1); END;
  IF ev==8 THEN gTSel:=IFTE(gTSel==n,1,gTSel+1); END;
  IF ev==2 THEN bumpSlot(sl(gTSel),1); END;
  IF ev==12 THEN bumpSlot(sl(gTSel),-1); END;
  IF ev==30 THEN editSlot(sl(gTSel)); END;
  IF ev==3 THEN toolHelp(gTool); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN goHome(); END;
  RETURN 1;
END;

// ================= main =================

EXPORT HVACAIR()
BEGIN
  LOCAL run:=1;
  LOCAL ev;
  DIMGROB_P(G1,320,240);
  DIMGROB_P(G9,320,30);
  gEnt:=0;
  WHILE run DO
    IF gScr==1 THEN drawHome(); ELSE drawTool(); END;
    ev:=WAIT(-1);   // key code, touch event list, or -1 after a 1 minute timeout
    IF gScr==1 THEN run:=onHome(ev); ELSE run:=onTool(ev); END;
  END;
END;
