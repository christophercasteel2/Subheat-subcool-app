// ================= number formatting =================

// whole number as text, independent of the calculator's number format
istr(v)
BEGIN
  LOCAL n:=ROUND(ABS(v),0);
  LOCAL s:="";
  IF n<1 THEN RETURN "0"; END;
  WHILE n>=1 DO
    s:=CHAR(48+(n MOD 10))+s;
    n:=IP(n/10);
  END;
  IF v<0 THEN s:="-"+s; END;
  RETURN s;
END;

// one decimal place
fmt1(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*10,0);
  LOCAL s:=istr(IP(n/10))+"."+CHAR(48+(n MOD 10));
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// v with d decimal places (d = 0 to 5)
fmtN(v,d)
BEGIN
  LOCAL m:=10^d;
  LOCAL n:=ROUND(ABS(v)*m,0);
  LOCAL s:=istr(IP(n/m));
  LOCAL f;
  IF d>0 THEN
    f:=istr(n MOD m);
    WHILE DIM(f)<d DO f:="0"+f; END;
    s:=s+"."+f;
  END;
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// up to d decimals, trailing zeros dropped
fmtTr(v,d)
BEGIN
  LOCAL s:=fmtN(v,d);
  IF d>0 THEN
    WHILE RIGHT(s,1)=="0" DO s:=LEFT(s,DIM(s)-1); END;
    IF RIGHT(s,1)=="." THEN s:=LEFT(s,DIM(s)-1); END;
  END;
  RETURN s;
END;

// any size of number; very large or small ones as 1.23E-6
fmtE(v)
BEGIN
  LOCAL a:=ABS(v);
  LOCAL ex, m;
  IF a<0.000000000001 THEN RETURN "0"; END;
  IF a>=0.000001 AND a<10000000 THEN
    // six significant figures
    ex:=5-FLOOR(LOG(a)+0.0000000001);
    IF ex>5 THEN RETURN fmtTr(v,MIN(11,ex)); END;
    ex:=5;
    IF a>=10 THEN ex:=4; END;
    IF a>=100 THEN ex:=3; END;
    IF a>=1000 THEN ex:=2; END;
    IF a>=10000 THEN ex:=1; END;
    IF a>=100000 THEN ex:=0; END;
    RETURN fmtTr(v,ex);
  END;
  ex:=FLOOR(LN(a)/LN(10));
  m:=a/10^ex;
  IF m>=9.9995 THEN
    m:=m/10;
    ex:=ex+1;
  END;
  IF m<1 THEN
    m:=m*10;
    ex:=ex-1;
  END;
  RETURN IFTE(v<0,"-","")+fmtTr(m,5)+"E"+istr(ex);
END;

// ================= circuit helpers =================

sqroot(x)
BEGIN
  RETURN MAX(0,x)^0.5;
END;

// two resistors in parallel; 0 means that resistor is not there
par2(a,b)
BEGIN
  IF a<=0 THEN RETURN MAX(0,b); END;
  IF b<=0 THEN RETURN a; END;
  RETURN a*b/(a+b);
END;

// ET and IT from RT and whichever of ET (ee) or IT (ii) was given
solveEI(rt,ee,ii)
BEGIN
  IF rt<=0 THEN RETURN {0,0}; END;
  IF ee>0 THEN RETURN {ee,ee/rt}; END;
  IF ii>0 THEN RETURN {ii*rt,ii}; END;
  RETURN {0,0};
END;

// a value with a metric prefix, e.g. 4.7 kΩ or 15 mA
engTxt(v,u)
BEGIN
  LOCAL a:=ABS(v);
  IF a==0 THEN RETURN "0 "+u; END;
  IF a>=1000000000 THEN RETURN fmtE(v/1000000000)+" G"+u; END;
  IF a>=1000000 THEN RETURN fmtE(v/1000000)+" M"+u; END;
  IF a>=1000 THEN RETURN fmtE(v/1000)+" k"+u; END;
  IF a>=1 THEN RETURN fmtE(v)+" "+u; END;
  IF a>=0.001 THEN RETURN fmtE(v*1000)+" m"+u; END;
  IF a>=0.000001 THEN RETURN fmtE(v*1000000)+" µ"+u; END;
  IF a>=0.000000001 THEN RETURN fmtE(v*1000000000)+" n"+u; END;
  RETURN fmtE(v*1000000000000)+" p"+u;
END;

// text shown in an input box
slotText(s)
BEGIN
  LOCAL u:=slotUnit(s);
  LOCAL v:=gV(s);
  LOCAL lst, t;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    RETURN lst(MAX(1,MIN(SIZE(lst),v)));
  END;
  IF v==0 AND gSZ(s)==1 THEN RETURN "--"; END;
  IF v==0 AND gSZ(s)==2 THEN RETURN "?"; END;
  IF gSD(s)<0 THEN t:=fmtTr(v,4); ELSE t:=fmtN(v,gSD(s)); END;
  IF u=="" THEN RETURN t; END;
  IF LEFT(u,1)=="°" THEN RETURN t+u; END;
  RETURN t+" "+u;
END;

setSlot(s,v)
BEGIN
  gV(s):=MAX(gSLo(s),MIN(gSHi(s),v));
  IF gSC(s)>0 THEN gV(s):=ROUND(gV(s),0); END;
END;

// Up/Down: step a number, or move through a choice list
bumpSlot(s,d)
BEGIN
  LOCAL n;
  IF gSC(s)>0 THEN
    n:=SIZE(choiceList(gSC(s)));
    gV(s):=gV(s)+IFTE(d>0,1,-1);
    IF gV(s)>n THEN gV(s):=1; END;
    IF gV(s)<1 THEN gV(s):=n; END;
  ELSE
    setSlot(s,gV(s)+d*gSS(s));
  END;
END;

// a two-way choice flips, a longer list opens, a number opens a dialog
editSlot(s)
BEGIN
  LOCAL v:=gV(s);
  LOCAL lst;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    IF SIZE(lst)==2 THEN
      setSlot(s,IFTE(v==1,2,1));
    ELSE
      IF CHOOSE(v,slotName(s),lst) THEN
        IF v>=1 THEN setSlot(s,v); END;
      END;
    END;
  ELSE
    IF INPUT(v,slotName(s),slotName(s)+":",gSH(s)) THEN
      IF TYPE(v)==0 THEN setSlot(s,v); END;
    END;
  END;
  gDown:=0;
END;

END;

entText()

  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK) DO
    s:=s+IFTE(gEK(k)==10,".",CHAR(48+gEK(k)));
  END;
  RETURN s;
END;

entVal()
BEGIN
  LOCAL v:=0;
  LOCAL sc:=0;
  LOCAL k, d;
  FOR k FROM 1 TO SIZE(gEK) DO
    d:=gEK(k);
    IF d==10 THEN
      sc:=1;
    ELSE
      IF sc>0 THEN
        sc:=sc/10;
        v:=v+d*sc;
      ELSE
        v:=v*10+d;
      END;
    END;
  END;
  RETURN IFTE(gENeg,-v,v);
END;

typeKey(d)
BEGIN
  LOCAL hasdot:=0;
  LOCAL k;
  IF gEnt==0 THEN
    gEnt:=1;
    gEK:={};
    gENeg:=0;
  END;
  FOR k FROM 1 TO SIZE(gEK) DO
    IF gEK(k)==10 THEN hasdot:=1; END;
  END;
  IF (d<10 OR hasdot==0) AND SIZE(gEK)<12 THEN gEK(SIZE(gEK)+1):=d; END;
END;

  END;
  gEK:=nl;

calcSer()
BEGIN
  LOCAL rs:={gV(3),gV(4),gV(5),gV(6),gV(7),gV(8),gV(9)};
  LOCAL rt:=0;
  LOCAL k, ei;
  FOR k FROM 1 TO 7 DO rt:=rt+MAX(0,rs(k)); END;
  IF rt<=0 THEN
    addRow("RT","enter resistors","",cWarn);
    RETURN 0;
  END;
  ei:=solveEI(rt,gV(1),gV(2));
  IF ei(2)<=0 THEN
    addRow("RT",fmtE(rt)+" Ω","",cSelc);
    addRow("Enter","ET or IT","",cWarn);
    RETURN 0;
  END;
  totRows(rt,ei(1),ei(2));
  FOR k FROM 1 TO 7 DO
    IF rs(k)>0 THEN addRow("E"+istr(k)+" P"+istr(k),fmtE(ei(2)*rs(k))+" V  "+fmtE(ei(2)^2*rs(k))+" W","",cText); END;
  END;
  addRow("","Same current "+fmtE(ei(2))+" A in each","",cDim);
  RETURN 1;
END;

calcSP()
BEGIN
  LOCAL rs:={gV(21),gV(22),gV(23),gV(24),gV(25),gV(26)};
  LOCAL nm:={"A","B","C"};
  LOCAL g:=0;
  LOCAL k, b, rb, rt, ei, ib;
  FOR b FROM 1 TO 3 DO
    rb:=MAX(0,rs(2*b-1))+MAX(0,rs(2*b));
    IF rb>0 THEN g:=g+1/rb; END;
  END;
  IF g<=0 THEN
    addRow("RT","enter resistors","",cWarn);
    RETURN 0;
  END;
  rt:=1/g;
  ei:=solveEI(rt,gV(19),gV(20));

    addRow("RT",fmtE(rt)+" Ω","",cSelc);

  totRows(rt,ei(1),ei(2));
  FOR b FROM 1 TO 3 DO

    IF rb>0 THEN

      END;

  END;
  RETURN 1;
END;

calcComb()
BEGIN
  LOCAL r:={gV(29),gV(30),gV(31),gV(32),gV(33),gV(34),gV(35)};
  LOCAL p1:=par2(gV(30),gV(31));
  LOCAL p2:=par2(gV(33),gV(34));

  LOCAL ei, it, ep, k;
  IF rt<=0 THEN
    addRow("RT","enter resistors","",cWarn);

  END;
  ei:=solveEI(rt,gV(27),gV(28));
  IF ei(2)<=0 THEN
    addRow("RT",fmtE(rt)+" Ω","",cSelc);
    addRow("Enter","ET or IT","",cWarn);
    RETURN 0;
  END;
  it:=ei(2);

// troubleshooting case h: 0 all good, 1-4 that resistor open, 5-8
// resistor h-4 shorted; predicted E1..E4 and IT
trbCase(h)
BEGIN
  LOCAL r:={gV(38),gV(39),gV(40),gV(41)};
  LOCAL op:={0,0,0,0};
  LOCAL ev:={0,0,0,0};
  LOCAL et:=gV(36);
  LOCAL it:=0;
  LOCAL b, a, c, ia;
  IF h>=1 AND h<=4 THEN op(h):=1; END;
  IF h>=5 THEN r(h-4):=0; END;
  FOR b FROM 0 TO 1 DO
    a:=2*b+1;
    c:=a+1;
    IF op(a)==1 OR op(c)==1 THEN
      IF op(a)==1 THEN ev(a):=et; ELSE ev(c):=et; END;
    ELSE
      IF r(a)+r(c)>0 THEN
        ia:=et/(r(a)+r(c));
        it:=it+ia;
        ev(a):=ia*r(a);
        ev(c):=ia*r(c);
      ELSE
        it:=it+1000000;
      END;
    END;
  END;
  RETURN {ev(1),ev(2),ev(3),ev(4),it};
END;

// how far case p is from the readings (0 = a perfect match)
trbErr(p)
BEGIN
  LOCAL m:={gV(42),gV(43),gV(44),gV(45)};
  LOCAL er:=0;
  LOCAL k;
  FOR k FROM 1 TO 4 DO er:=er+ABS(p(k)-m(k))/MAX(gV(36),0.000001); END;
  IF gV(37)>0 THEN er:=er+ABS(p(5)-gV(37))/gV(37); END;
  RETURN er;
END;

calcTrb()
BEGIN
  LOCAL nm:={"All resistors OK","R1 is OPEN","R2 is OPEN","R3 is OPEN","R4 is OPEN","R1 is SHORTED","R2 is SHORTED","R3 is SHORTED","R4 is SHORTED"};
  LOCAL m:={gV(42),gV(43),gV(44),gV(45)};
  LOCAL nrm:=trbCase(0);
  LOCAL best:=0;
  LOCAL be:=trbErr(trbCase(0));
  LOCAL h, er, p, k;
  FOR h FROM 1 TO 8 DO
    er:=trbErr(trbCase(h));
    IF er<be-0.000001 THEN
      be:=er;
      best:=h;
    END;
  END;
  addRow("IT normal",engTxt(nrm(5),"A"),"",cText);
  addRow("IT meas.",IFTE(gV(37)>0,engTxt(gV(37),"A"),"not given"),"",cText);
  FOR k FROM 1 TO 4 DO
    addRow("E"+istr(k)+" norm/meas",fmtE(nrm(k))+" / "+fmtE(m(k))+" V","",IFTE(ABS(nrm(k)-m(k))>0.05*gV(36),cWarn,cText));
  END;
  addRow("Answer",nm(best+1),"",IFTE(best==0,cGood,cBad));
  p:=trbCase(best);
  addRow("Predicted IT",engTxt(p(5),"A"),"",cText);
  addRow("Match",IFTE(be<0.03,"close",IFTE(be<0.15,"fair","poor: check readings")),"",IFTE(be<0.03,cGood,cWarn));
  IF best>=1 AND best<=4 THEN addRow("","Open: full ET across it, 0 A","",cDim); END;
  IF best>=5 THEN addRow("","Shorted: 0 V across it, I rises","",cDim); END;
  RETURN 1;
END;

calcKVL()
BEGIN
  LOCAL n:=gV(47);
  LOCAL et:=gV(46);
  LOCAL sm:=0;
  LOCAL un:=0;
  LOCAL ux:=0;
  LOCAL k, ek;
  FOR k FROM 1 TO n DO
    ek:=gV(47+k);
    IF ek>0 THEN
      sm:=sm+ek;
    ELSE
      un:=un+1;
      ux:=k;
    END;
  END;
  addRow("Known drops",fmtE(sm)+" V","",cText);
  IF et<=0 AND un==0 THEN
    addRow("ET",fmtE(sm)+" V","",cSelc);
    addRow("","ET = sum of the drops","",cDim);
    RETURN 1;
  END;
  IF un==1 AND et>0 THEN
    addRow("E"+istr(ux),fmtE(et-sm)+" V","",IFTE(et-sm>=0,cSelc,cBad));
    addRow("","E"+istr(ux)+" = ET - known drops","",cDim);
    RETURN 1;
  END;
  IF un==0 THEN
    addRow("ET",fmtE(et)+" V","",cText);
    addRow("Difference",fmtE(et-sm)+" V",IFTE(ABS(et-sm)<0.0001,"","check"),IFTE(ABS(et-sm)<0.0001*MAX(1,et),cGood,cWarn));
    RETURN 1;
  END;
  addRow("Enter","one unknown only","",cWarn);
  RETURN 0;
END;

calcNet()
BEGIN
  LOCAL e1:=gV(63);
  LOCAL e2:=IFTE(gV(65)==1,gV(64),-gV(64));
  LOCAL r1:=gV(66);
  LOCAL r2:=gV(67);
  LOCAL r3:=gV(68);
  LOCAL v, i1, i2, i3;
  IF r1<=0 OR r2<=0 OR r3<=0 THEN
    addRow("Enter","R1, R2 and R3","",cWarn);
    RETURN 0;
  END;
  v:=(e1/r1+e2/r2)/(1/r1+1/r2+1/r3);
  i1:=(e1-v)/r1;
  i2:=(e2-v)/r2;
  i3:=v/r3;
  addRow("Top node",fmtE(v)+" V","",cText);
  addRow("I1",fmtE(i1)+" A","",IFTE(i1<0,cWarn,cSelc));
  addRow("I2",fmtE(i2)+" A","",IFTE(i2<0,cWarn,cSelc));
  addRow("I3",fmtE(i3)+" A","",IFTE(i3<0,cWarn,cSelc));
  addRow("E R1",fmtE(i1*r1)+" V","",cText);
  addRow("E R2",fmtE(i2*r2)+" V","",cText);
  addRow("E R3",fmtE(i3*r3)+" V","",cText);
  addRow("KCL check","I1 + I2 = "+fmtE(i1+i2),"",cDim);
  IF i1<0 OR i2<0 OR i3<0 THEN
    addRow("","Negative = flows opposite","",cWarn);
    addRow("","to the assumed direction","",cWarn);
  END;
  RETURN 1;
END;

calcOhm()
BEGIN
  LOCAL v:=gV(69);
  LOCAL a:=gV(70);
  LOCAL r:=gV(71);
  LOCAL p:=gV(72);
  LOCAL n:=(v>0)+(a>0)+(r>0)+(p>0);
  LOCAL md:=0;
  LOCAL fm:={"R = E/I  P = E x I","I = E/R  P = E²/R","I = P/E  R = E²/P","E = I x R  P = I² x R","E = P/I  R = P/I²","E = √(PR)  I = √(P/R)"};
  IF n<2 THEN
    addRow("Enter","any two values","",cWarn);
    RETURN 0;
  END;
  IF v>0 AND a>0 THEN md:=1; END;
  IF md==0 AND v>0 AND r>0 THEN md:=2; END;
  IF md==0 AND v>0 AND p>0 THEN md:=3; END;
  IF md==0 AND a>0 AND r>0 THEN md:=4; END;
  IF md==0 AND a>0 AND p>0 THEN md:=5; END;
  IF md==0 THEN md:=6; END;
  IF md==1 THEN
    r:=v/a;
    p:=v*a;
  END;
  IF md==2 THEN
    a:=v/r;
    p:=v*v/r;
  END;
  IF md==3 THEN
    a:=p/v;
    r:=v*v/p;
  END;
  IF md==4 THEN
    v:=a*r;
    p:=a*a*r;
  END;
  IF md==5 THEN
    v:=p/a;
    r:=p/(a*a);
  END;
  IF md==6 THEN
    v:=sqroot(p*r);
    a:=sqroot(p/r);
  END;
  addRow("Volts E",fmtE(v)+" V","",cSelc);
  addRow("Amps I",fmtE(a)+" A","",cSelc);
  addRow("Ohms R",fmtE(r)+" Ω","",cSelc);
  addRow("Watts P",fmtE(p)+" W","",cSelc);
  addRow("",fm(md),"",cDim);
  IF n>2 THEN addRow("","Used 2 values; zero the rest","",cWarn); END;
  RETURN 1;
END;

  TEXTOUT_P(gTName(gTool),G1,5,3,3,cWhite);
  chip("HELP",272);
  // input boxes down the left
  FOR k FROM 1 TO n DO
    s:=sl(k);
    y:=23+(k-1)*19;
    TEXTOUT_P(slotName(s),G1,3,y+4,1,cDim);

    TEXTOUT_P(txt,G1,64,y+IFTE(big,2,4),IFTE(big,2,1),cText);
  END;

  LINE_P(G1,160,22,160,217,cEdge);
  y:=23;
  FOR k FROM 1 TO MIN(13,SIZE(gRL)) DO

openTool(t)
BEGIN
  gTool:=t;
  gTSel:=1;
  gEnt:=0;
  gScr:=2;
END;

goHome()
BEGIN
  gHome:=gTool;
  gEnt:=0;
  gScr:=1;
END;

toolsMenu()
BEGIN
  LOCAL t:=gTool;
  IF CHOOSE(t,"Tools",gTName) THEN
    IF t>=1 THEN openTool(t); END;
  END;
  gDown:=0;
END;

// RESET: the tool's example values
resetTool()
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL k;
  FOR k FROM 1 TO SIZE(sl) DO
    gV(sl(k)):=gDef(sl(k));
  END;
  gEnt:=0;
END;

// open tile k: a tool, or the Learn pages
openTile(k)
BEGIN
  gHome:=k;
  IF k>gNT THEN
    learnMenu();
  ELSE
    openTool(k);
  END;
END;

// ================= input handling =================

// returns 0 to quit
onHome(ev)
BEGIN
  LOCAL mx, my, k;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    IF my<21 THEN
      IF mx>=272 THEN RETURN 0; END;
      IF mx>=224 THEN showHelp(); END;
      RETURN 1;
    END;
    k:=IP((my-23)/50)*4+MIN(3,IP(mx/80))+1;
    IF my>=23 AND k>=1 AND k<=gNT+1 THEN openTile(k); END;
    RETURN 1;
  END;
  IF ev==7 THEN gHome:=IFTE(gHome==1,gNT+1,gHome-1); END;
  IF ev==8 THEN gHome:=IFTE(gHome==gNT+1,1,gHome+1); END;
  IF ev==2 THEN
    gHome:=gHome-4;
    WHILE gHome<1 DO gHome:=gHome+4*IP((gNT+4)/4); END;
    WHILE gHome>gNT+1 DO gHome:=gHome-4; END;
  END;
  IF ev==12 THEN
    gHome:=gHome+4;
    IF gHome>gNT+1 THEN gHome:=((gHome-1) MOD 4)+1; END;
  END;
  IF ev==30 THEN openTile(gHome); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN RETURN 0; END;
  RETURN 1;
END;

onTool(ev)
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, mx, my, rw, idx, dg;
  IF gTSel>n THEN gTSel:=1; END;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    commitEnt();
    IF my>=220 THEN
      k:=IP(mx/64)+1;
      IF k==1 THEN editSlot(sl(gTSel)); END;
      IF k==2 THEN resetTool(); END;
      IF k==3 THEN openTool(IFTE(gTool==1,gNT,gTool-1)); END;
      IF k==4 THEN openTool(IFTE(gTool==gNT,1,gTool+1)); END;
      IF k==5 THEN goHome(); END;
      RETURN 1;
    END;
    IF my<21 THEN
      IF mx>=272 THEN toolHelp(gTool); END;
      RETURN 1;
    END;
    // tap an input box to change it
    rw:=IP((my-23)/19);
    idx:=rw+1;
    IF my>=23 AND mx<160 AND idx>=1 AND idx<=n THEN
      gTSel:=idx;
      editSlot(sl(idx));
    END;
    RETURN 1;
  END;
  // digits and the decimal point type straight into a number box
  dg:=keyDigit(ev);
  IF dg>=0 THEN
    IF gSC(sl(gTSel))==0 THEN typeKey(dg); END;
    RETURN 1;
  END;
  IF gEnt THEN
    IF ev==45 OR ev==27 THEN
      gENeg:=1-gENeg;
      RETURN 1;
    END;
    IF ev==19 THEN
      backKey();
      RETURN 1;
    END;
    IF ev==4 THEN
      gEnt:=0;
      RETURN 1;
    END;
    // any other key keeps the typed number first
    commitEnt();
    IF ev==30 THEN RETURN 1; END;
  ELSE
    IF (ev==45 OR ev==27) AND gSC(sl(gTSel))==0 THEN
      gEnt:=1;
      gEK:={};
      gENeg:=1;
      RETURN 1;
    END;
  END;
  IF ev==7 THEN gTSel:=IFTE(gTSel==1,n,gTSel-1); END;
  IF ev==8 THEN gTSel:=IFTE(gTSel==n,1,gTSel+1); END;
  IF ev==2 THEN bumpSlot(sl(gTSel),1); END;
  IF ev==12 THEN bumpSlot(sl(gTSel),-1); END;
  IF ev==30 THEN editSlot(sl(gTSel)); END;
  IF ev==3 THEN toolHelp(gTool); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN goHome(); END;
  RETURN 1;
END;

EXPORT CIRCUITS()
BEGIN
  LOCAL run:=1;
  LOCAL ev;
  DIMGROB_P(G1,320,240);
  DIMGROB_P(G9,320,30);
  gEnt:=0;
  WHILE run DO
    IF gScr==1 THEN drawHome(); ELSE drawTool(); END;
    ev:=WAIT(-1);   // key code, touch event list, or -1 after a 1 minute timeout
    IF gScr==1 THEN run:=onHome(ev); ELSE run:=onTool(ev); END;
  END;
END;
