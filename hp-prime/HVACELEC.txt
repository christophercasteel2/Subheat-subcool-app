// ================= number formatting =================

// whole number as text, independent of the calculator's number format
istr(v)
BEGIN
  LOCAL n:=ROUND(ABS(v),0);
  LOCAL s:="";
  IF n<1 THEN RETURN "0"; END;
  WHILE n>=1 DO
    s:=CHAR(48+(n MOD 10))+s;
    n:=IP(n/10);
  END;
  IF v<0 THEN s:="-"+s; END;
  RETURN s;
END;

// one decimal place
fmt1(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*10,0);
  LOCAL s:=istr(IP(n/10))+"."+CHAR(48+(n MOD 10));
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// v with d decimal places (d = 0 to 5)
fmtN(v,d)
BEGIN
  LOCAL m:=10^d;
  LOCAL n:=ROUND(ABS(v)*m,0);
  LOCAL s:=istr(IP(n/m));
  LOCAL f;
  IF d>0 THEN
    f:=istr(n MOD m);
    WHILE DIM(f)<d DO f:="0"+f; END;
    s:=s+"."+f;
  END;
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// up to d decimals, trailing zeros dropped
fmtTr(v,d)
BEGIN
  LOCAL s:=fmtN(v,d);
  IF d>0 THEN
    WHILE RIGHT(s,1)=="0" DO s:=LEFT(s,DIM(s)-1); END;
    IF RIGHT(s,1)=="." THEN s:=LEFT(s,DIM(s)-1); END;
  END;
  RETURN s;
END;

// about three significant figures
fmt3(v)
BEGIN
  LOCAL a:=ABS(v);
  IF a<0.0005 THEN RETURN "0"; END;
  IF a>=99.95 THEN RETURN fmtN(v,0); END;
  IF a>=9.995 THEN RETURN fmtN(v,1); END;
  IF a>=0.9995 THEN RETURN fmtN(v,2); END;
  RETURN fmtN(v,3);
END;

// any size of number; very large or small ones as 1.23E-6
fmtE(v)
BEGIN
  LOCAL a:=ABS(v);
  LOCAL ex, m;
  IF a<0.000000000001 THEN RETURN "0"; END;
  IF a>=0.001 AND a<10000000 THEN
    // six significant figures
    ex:=5;
    IF a>=10 THEN ex:=4; END;
    IF a>=100 THEN ex:=3; END;
    IF a>=1000 THEN ex:=2; END;
    IF a>=10000 THEN ex:=1; END;
    IF a>=100000 THEN ex:=0; END;
    RETURN fmtTr(v,ex);
  END;
  ex:=FLOOR(LN(a)/LN(10));
  m:=a/10^ex;
  IF m>=9.9995 THEN
    m:=m/10;
    ex:=ex+1;
  END;
  IF m<1 THEN
    m:=m*10;
    ex:=ex-1;
  END;
  RETURN IFTE(v<0,"-","")+fmtTr(m,3)+"E"+istr(ex);
END;

END;

END;

BEGIN

END;

cuberoot(x)
BEGIN
  IF x<0 THEN RETURN -((-x)^(1/3)); END;
  RETURN x^(1/3);
END;

// smallest standard fuse / breaker rating (NEC 240.6(A)) of at least a
stdUp(a)
BEGIN
  LOCAL k;
  FOR k FROM 1 TO SIZE(gStd) DO
    IF gStd(k)>=a-0.0001 THEN RETURN gStd(k); END;
  END;
  RETURN gStd(SIZE(gStd));
END;

// largest standard rating of at most a (0 if a is under 15 A)
stdDn(a)
BEGIN
  LOCAL k;
  LOCAL r:=0;
  FOR k FROM 1 TO SIZE(gStd) DO
    IF gStd(k)<=a+0.0001 THEN r:=gStd(k); END;
  END;
  RETURN r;
END;

// fuses also come in 1, 3, 6 and 10 A
fuseUp(a)
BEGIN
  IF a<=1 THEN RETURN 1; END;
  IF a<=3 THEN RETURN 3; END;
  IF a<=6 THEN RETURN 6; END;
  IF a<=10 THEN RETURN 10; END;
  RETURN stdUp(a);
END;

// NEC 310.16 ampacity: wire k, material m (1 Cu, 2 Al), column c (1 60C,
// 2 75C, 3 90C); 0 = not listed
ampac(k,m,c)
BEGIN
  RETURN gAmp((m-1)*3+c,k);
END;

// smallest wire (14 AWG up) with at least a amps in column c, 0 = none
wireFor(a,m,c)
BEGIN
  LOCAL k;
  FOR k FROM 3 TO SIZE(gWire) DO
    IF ampac(k,m,c)>=a AND ampac(k,m,c)>0 THEN RETURN k; END;
  END;
  RETURN 0;
END;

wireName(k)
BEGIN
  IF k<1 THEN RETURN "over 500 kcmil"; END;
  IF k>=16 THEN RETURN gWire(k)+" kcmil"; END;
  RETURN gWire(k)+" AWG";
END;

// NEMA MG 1 derating factor for a voltage unbalance of u %; 0 = over 5%
nemaDer(u)
BEGIN
  LOCAL x:={0,1,2,3,4,5};
  LOCAL y:={1,1,0.95,0.88,0.82,0.75};
  LOCAL k;
  IF u>5 THEN RETURN 0; END;
  FOR k FROM 1 TO 5 DO
    IF u<=x(k+1) THEN RETURN y(k)+(y(k+1)-y(k))*(u-x(k)); END;
  END;
  RETURN 0.75;
END;

// thermistor temperature (F) for r ohms on curve cv
thTemp(cv,r)
BEGIN
  LOCAL xv:=LN(r);
  LOCAL tk;
  IF cv==3 THEN
    tk:=1/(1/298.15+LN(r/(gV(87)*1000))/gV(88));
  ELSE
    tk:=1/(gShA(cv)+gShB(cv)*xv+gShC(cv)*xv^3);
  END;
  RETURN (tk-273.15)*1.8+32;
END;

// thermistor resistance (ohms) at tf F on curve cv
thRes(cv,tf)
BEGIN
  LOCAL tk:=(tf-32)/1.8+273.15;
  LOCAL y, b3, xx;
  IF cv==3 THEN RETURN gV(87)*1000*EXP(gV(88)*(1/tk-1/298.15)); END;
  // Steinhart-Hart solved for ln R
  y:=(gShA(cv)-1/tk)/gShC(cv);
  b3:=gShB(cv)/(3*gShC(cv));
  xx:=sqroot(b3^3+y*y/4);
  RETURN EXP(cuberoot(xx-y/2)-cuberoot(xx+y/2));
END;

// the series / parallel values are R1-R4 or C1-C4
slotName(s)
BEGIN
  IF s>=15 AND s<=18 THEN RETURN IFTE(gV(12)==2,"C","R")+istr(s-14); END;
  RETURN gSL(s);
END;

slotUnit(s)
BEGIN
  IF s>=15 AND s<=18 THEN RETURN IFTE(gV(12)==2,"µF","Ω"); END;
  RETURN gSU(s);
END;

// text shown in an input box
slotText(s)
BEGIN
  LOCAL u:=slotUnit(s);
  LOCAL v:=gV(s);
  LOCAL lst, t;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    RETURN lst(MAX(1,MIN(SIZE(lst),v)));
  END;
  IF v==0 AND gSZ(s)==1 THEN RETURN "--"; END;
  IF v==0 AND gSZ(s)==2 THEN RETURN "?"; END;
  IF gSD(s)<0 THEN t:=fmtTr(v,4); ELSE t:=fmtN(v,gSD(s)); END;
  IF s==93 THEN
    lst:=convUnits(gV(92));
    RETURN t+" "+lst(MAX(1,MIN(SIZE(lst),gV(94))));
  END;
  IF u=="" THEN RETURN t; END;
  IF LEFT(u,1)=="°" THEN RETURN t+u; END;
  RETURN t+" "+u;
END;

setSlot(s,v)
BEGIN
  gV(s):=MAX(gSLo(s),MIN(gSHi(s),v));
  IF gSC(s)>0 THEN gV(s):=ROUND(gV(s),0); END;
  IF s==92 THEN gV(94):=1; END;
  IF s==94 THEN gV(94):=MIN(gV(94),SIZE(convUnits(gV(92)))); END;
END;

// Up/Down: step a number, or move through a choice list
bumpSlot(s,d)
BEGIN
  LOCAL n;
  IF gSC(s)>0 THEN
    n:=SIZE(choiceList(gSC(s)));
    gV(s):=gV(s)+IFTE(d>0,1,-1);
    IF gV(s)>n THEN gV(s):=1; END;
    IF gV(s)<1 THEN gV(s):=n; END;
    IF s==92 THEN gV(94):=1; END;
  ELSE
    setSlot(s,gV(s)+d*gSS(s));
  END;
END;

// a two-way choice flips, a longer list opens, a number opens a dialog
editSlot(s)
BEGIN
  LOCAL v:=gV(s);
  LOCAL lst;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    IF SIZE(lst)==2 THEN
      setSlot(s,IFTE(v==1,2,1));
    ELSE
      IF CHOOSE(v,slotName(s),lst) THEN
        IF v>=1 THEN setSlot(s,v); END;
      END;
    END;
  ELSE
    IF INPUT(v,slotName(s),slotName(s)+":",gSH(s)) THEN
      IF TYPE(v)==0 THEN setSlot(s,v); END;
    END;
  END;
  gDown:=0;
END;

  LOCAL kc:={47,42,43,44,37,38,39,32,33,34,48};

entText()
BEGIN
  LOCAL s:=IFTE(gENeg,"-","");
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK) DO
    s:=s+IFTE(gEK(k)==10,".",CHAR(48+gEK(k)));
  END;
  RETURN s;
END;

entVal()
BEGIN
  LOCAL v:=0;
  LOCAL sc:=0;
  LOCAL k, d;
  FOR k FROM 1 TO SIZE(gEK) DO
    d:=gEK(k);
    IF d==10 THEN
      sc:=1;
    ELSE
      IF sc>0 THEN
        sc:=sc/10;
        v:=v+d*sc;
      ELSE
        v:=v*10+d;
      END;
    END;
  END;
  RETURN IFTE(gENeg,-v,v);
END;

typeKey(d)
BEGIN
  LOCAL hasdot:=0;
  LOCAL k;
  IF gEnt==0 THEN
    gEnt:=1;
    gEK:={};
    gENeg:=0;
  END;
  FOR k FROM 1 TO SIZE(gEK) DO
    IF gEK(k)==10 THEN hasdot:=1; END;
  END;
  IF (d<10 OR hasdot==0) AND SIZE(gEK)<12 THEN gEK(SIZE(gEK)+1):=d; END;
END;

backKey()
BEGIN
  LOCAL nl:={};
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK)-1 DO
    nl(k):=gEK(k);
  END;
  gEK:=nl;
  IF SIZE(gEK)==0 THEN gEnt:=0; END;
END;

calcOhm()
BEGIN
  LOCAL v:=gV(1);
  LOCAL a:=gV(2);
  LOCAL r:=gV(3);
  LOCAL p:=gV(4);
  LOCAL n:=(v>0)+(a>0)+(r>0)+(p>0);
  LOCAL md:=0;
  LOCAL fm:={"R = E/I   P = E x I","I = E/R   P = E²/R","I = P/E   R = E²/P",
    "E = I x R   P = I² x R","E = P/I   R = P/I²","E = √(P x R)   I = √(P/R)"};
  LOCAL pr:={"E and I","E and R","E and P","I and R","I and P","R and P"};
  LOCAL gv, ga, gr, gp;
  IF n<2 THEN
    addRow("Enter","any two values","leave the rest 0",cWarn);
    RETURN 0;
  END;
  // first pair entered, in the order E, I, R, P
  IF v>0 AND a>0 THEN md:=1; END;
  IF md==0 AND v>0 AND r>0 THEN md:=2; END;
  IF md==0 AND v>0 AND p>0 THEN md:=3; END;
  IF md==0 AND a>0 AND r>0 THEN md:=4; END;
  IF md==0 AND a>0 AND p>0 THEN md:=5; END;
  IF md==0 THEN md:=6; END;
  IF md==1 THEN
    r:=v/a;
    p:=v*a;
  END;
  IF md==2 THEN
    a:=v/r;
    p:=v*v/r;
  END;
  IF md==3 THEN
    a:=p/v;
    r:=v*v/p;
  END;
  IF md==4 THEN
    v:=a*r;
    p:=a*a*r;
  END;
  IF md==5 THEN
    v:=p/a;
    r:=p/(a*a);
  END;
  IF md==6 THEN
    v:=sqroot(p*r);
    a:=sqroot(p/r);
  END;
  gv:=md<=3;
  ga:=md==1 OR md==4 OR md==5;
  gr:=md==2 OR md==4 OR md==6;
  gp:=md==3 OR md==5 OR md==6;
  addRow("Volts (E)",fmt3(v)+" V",IFTE(gv,"entered","worked out"),IFTE(gv,cText,cSelc));
  addRow("Amps (I)",fmt3(a)+" A",IFTE(ga,"entered","worked out"),IFTE(ga,cText,cSelc));
  addRow("Ohms (R)",fmtOhm(r),IFTE(gr,"entered","worked out"),IFTE(gr,cText,cSelc));
  addRow("Watts (P)",fmtU(p,"W"),IFTE(gp,"entered","worked out"),IFTE(gp,cText,cSelc));
  addRow("Formula",fm(md),"from "+pr(md),cDim);
  IF n>2 THEN addRow("Note","used "+pr(md),"set the others to 0",cWarn); END;
  RETURN 1;
END;

calcSerPar()
BEGIN
  LOCAL isc:=gV(12)==2;
  LOCAL isp:=gV(13)==2;
  LOCAL v:=gV(14);
  LOCAL n:=0;
  LOCAL tot:=0;
  LOCAL rcp:=0;
  LOCAL k, xv, tt, ti;
  FOR k FROM 15 TO 18 DO
    xv:=gV(k);
    IF xv>0 THEN
      n:=n+1;
      tot:=tot+xv;
      rcp:=rcp+1/xv;
    END;
  END;
  IF n==0 THEN
    addRow("Enter","at least one value","",cWarn);
    RETURN 0;
  END;
  // resistors add in series, capacitors add in parallel
  IF isc==isp THEN tt:=tot; ELSE tt:=1/rcp; END;
  IF isc THEN
    addRow("Total",fmt3(tt)+" µF",IFTE(isp,"C1 + C2 + ...","1 / (1/C1 + 1/C2 ...)"),cSelc);
    IF isp THEN
      addRow("Voltage","same on each","rate each one for it",cText);
    ELSE
      FOR k FROM 15 TO 18 DO
        IF gV(k)>0 AND v>0 THEN addRow("C"+istr(k-14)+" gets",fmt3(v*tt/gV(k))+" V","of "+fmtTr(v,1)+" V",cText); END;
      END;
    END;
    RETURN 1;
  END;
  addRow("Total",fmtOhm(tt),IFTE(isp,"1 / (1/R1 + 1/R2 ...)","R1 + R2 + ..."),cSelc);
  IF v<=0 THEN RETURN 1; END;
  ti:=v/tt;
  addRow("Current",fmt3(ti)+" A","total, V / Rt",cText);
  addRow("Power",fmtU(v*ti,"W"),"total, V x I",cText);
  FOR k FROM 15 TO 18 DO
    xv:=gV(k);
    IF xv>0 THEN
      IF isp THEN
        addRow("R"+istr(k-14)+" "+fmtOhm(xv),fmt3(v/xv)+" A",fmtU(v*v/xv,"W"),cText);
      ELSE
        addRow("R"+istr(k-14)+" "+fmtOhm(xv),fmt3(ti*xv)+" V",fmtU(ti*ti*xv,"W"),cText);
      END;
    END;
  END;
  RETURN 1;
END;

calc3ph()
BEGIN
  LOCAL v:={gV(24),gV(25),gV(26)};
  LOCAL a:={gV(27),gV(28),gV(29)};
  LOCAL vn:={"L1-L2","L2-L3","L1-L3"};
  LOCAL an:={"L1","L2","L3"};
  LOCAL va:=(v(1)+v(2)+v(3))/3;
  LOCAL aa:=(a(1)+a(2)+a(3))/3;
  LOCAL dv:=0;
  LOCAL da:=0;
  LOCAL kv:=1;
  LOCAL hi:=1;
  LOCAL lo:=1;
  LOCAL k, pu, iu, c, t, f;
  FOR k FROM 1 TO 3 DO
    IF ABS(v(k)-va)>dv THEN
      dv:=ABS(v(k)-va);
      kv:=k;
    END;
    IF ABS(a(k)-aa)>da THEN da:=ABS(a(k)-aa); END;
    IF a(k)>a(hi) THEN hi:=k; END;
    IF a(k)<a(lo) THEN lo:=k; END;
  END;
  IF va<=0 THEN
    addRow("Volts","enter all three","",cWarn);
    RETURN 0;
  END;
  pu:=100*dv/va;
  c:=cGood;
  t:="NEMA: under 1%";
  IF pu>1 THEN
    c:=cWarn;
    t:="most makers: 2% max";
  END;
  IF pu>2 THEN
    c:=cBad;
    t:="too high: fix supply";
  END;
  addRow("Avg volts",fmt1(va)+" V","most off: "+vn(kv)+" "+fmt1(dv)+" V",cText);
  addRow("V unbalance",fmtN(pu,2)+"%",t,c);
  f:=nemaDer(pu);
  IF f>0 THEN
    addRow("NEMA derate",fmtN(f,2)+" x HP",IFTE(f<1,"load the motor less","no derating"),IFTE(f<1,cWarn,cGood));
  ELSE
    addRow("NEMA derate","do not run","over 5% unbalance",cBad);
  END;
  IF aa<=0 THEN RETURN 1; END;
  iu:=100*da/aa;
  addRow("Avg amps",fmt3(aa)+" A","most off: "+fmt3(da)+" A",cText);
  addRow("A unbalance",fmtN(iu,1)+"%",IFTE(iu<=10,"OK, under 10%","over 10%: check"),IFTE(iu<=10,cGood,cBad));
  addRow("Legs","high "+an(hi)+", low "+an(lo),"roll leads to test",cText);
  RETURN 1;
END;

calcWind()
BEGIN
  LOCAL r:={gV(37),gV(38),gV(39)};
  LOCAL pa1:={1,2,1};
  LOCAL pa2:={2,3,3};
  LOCAL tf:=(234.5+25)/(234.5+(gV(40)-32)/1.8);
  LOCAL op:=0;
  LOCAL mx:=1;
  LOCAL ja:=0;
  LOCAL jb:=0;
  LOCAL k, av, dv, pu, cm, tr, ts, sm, d;
  FOR k FROM 1 TO 3 DO
    IF r(k)>=10000 THEN op:=op+1; END;
    IF r(k)>r(mx) THEN mx:=k; END;
  END;
  IF gV(36)==2 THEN
    // three phase: the three pairs should match
    IF op>0 THEN
      addRow("Winding","OPEN","a pair reads OL",cBad);
      RETURN 0;
    END;
    av:=(r(1)+r(2)+r(3))/3;
    dv:=0;
    FOR k FROM 1 TO 3 DO dv:=MAX(dv,ABS(r(k)-av)); END;
    pu:=IFTE(av>0,100*dv/av,0);
    addRow("Average",fmtOhm(av),"of the 3 pairs",cText);
    addRow("Most off",fmtN(pu,1)+"%",fmtOhm(dv),IFTE(pu<=5,cGood,cWarn));
    IF av<0.1 THEN
      addRow("Windings","near 0 Ω","shorted?",cBad);
    ELSE
      addRow("Windings",IFTE(pu<=5,"Balanced","Unbalanced"),IFTE(pu<=5,"pairs match","shorted turns?"),IFTE(pu<=5,cGood,cBad));
    END;
    IF ABS(tf-1)>0.005 THEN addRow("At 77°F",fmtOhm(av*tf),"average, for spec",cText); END;
    RETURN 1;
  END;
  // single phase compressor: find common, run and start
  IF op==3 THEN
    addRow("All pairs","OPEN (OL)","",cBad);
    addRow("Likely","overload open","let it cool, retest",cWarn);
    RETURN 0;
  END;
  IF op==2 THEN
    addRow("Only one pair","reads ohms","the other two OL",cWarn);
    addRow("If it is S-R","overload open","let it cool, retest",cWarn);
    addRow("Otherwise","open winding","compressor failed",cBad);
    RETURN 0;
  END;
  IF op==1 THEN
    addRow("Readings","do not add up","recheck the leads",cWarn);
    RETURN 0;
  END;
  // the highest reading is start + run; common is the terminal left out
  cm:=IFTE(mx==1,3,IFTE(mx==2,1,2));
  FOR k FROM 1 TO 3 DO
    IF k<>mx THEN
      IF ja==0 THEN ja:=k; ELSE jb:=k; END;
    END;
  END;
  IF r(ja)>r(jb) THEN
    k:=ja;
    ja:=jb;
    jb:=k;
  END;
  tr:=IFTE(pa1(ja)==cm,pa2(ja),pa1(ja));
  ts:=IFTE(pa1(jb)==cm,pa2(jb),pa1(jb));
  addRow("Common (C)","terminal "+istr(cm),"",cSelc);
  addRow("Run (R)","terminal "+istr(tr),"C-R "+fmtOhm(r(ja)),cSelc);
  addRow("Start (S)","terminal "+istr(ts),"C-S "+fmtOhm(r(jb)),cSelc);
  sm:=r(ja)+r(jb);
  d:=IFTE(r(mx)>0,100*ABS(sm-r(mx))/r(mx),0);
  addRow("C-R + C-S",fmtOhm(sm),"S-R reads "+fmtOhm(r(mx)),IFTE(d<=5,cGood,cWarn));
  IF r(ja)<0.1 THEN
    addRow("Windings","near 0 Ω","shorted?",cBad);
  ELSE
    addRow("Windings",IFTE(d<=5,"look normal","do not add up"),IFTE(d<=5,"sum matches S-R","recheck, shorted turns?"),IFTE(d<=5,cGood,cWarn));
  END;
  IF ABS(tf-1)>0.005 THEN addRow("At 77°F","R "+fmtOhm(r(ja)*tf)+"  S "+fmtOhm(r(jb)*tf),"for the spec",cText); END;
  RETURN 1;
END;

    IF best==0 AND 100*m*kk*a*lf/gCM(k)/vs<=mx THEN best:=k; END;
  END;
  addRow("Voltage drop",fmtN(vd,2)+" V",IFTE(m==2,"2","1.732")+" x K x I x L / CM",cSelc);
  addRow("Drop",fmtN(pc,2)+"%",IFTE(pc<=mx,"OK, ","over ")+fmtTr(mx,1)+"% limit",IFTE(pc<=mx,cGood,cBad));
  addRow("At the load",fmt1(vs-vd)+" V","from "+fmtTr(vs,1)+" V",cText);
  addRow("Smallest OK",IFTE(best>0,wireName(best),"over 500 kcmil"),"for "+fmtTr(mx,1)+"% drop",cText);
  am:=ampac(wk,mt,2);
  IF am>0 THEN
    addRow("Ampacity",istr(am)+" A","75°C, "+IFTE(am>=a,"carries the load","TOO SMALL"),IFTE(am>=a,cText,cBad));
  ELSE
    addRow("Ampacity","control wire","not in 310.16 at 75°C",cDim);
  END;
  addRow("Each wire",fmtOhm(kk*lf/gCM(wk)),istr(lf)+" ft, K = "+fmt1(kk),cText);
  RETURN 1;
END;

calcAmpac()
BEGIN
  LOCAL wk:=gV(66);
  LOCAL mt:=gV(67);
  LOCAL ic:=gV(68);
  LOCAL tcd:=45+15*ic;
  LOCAL ta:=(gV(70)-32)/1.8+IFTE(gV(72)==2,33,0);
  LOCAL nc:=ROUND(gV(71),0);
  LOCAL tb:=ampac(wk,mt,ic);
  LOCAL af:=1;
  LOCAL oc:=0;
  LOCAL cf, cor, tl, al;
  IF tb<=0 THEN
    addRow("Ampacity","not in 310.16",IFTE(mt==2,"Al starts at 12 AWG","18-16 AWG: 90°C only"),cWarn);
    RETURN 0;
  END;
  cf:=IFTE(ta>=tcd,0,sqroot((tcd-ta)/(tcd-30)));
  IF nc>3 THEN af:=0.8; END;
  IF nc>6 THEN af:=0.7; END;
  IF nc>9 THEN af:=0.5; END;
  IF nc>20 THEN af:=0.45; END;
  IF nc>30 THEN af:=0.4; END;
  IF nc>40 THEN af:=0.35; END;
  cor:=tb*cf*af;
  tl:=ampac(wk,mt,gV(69));
  IF wk<=2 THEN tl:=tb; END;
  al:=MIN(cor,tl);
  addRow("Table 310.16",istr(tb)+" A",istr(tcd)+"°C "+IFTE(mt==2,"aluminum","copper"),cText);
  addRow("Ambient",fmtN(cf,2)+" x",fmtTr(ta*1.8+32,1)+"°F"+IFTE(gV(72)==2," (incl. roof +60)",""),IFTE(cf<1,cWarn,cText));
  addRow("Bundling",fmtN(af,2)+" x",istr(nc)+" current-carrying",IFTE(af<1,cWarn,cText));
  addRow("Derated",fmt1(cor)+" A","table x factors",cText);
  addRow("Terminal limit",istr(tl)+" A",IFTE(gV(69)==2,"75°C","60°C")+" column",cText);
  addRow("Allowable",fmt1(al)+" A",IFTE(al<cor,"set by the terminals","set by derating"),cSelc);
  // NEC 240.4(D) small conductors
  IF mt==1 THEN
    IF wk==1 THEN oc:=7; END;
    IF wk==2 THEN oc:=10; END;
    IF wk==3 THEN oc:=15; END;
    IF wk==4 THEN oc:=20; END;
    IF wk==5 THEN oc:=30; END;
  ELSE
    IF wk==4 THEN oc:=15; END;
    IF wk==5 THEN oc:=25; END;
  END;
  IF al<=0 THEN
    addRow("Max breaker","none","too hot for this wire",cBad);
  ELSE
    IF oc>0 THEN
      addRow("Max breaker",istr(MIN(oc,stdUp(al)))+" A","240.4(D) small wire",cSelc);
    ELSE
      addRow("Max breaker",istr(stdUp(al))+" A","next size up, 240.4(B)",cSelc);
    END;
  END;
  RETURN 1;
END;

calcXfmr()
BEGIN
  LOCAL va:=gV(73);
  LOCAL k:=phk(gV(74));
  LOCAL vp:=gV(75);
  LOCAL vs:=gV(76);
  LOCAL la:=gV(77);
  LOCAL ipri:=va/(vp*k);
  LOCAL isec:=va/(vs*k);
  LOCAL lv:=vs*la*k;
  LOCAL pc:=100*lv/va;
  LOCAL pf:=3;
  LOCAL c, t;
  addRow("Turns ratio",fmt3(vp/vs)+" : 1",IFTE(vp>=vs,"step down","step up"),cText);
  addRow("Primary FLA",fmt3(ipri)+" A",IFTE(k>1,"VA / (V x 1.732)","VA / V"),cText);
  addRow("Secondary FLA",fmt3(isec)+" A","full load",cSelc);
  IF la>0 THEN
    c:=cGood;
    t:="OK";
    IF pc>80 THEN
      c:=cWarn;
      t:="near full load";
    END;
    IF pc>100 THEN
      c:=cBad;
      t:="OVERLOADED";
    END;
    addRow("Load",fmtN(pc,0)+"%  "+fmtU(lv,"VA"),t,c);
  END;
  // NEC 450.3(B), primary-only protection
  IF ipri>=2 THEN pf:=1.67; END;
  IF ipri>=9 THEN pf:=1.25; END;
  IF ipri>=9 THEN
    addRow("Primary OCPD",istr(stdUp(pf*ipri))+" A max","125%, NEC 450.3(B)",cText);
  ELSE
    addRow("Primary OCPD",fmt3(pf*ipri)+" A max",istr(pf*100)+"%, NEC 450.3(B)",cText);
  END;
  RETURN 1;
END;

// usual system name for a wye voltage
sysName(v)
BEGIN
  IF ABS(v-208)<=2 THEN RETURN "208Y/120 V"; END;
  IF ABS(v-480)<=5 THEN RETURN "480Y/277 V"; END;
  IF ABS(v-600)<=5 THEN RETURN "600Y/347 V"; END;
  IF ABS(v-415)<=5 THEN RETURN "415Y/240 V"; END;
  IF ABS(v-400)<=4 THEN RETURN "400Y/230 V"; END;
  RETURN "";
END;

calcWye()
BEGIN
  LOCAL v:=gV(96);
  LOCAL a:=gV(97);
  addRow("Power",fmtU(gRt3*v*a,"VA"),"1.732 x V x A",cSelc);
  IF gV(95)==1 THEN
    addRow("Line-neutral",fmt1(v/gRt3)+" V","V / 1.732",cText);
    addRow("Phase volts",fmt1(v/gRt3)+" V","across each winding",cText);
    addRow("Phase amps",fmt3(a)+" A","same as line amps",cText);
    IF sysName(v)<>"" THEN addRow("System",sysName(v),"",cText); END;
  ELSE
    addRow("Phase volts",fmt1(v)+" V","same as line volts",cText);
    addRow("Phase amps",fmt3(a/gRt3)+" A","line amps / 1.732",cText);
    addRow("High leg-N",fmt1(v*gRt3/2)+" V","center-tapped delta",cWarn);
    addRow("Other legs-N",fmt1(v/2)+" V","for 120 V loads",cText);
  END;

END;

calcHeat()
BEGIN
  LOCAL kw:=gV(78);
  LOCAL vr:=gV(79);
  LOCAL va:=gV(80);
  LOCAL k:=phk(gV(81));
  LOCAL am:=gV(82);
  LOCAL dt:=gV(83);
  LOCAL kn:=kw*(va/vr)^2;
  LOCAL btu:=kn*3412.14;
  LOCAL km;
  addRow("Output now",fmt3(kn)+" kW",fmtN(100*kn/kw,0)+"% of rated, at "+fmtTr(va,1)+" V",cSelc);
  addRow("Heat now",fmtN(btu,0)+" BTU/h","kW x 3412",cText);
  addRow("Amps now",fmt3(kn*1000/(va*k))+" A","rated amps "+fmt3(kw*1000/(vr*k)),cText);
  IF k==1 THEN addRow("Element",fmtOhm(vr*vr/(kw*1000)),"total, from the rating",cText); END;
  IF am>0 THEN
    km:=va*am*k/1000;
    btu:=km*3412.14;
    addRow("Measured",fmt3(km)+" kW",fmtN(btu,0)+" BTU/h",cText);
  END;
  IF dt>0 THEN addRow("Airflow",fmtN(btu/(1.08*dt),0)+" CFM","BTU/h / (1.08 x "+fmtTr(dt,1)+"°F)",cSelc); END;
  addRow("Circuit 125%",fmt3(1.25*kw*1000/(vr*k))+" A","at rated volts",cText);
  RETURN 1;
END;

calcTherm()
BEGIN
  LOCAL cv:=gV(84);
  LOCAL rk:=gV(85)*1000;
  LOCAL tf:=gV(86);
  LOCAL tr, rx, d, c, t;
  IF rk>=5000000 THEN
    addRow("Sensor","OPEN","reads OL",cBad);
    RETURN 0;
  END;
  IF rk<=50 THEN
    addRow("Sensor","SHORTED","reads near 0",cBad);
    RETURN 0;
  END;
  tr:=thTemp(cv,rk);
  rx:=thRes(cv,tf);
  d:=tr-tf;
  c:=cGood;
  t:="sensor OK";
  IF ABS(d)>2 THEN
    c:=cWarn;
    t:="check the sensor";
  END;
  IF ABS(d)>5 THEN
    c:=cBad;
    t:="bad sensor / curve";
  END;
  addRow("Reads as",fmt1(tr)+"°F",fmt1((tr-32)/1.8)+"°C",cSelc);
  addRow("Should read",fmt3(rx/1000)+" kΩ","at "+fmtTr(tf,1)+"°F",cText);
  addRow("Difference",IFTE(d>=0,"+","")+fmt1(d)+"°F",t,c);
  addRow("At 32°F",fmt3(thRes(cv,32)/1000)+" kΩ","this curve",cText);
  addRow("At 77°F",fmt3(thRes(cv,77)/1000)+" kΩ","",cText);
  addRow("At 100°F",fmt3(thRes(cv,100)/1000)+" kΩ","",cText);
  RETURN 1;
END;

calcCost()
BEGIN
  LOCAL kw:=gV(5)*gV(6)*phk(gV(7))*gV(8)/1000;
  LOCAL rate:=gV(89);
  LOCAL hr:=gV(90);
  LOCAL cap:=gV(91);
  addRow("Power",fmt3(kw)+" kW","V x A x PF"+IFTE(gV(7)==2," x 1.732",""),cSelc);
  addRow("Per hour",money(kw*rate),fmt3(kw)+" kWh",cText);
  addRow("Per day",money(kw*rate*hr),fmtTr(hr,1)+" h a day",cText);
  addRow("Per month",money(kw*rate*hr*30),fmtN(kw*hr*30,0)+" kWh, 30 days",cSelc);
  addRow("Per year",money(kw*rate*hr*365),fmtN(kw*hr*365,0)+" kWh",cText);
  IF cap>0 AND kw>0 THEN
    addRow("EER now",fmtN(cap/(kw*1000),1),"COP "+fmtN(cap/(kw*3412.14),2),cText);
    addRow("kW per ton",fmtN(kw/(cap/12000),2),fmtN(cap/12000,2)+" tons",cText);
  END;
  RETURN 1;
END;

convFactor(ct)
BEGIN
  IF ct==1 THEN RETURN {1,1000,745.7,0.29307107,293.07107,3516.8528}; END;
  IF ct==2 THEN RETURN {1,0.001,0.00029307107,29.307107,0.27777778}; END;
  IF ct==3 THEN RETURN {1,0.001,0.000001,1000}; END;
  IF ct==4 THEN RETURN {1,0.001,1000}; END;
  IF ct==5 THEN RETURN {1,1000,1000000}; END;
  RETURN {1,0.001,0.000001,1000000};
END;

// temperature in unit u (1 F, 2 C, 3 K) to kelvin, and back
degK(u,v)
BEGIN
  IF u==1 THEN RETURN (v-32)/1.8+273.15; END;
  IF u==2 THEN RETURN v+273.15; END;
  RETURN v;
END;

degOf(u,k)
BEGIN
  IF u==1 THEN RETURN (k-273.15)*1.8+32; END;
  IF u==2 THEN RETURN k-273.15; END;
  RETURN k;
END;

calcConv()
BEGIN
  LOCAL ct:=gV(92);
  LOCAL u:=convUnits(ct);
  LOCAL n:=SIZE(u);
  LOCAL fr:=MAX(1,MIN(n,gV(94)));
  LOCAL v:=gV(93);
  LOCAL f:=convFactor(ct);
  LOCAL bv, k, ov;
  IF ct==7 THEN bv:=degK(fr,v); ELSE bv:=v*f(fr); END;
  FOR k FROM 1 TO n DO
    IF ct==7 THEN ov:=degOf(k,bv); ELSE ov:=bv/f(k); END;
    addRow(u(k),fmtE(ov),IFTE(k==fr,"entered",""),IFTE(k==fr,cSelc,cText));
  END;
  IF ct==7 AND bv<0 THEN addRow("Note","below absolute zero","",cBad); END;
  RETURN 1;
END;

// ================= input handling =================

// returns 0 to quit
onHome(ev)
BEGIN
  LOCAL mx, my, k;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    IF my<21 THEN
      IF mx>=272 THEN RETURN 0; END;
      IF mx>=224 THEN showHelp(); END;
      RETURN 1;
    END;
    k:=IP((my-23)/43)*4+IP(mx/80)+1;
    IF my>=23 AND k>=1 AND k<=gNT THEN openTool(k); END;
    RETURN 1;
  END;
  IF ev==7 THEN gHome:=IFTE(gHome==1,gNT,gHome-1); END;
  IF ev==8 THEN gHome:=IFTE(gHome==gNT,1,gHome+1); END;
  IF ev==2 THEN gHome:=IFTE(gHome<=4,gHome+gNT-4,gHome-4); END;
  IF ev==12 THEN gHome:=IFTE(gHome>gNT-4,gHome-gNT+4,gHome+4); END;
  IF ev==30 THEN openTool(gHome); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN RETURN 0; END;
  RETURN 1;
END;

onTool(ev)
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, mx, my, rw, idx, dg;
  IF gTSel>n THEN gTSel:=1; END;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    commitEnt();
    IF my>=220 THEN
      k:=IP(mx/64)+1;
      IF k==1 THEN editSlot(sl(gTSel)); END;
      IF k==2 THEN resetTool(); END;
      IF k==3 THEN openTool(IFTE(gTool==1,gNT,gTool-1)); END;
      IF k==4 THEN openTool(IFTE(gTool==gNT,1,gTool+1)); END;
      IF k==5 THEN goHome(); END;
      RETURN 1;
    END;
    IF my<21 THEN
      IF mx>=272 THEN toolHelp(gTool); END;
      RETURN 1;
    END;
    // tap an input box to change it
    rw:=IP((my-24)/20);
    idx:=rw*2+IP(mx/160)+1;
    IF my>=24 AND rw<IP((n+1)/2) AND idx<=n THEN
      gTSel:=idx;
      editSlot(sl(idx));
    END;
    RETURN 1;
  END;
  // digits and the decimal point type straight into a number box
  dg:=keyDigit(ev);
  IF dg>=0 THEN
    IF gSC(sl(gTSel))==0 THEN typeKey(dg); END;
    RETURN 1;
  END;
  IF gEnt THEN
    IF ev==45 OR ev==27 THEN
      gENeg:=1-gENeg;
      RETURN 1;
    END;
    IF ev==19 THEN
      backKey();
      RETURN 1;
    END;
    IF ev==4 THEN
      gEnt:=0;
      RETURN 1;
    END;
    // any other key keeps the typed number first
    commitEnt();
    IF ev==30 THEN RETURN 1; END;
  ELSE
    IF (ev==45 OR ev==27) AND gSC(sl(gTSel))==0 THEN
      gEnt:=1;
      gEK:={};
      gENeg:=1;
      RETURN 1;
    END;
  END;
  IF ev==7 THEN gTSel:=IFTE(gTSel==1,n,gTSel-1); END;
  IF ev==8 THEN gTSel:=IFTE(gTSel==n,1,gTSel+1); END;
  IF ev==2 THEN bumpSlot(sl(gTSel),1); END;
  IF ev==12 THEN bumpSlot(sl(gTSel),-1); END;
  IF ev==30 THEN editSlot(sl(gTSel)); END;
  IF ev==3 THEN toolHelp(gTool); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN goHome(); END;
  RETURN 1;
END;

EXPORT HVACELEC()
BEGIN
  LOCAL run:=1;
  LOCAL ev;
  DIMGROB_P(G1,320,240);
  DIMGROB_P(G9,320,30);
  gEnt:=0;
  WHILE run DO
    IF gScr==1 THEN drawHome(); ELSE drawTool(); END;
    ev:=WAIT(-1);   // key code, touch event list, or -1 after a 1 minute timeout
    IF gScr==1 THEN run:=onHome(ev); ELSE run:=onTool(ev); END;
  END;
END;
