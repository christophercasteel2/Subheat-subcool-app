// ================= number formatting =================

// whole number as text, independent of the calculator's number format
istr(v)
BEGIN
  LOCAL n:=ROUND(ABS(v),0);
  LOCAL s:="";
  IF n<1 THEN RETURN "0"; END;
  WHILE n>=1 DO
    s:=CHAR(48+(n MOD 10))+s;
    n:=IP(n/10);
  END;
  IF v<0 THEN s:="-"+s; END;
  RETURN s;
END;

// one decimal place
fmt1(v)
BEGIN
  LOCAL n:=ROUND(ABS(v)*10,0);
  LOCAL s:=istr(IP(n/10))+"."+CHAR(48+(n MOD 10));
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// v with d decimal places (d = 0 to 5)
fmtN(v,d)
BEGIN
  LOCAL m:=10^d;
  LOCAL n:=ROUND(ABS(v)*m,0);
  LOCAL s:=istr(IP(n/m));
  LOCAL f;
  IF d>0 THEN
    f:=istr(n MOD m);
    WHILE DIM(f)<d DO f:="0"+f; END;
    s:=s+"."+f;
  END;
  IF v<0 AND n>0 THEN s:="-"+s; END;
  RETURN s;
END;

// up to d decimals, trailing zeros dropped
fmtTr(v,d)
BEGIN
  LOCAL s:=fmtN(v,d);
  IF d>0 THEN
    WHILE RIGHT(s,1)=="0" DO s:=LEFT(s,DIM(s)-1); END;
    IF RIGHT(s,1)=="." THEN s:=LEFT(s,DIM(s)-1); END;
  END;
  RETURN s;
END;

// any size of number; very large or small ones as 1.23E-6
fmtE(v)
BEGIN
  LOCAL a:=ABS(v);
  LOCAL ex, m;
  IF a<0.000000000001 THEN RETURN "0"; END;
  IF a>=0.000001 AND a<10000000 THEN
    // six significant figures
    ex:=5-FLOOR(LOG(a)+0.0000000001);
    IF ex>5 THEN RETURN fmtTr(v,MIN(11,ex)); END;
    ex:=5;
    IF a>=10 THEN ex:=4; END;
    IF a>=100 THEN ex:=3; END;
    IF a>=1000 THEN ex:=2; END;
    IF a>=10000 THEN ex:=1; END;
    IF a>=100000 THEN ex:=0; END;
    RETURN fmtTr(v,ex);
  END;
  ex:=FLOOR(LN(a)/LN(10));
  m:=a/10^ex;
  IF m>=9.9995 THEN
    m:=m/10;
    ex:=ex+1;
  END;
  IF m<1 THEN
    m:=m*10;
    ex:=ex-1;
  END;
  RETURN IFTE(v<0,"-","")+fmtTr(m,5)+"E"+istr(ex);
END;

// ================= circuit helpers =================

sqroot(x)
BEGIN
  RETURN MAX(0,x)^0.5;
END;

// two resistors in parallel; 0 means that resistor is not there
par2(a,b)
BEGIN
  IF a<=0 THEN RETURN MAX(0,b); END;
  IF b<=0 THEN RETURN a; END;
  RETURN a*b/(a+b);
END;

// a value with a metric prefix, e.g. 4.7 kΩ or 15 mA
engTxt(v,u)
BEGIN
  LOCAL a:=ABS(v);
  IF a==0 THEN RETURN "0 "+u; END;
  IF a>=1000000 THEN RETURN fmtE(v/1000000)+" M"+u; END;
  IF a>=1000 THEN RETURN fmtE(v/1000)+" k"+u; END;
  IF a>=1 THEN RETURN fmtE(v)+" "+u; END;
  IF a>=0.001 THEN RETURN fmtE(v*1000)+" m"+u; END;
  RETURN fmtE(v*1000000)+" µ"+u;
END;

// text shown in an input box
slotText(s)
BEGIN
  LOCAL u:=slotUnit(s);
  LOCAL v:=gV(s);
  LOCAL lst, t;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    RETURN lst(MAX(1,MIN(SIZE(lst),v)));
  END;
  IF v==0 AND gSZ(s)==1 THEN RETURN "--"; END;
  IF v==0 AND gSZ(s)==2 THEN RETURN "?"; END;
  IF gSD(s)<0 THEN t:=fmtTr(v,4); ELSE t:=fmtN(v,gSD(s)); END;
  IF u=="" THEN RETURN t; END;
  IF LEFT(u,1)=="°" THEN RETURN t+u; END;
  RETURN t+" "+u;
END;

setSlot(s,v)
BEGIN
  gV(s):=MAX(gSLo(s),MIN(gSHi(s),v));
  IF gSC(s)>0 THEN gV(s):=ROUND(gV(s),0); END;
END;

// Up/Down: step a number, or move through a choice list
bumpSlot(s,d)
BEGIN
  LOCAL n;
  IF gSC(s)>0 THEN
    n:=SIZE(choiceList(gSC(s)));
    gV(s):=gV(s)+IFTE(d>0,1,-1);
    IF gV(s)>n THEN gV(s):=1; END;
    IF gV(s)<1 THEN gV(s):=n; END;
  ELSE
    setSlot(s,gV(s)+d*gSS(s));
  END;
END;

// a two-way choice flips, a longer list opens, a number opens a dialog
editSlot(s)
BEGIN
  LOCAL v:=gV(s);
  LOCAL lst;
  IF gSC(s)>0 THEN
    lst:=choiceList(gSC(s));
    IF SIZE(lst)==2 THEN
      setSlot(s,IFTE(v==1,2,1));
    ELSE
      IF CHOOSE(v,slotName(s),lst) THEN
        IF v>=1 THEN setSlot(s,v); END;
      END;
    END;
  ELSE
    IF INPUT(v,slotName(s),slotName(s)+":",gSH(s)) THEN
      IF TYPE(v)==0 THEN setSlot(s,v); END;
    END;
  END;
  gDown:=0;
END;

  LOCAL kc:={47,42,43,44,37,38,39,32,33,34,48};

entText()
BEGIN
  LOCAL s:=IFTE(gENeg,"-","");
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK) DO
    s:=s+IFTE(gEK(k)==10,".",CHAR(48+gEK(k)));
  END;
  RETURN s;
END;

entVal()
BEGIN
  LOCAL v:=0;
  LOCAL sc:=0;
  LOCAL k, d;
  FOR k FROM 1 TO SIZE(gEK) DO
    d:=gEK(k);
    IF d==10 THEN
      sc:=1;
    ELSE
      IF sc>0 THEN
        sc:=sc/10;
        v:=v+d*sc;
      ELSE
        v:=v*10+d;
      END;
    END;
  END;
  RETURN IFTE(gENeg,-v,v);
END;

typeKey(d)
BEGIN
  LOCAL hasdot:=0;
  LOCAL k;
  IF gEnt==0 THEN
    gEnt:=1;
    gEK:={};
    gENeg:=0;
  END;
  FOR k FROM 1 TO SIZE(gEK) DO
    IF gEK(k)==10 THEN hasdot:=1; END;
  END;
  IF (d<10 OR hasdot==0) AND SIZE(gEK)<12 THEN gEK(SIZE(gEK)+1):=d; END;
END;

backKey()
BEGIN
  LOCAL nl:={};
  LOCAL k;
  FOR k FROM 1 TO SIZE(gEK)-1 DO
    nl(k):=gEK(k);
  END;
  gEK:=nl;
  IF SIZE(gEK)==0 THEN gEnt:=0; END;
END;

  IF gEnt AND SIZE(gEK)>0 THEN setSlot(sl(gTSel),entVal()); END;
  gEnt:=0;
END;

// arrow from (x1,y1) to (x2,y2)
arr(x1,y1,x2,y2,c)
BEGIN
  LOCAL dx:=x2-x1;
  LOCAL dy:=y2-y1;
  LOCAL l:=MAX(0.001,sqroot(dx*dx+dy*dy));
  LOCAL ux, uy;
  ux:=dx/l;
  uy:=dy/l;
  LINE_P(G1,x1,y1,x2,y2,c);
  TRIANGLE_P(G1,x2,y2,ROUND(x2-5*ux-3*uy,0),ROUND(y2-5*uy+3*ux,0),ROUND(x2-5*ux+3*uy,0),ROUND(y2-5*uy-3*ux,0),c);
END;

// one long page of lines t, scrolled with Up/Down, + / -, or a finger;
// Left/Right jump to the previous / next section (st: first line of
// each section); Esc or Enter (or CLOSE) goes back
showScroll(ttl,t,st)
BEGIN
  LOCAL n:=SIZE(t);
  LOCAL nr:=15;
  LOCAL mx:=MAX(1,n-nr+1);
  LOCAL done:=0;
  LOCAL dragged:=0;
  LOCAL y0:=0;
  LOCAL dg:={};
  LOCAL k, ev, dy, tx, bh, by, j;
  // drawing number of each line: 0 text, -1 part of a drawing
  FOR k FROM 1 TO n DO
    tx:=t(k);
    dg(k):=0;
    IF DIM(tx)>0 THEN
      IF LEFT(tx,1)=="@" THEN
        dg(k):=-1;
        FOR j FROM 1 TO 42 DO
          IF tx=="@"+istr(j) THEN dg(k):=j; END;
        END;
      END;
    END;
  END;
  gSTop:=1;
  gCurS:=0;
  gDgN:=1;
  REPEAT
    RECT_P(G1,0,0,319,239,cBg);
    FOR k FROM -5 TO nr-1 DO
      j:=gSTop+k;
      IF j>=1 AND j<=n THEN
        IF dg(j)>0 THEN
          drawNoteDg(dg(j),82,24+k*14+2);
        ELSE
          IF k>=0 AND dg(j)==0 THEN
            tx:=t(j);
            TEXTOUT_P(tx,G1,6,24+k*14,1,IFTE(LEFT(tx,1)==" ",cText,cSelc));
          END;
        END;
      END;
    END;
    RECT_P(G1,0,0,319,21,cPanel);
    TEXTOUT_P(ttl,G1,5,4,2,cText);
    rtext(istr(gSTop)+"-"+istr(MIN(n,gSTop+nr-1))+" of "+istr(n),262,6,1,cDim);
    chip("CLOSE",268);
    // scroll bar
    bh:=MAX(10,ROUND(214*MIN(1,nr/MAX(1,n)),0));
    by:=22+ROUND((214-bh)*(gSTop-1)/MAX(1,mx-1),0);
    RECT_P(G1,316,22,319,236,cRow);
    RECT_P(G1,316,by,319,by+bh,cEdge);
    BLIT_P(G0,G1);
    ev:=WAIT(-1);
    IF TYPE(ev)==6 THEN
      IF ev(1)==0 AND SIZE(ev)>=3 THEN
        y0:=ev(3);
        dragged:=0;
      END;
      IF ev(1)==1 AND SIZE(ev)>=3 THEN
        dy:=IP((y0-ev(3))/14);
        IF dy<>0 THEN
          gSTop:=gSTop+dy;
          y0:=y0-dy*14;
          dragged:=1;
        END;
      END;
      IF ev(1)==3 AND SIZE(ev)>=3 AND dragged==0 THEN
        IF ev(3)<21 THEN
          IF ev(2)>=268 THEN done:=1; END;
        ELSE
          gSTop:=gSTop+IFTE(ev(3)>=130,nr-1,1-nr);
        END;
      END;
    ELSE
      IF ev==2 THEN gSTop:=gSTop-1; END;
      IF ev==12 THEN gSTop:=gSTop+1; END;
      IF ev==50 THEN gSTop:=gSTop+nr-1; END;
      IF ev==45 THEN gSTop:=gSTop-nr+1; END;
      IF ev==8 THEN
        k:=SIZE(st);
        WHILE k>=1 AND st(k)>gSTop DO k:=k-1; END;
        IF k<SIZE(st) THEN gSTop:=st(k+1); ELSE gSTop:=mx; END;
      END;
      IF ev==7 THEN
        k:=1;
        WHILE k<=SIZE(st) AND st(k)<gSTop DO k:=k+1; END;
        gSTop:=IFTE(k>=2,st(k-1),1);
      END;
      IF ev==4 OR ev==30 THEN done:=1; END;
    END;
    gSTop:=MAX(1,MIN(mx,gSTop));
  UNTIL done;
  gDgN:=0;
  gDown:=0;
END;

// 1 Kirchhoff's current law at one point
calcKCL()
BEGIN
  LOCAL u:=gV(1);
  LOCAL sn:=0;
  LOCAL so:=0;
  LOCAL k, x;
  LOCAL lbl:=choiceList(1);
  FOR k FROM 1 TO 4 DO
    IF k<>u THEN sn:=sn+gV(1+k); END;
    IF k+4<>u THEN so:=so+gV(5+k); END;
  END;
  addRow("Known in",fmtE(sn)+" A","",cText);
  addRow("Known out",fmtE(so)+" A","",cText);
  IF u==9 THEN
    addRow("In - out",fmtE(sn-so)+" A","",IFTE(ABS(sn-so)<0.000001,cGood,cWarn));
    addRow("","In (+) and out (-) add to 0","",cDim);
    RETURN 1;
  END;
  x:=IFTE(u<=4,so-sn,sn-so);
  addRow(lbl(u),fmtE(x)+" A","",IFTE(x>=0,cSelc,cWarn));
  addRow("","Entering = +, leaving = -","",cDim);
  IF x<0 THEN
    addRow("","Negative: the assumed","",cWarn);
    addRow("","direction was wrong; it flows","",cWarn);
    addRow("",IFTE(u<=4,"OUT of","INTO")+" the point, "+fmtE(-x)+" A","",cWarn);
  END;
  RETURN 1;
END;

// 2 Kirchhoff's voltage law around one loop
calcKVL()
BEGIN
  LOCAL n:=gV(11);
  LOCAL et:=gV(10);
  LOCAL sm:=0;
  LOCAL un:=0;
  LOCAL ux:=0;
  LOCAL k, ek;
  FOR k FROM 1 TO n DO
    ek:=gV(11+k);
    IF ek>0 THEN
      sm:=sm+ek;
    ELSE
      un:=un+1;
      ux:=k;
    END;
  END;
  addRow("Known drops",fmtE(sm)+" V","",cText);
  IF et<=0 AND un==0 THEN
    addRow("ET",fmtE(sm)+" V","",cSelc);
    addRow("","ET = sum of the drops","",cDim);
    RETURN 1;
  END;
  IF un==1 AND et>0 THEN
    addRow("E"+istr(ux),fmtE(et-sm)+" V","",IFTE(et-sm>=0,cSelc,cBad));
    addRow("","E"+istr(ux)+" = ET - known drops","",cDim);
    RETURN 1;
  END;
  IF un==0 THEN
    addRow("ET",fmtE(et)+" V","",cText);
    addRow("-drops + ET",fmtE(et-sm)+" V","",IFTE(ABS(et-sm)<0.0001*MAX(1,et),cGood,cWarn));
    addRow("","Loop sum must be 0","",cDim);
    RETURN 1;
  END;
  addRow("Enter","one unknown only","",cWarn);
  RETURN 0;
END;

// 3 two sources, each with a resistor, feeding a middle resistor
calcNet()
BEGIN
  LOCAL e1:=gV(19);
  LOCAL e2:=IFTE(gV(21)==1,gV(20),-gV(20));
  LOCAL r1:=gV(22);
  LOCAL r2:=gV(23);
  LOCAL r3:=gV(24);
  LOCAL v, i1, i2, i3;
  IF r1<=0 OR r2<=0 OR r3<=0 THEN
    addRow("Enter","R1, R2 and R3","",cWarn);
    RETURN 0;
  END;
  v:=(e1/r1+e2/r2)/(1/r1+1/r2+1/r3);
  i1:=(e1-v)/r1;
  i2:=(e2-v)/r2;
  i3:=v/r3;
  addRow("I1",fmtE(i1)+" A","",IFTE(i1<0,cWarn,cSelc));
  addRow("I2",fmtE(i2)+" A","",IFTE(i2<0,cWarn,cSelc));
  addRow("I3",fmtE(i3)+" A","",IFTE(i3<0,cWarn,cSelc));
  addRow("E1 (R1)",fmtE(i1*r1)+" V","",cSelc);
  addRow("E2 (R2)",fmtE(i2*r2)+" V","",cSelc);
  addRow("E3 (R3)",fmtE(i3*r3)+" V","",cSelc);
  addRow("KCL check","I1 + I2 = "+fmtE(i1+i2),"",cDim);
  addRow("","L1: "+fmtE(e1)+" = "+fmtE(r1+r3)+" I1 + "+fmtE(r3)+" I2","",cDim);
  addRow("","L2: "+fmtE(e2)+" = "+fmtE(r3)+" I1 + "+fmtE(r2+r3)+" I2","",cDim);
  IF i1<0 OR i2<0 OR i3<0 THEN
    addRow("","Negative = flows opposite","",cWarn);
  END;
  RETURN 1;
END;

// 4 Thevenin and Norton equivalents of a divider (Fig 9-34)
calcThev()
BEGIN
  LOCAL es:=gV(29);
  LOCAL r1:=gV(30);
  LOCAL r2:=gV(31);
  LOCAL r3:=gV(32);
  LOCAL rl:=gV(33);
  LOCAL eth, rth, il;
  IF r1<=0 OR r2<=0 THEN
    addRow("Enter","R1 and R2","",cWarn);
    RETURN 0;
  END;
  eth:=es*r2/(r1+r2);
  rth:=r1*r2/(r1+r2)+MAX(0,r3);
  addRow("ETH",fmtE(eth)+" V","",cSelc);
  addRow("RTH",fmtE(rth)+" Ω","",cSelc);
  addRow("IN",fmtE(eth/rth)+" A","",cSelc);
  addRow("RN",fmtE(rth)+" Ω","",cSelc);
  addRow("","ETH = Es x R2 / (R1 + R2)","",cDim);
  addRow("","RTH = R1 || R2 (+ R3); IN = ETH/RTH","",cDim);
  IF rl>0 THEN
    il:=eth/(rth+rl);
    addRow("I load",fmtE(il)+" A","",cGood);
    addRow("E load",fmtE(il*rl)+" V","",cGood);
    addRow("P load",fmtE(il*il*rl)+" W","",cText);
  END;
  RETURN 1;
END;

// 5 a Thevenin or Norton equivalent with a load
calcEqv()
BEGIN
  LOCAL nrt:=gV(34)==2;
  LOCAL v:=gV(35);
  LOCAL r:=gV(36);
  LOCAL rl:=gV(37);
  LOCAL eth, il;
  IF r<=0 THEN
    addRow("Enter","the resistance","",cWarn);
    RETURN 0;
  END;
  eth:=IFTE(nrt,v*r,v);
  addRow("ETH",fmtE(eth)+" V","",IFTE(nrt,cSelc,cText));
  addRow("RTH = RN",fmtE(r)+" Ω","",cText);
  addRow("IN",fmtE(eth/r)+" A","",IFTE(nrt,cText,cSelc));
  IF rl>0 THEN
    il:=eth/(r+rl);
    addRow("I load",fmtE(il)+" A","",cGood);
    addRow("E load",fmtE(il*rl)+" V","",cGood);
    addRow("R total",fmtE(r+rl)+" Ω","Thevenin: RTH + RL",cText);
    addRow("RN || RL",fmtE(par2(r,rl))+" Ω","Norton",cText);
    addRow("I in RN",fmtE(il*rl/r)+" A","Norton",cText);
  END;
  RETURN 1;
END;

// 6 superposition: current through the middle resistor R2
calcSup()
BEGIN
  LOCAL e1:=gV(38);
  LOCAL r1:=gV(39);
  LOCAL r2:=gV(40);
  LOCAL r3:=gV(41);
  LOCAL e2:=gV(42);
  LOCAL ad:=gV(43)==1;
  LOCAL t1, i1, ia, t2, i2, ib, it;
  IF r1<=0 OR r2<=0 OR r3<=0 THEN
    addRow("Enter","R1, R2 and R3","",cWarn);
    RETURN 0;
  END;
  t1:=r1+par2(r2,r3);
  i1:=e1/t1;
  ia:=i1*par2(r2,r3)/r2;
  t2:=r3+par2(r1,r2);
  i2:=e2/t2;
  ib:=i2*par2(r1,r2)/r2;
  it:=IFTE(ad,ia+ib,ABS(ia-ib));
  addRow("1 Es2 short","RT "+fmtE(t1)+" Ω","",cDim);
  addRow(" IT, I2","   "+fmtE(i1)+", "+fmtE(ia)+" A","",cText);
  addRow("2 Es1 short","RT "+fmtE(t2)+" Ω","",cDim);
  addRow(" IT, I2","   "+fmtE(i2)+", "+fmtE(ib)+" A","",cText);
  addRow("3 I2 total",fmtE(it)+" A","",cSelc);
  addRow("E2 (R2)",fmtE(it*r2)+" V","",cSelc);
  addRow("",IFTE(ad,"Same direction: add","Opposite: subtract, larger"),"",cDim);
  IF NOT ad THEN addRow("","direction: from Es"+IFTE(ia>=ib,"1","2"),"",cDim); END;
  RETURN 1;
END;

// 7 R = K x L / CM, solved for R, CM, L or K
calcWire()
BEGIN
  LOCAL fd:=gV(44);
  LOCAL k20:=kOf(45,46);
  LOCAL al:=IFTE(gV(45)<=14,gAl(MIN(14,gV(45))),0);
  LOCAL k:=k20*(1+al*(gV(51)-20));
  LOCAL cm:=cmOf(47,48);
  LOCAL l:=gV(49);
  LOCAL r:=gV(50);
  LOCAL x, j, b;
  IF gV(51)<>20 THEN addRow("K at "+fmtE(gV(51))+"°C",fmtE(k),"K20 (1 + α ΔT)",cText); END;
  IF fd==1 THEN
    IF cm<=0 THEN RETURN 0; END;
    addRow("R",fmtE(k*l/cm)+" Ω","K x L / CM",cSelc);
    addRow("K, L, CM",fmtE(k)+", "+fmtE(l)+", "+fmtE(cm),"",cDim);
  END;
  IF fd==2 THEN
    IF r<=0 THEN
      addRow("Enter","R","",cWarn);
      RETURN 0;
    END;
    x:=k*l/r;
    addRow("CM",fmtE(x)+" CM","K x L / R",cSelc);
    j:=sizeFor(x);
    addRow("Use size",IFTE(j>0,sizeName(j)+" ("+fmtE(gCM(MAX(1,j)))+")","over 2000 kcmil"),"next size up",cGood);
  END;
  IF fd==3 THEN
    IF k<=0 THEN RETURN 0; END;
    addRow("L",fmtE(r*cm/k)+" ft","R x CM / K",cSelc);
  END;
  IF fd==4 THEN
    IF l<=0 THEN
      addRow("Enter","L","",cWarn);
      RETURN 0;
    END;
    x:=r*cm/l;
    addRow("K",fmtE(x)+" Ω-CM/ft","R x CM / L",cSelc);
    b:=1;
    FOR j FROM 2 TO 14 DO
      IF ABS(gK(j)-x)<ABS(gK(b)-x) THEN b:=j; END;
    END;
    addRow("Closest",matName(b)+" (K "+fmtE(gK(b))+")","",cGood);
  END;
  addRow("","L = total wire length, ft","",cDim);
  RETURN 1;
END;

// 9 conductor size for a long run, from the allowed voltage drop

// 11 conductor size for a load (the book's Example 11-8 method)
calcLoad()
BEGIN
  LOCAL u:=gV(74);

  LOCAL p:=gV(73)*IFTE(u==3 OR u==6,1000,IFTE(u==4 OR u==7,746,1));

  LOCAL ic:=gInsCol(gV(77));
  LOCAL t:=ambC(79,80);
  LOCAL tf:=corrF(IFTE(ic==4,3,ic),t);
  LOCAL af:=adjF(gV(81));

  IF tf==0 THEN
    addRow("Temp factor","not allowed","",cBad);
    RETURN 0;
  END;
  rq:=ia/(tf*af);
  addRow("÷ factors",fmtE(rq)+" A","T "+fmtE(tf)+"  adj "+fmtE(af),cSelc);
  c:=IFTE(gV(82)==1,IFTE(rq<=100,1,2),gV(82)-1);

  IF j==0 THEN

  END;
  addRow("Column",colTxt(c),IFTE(gV(82)==1,IFTE(rq<=100,"100 A or less","over 100 A"),"termination"),cText);
  addRow("Use size",sizeName(j),fmtE(ampT(1,j,gV(78),c))+" A in the table",cGood);

// 18 Ohm's law and power
calcOhm()
BEGIN
  LOCAL v:=gV(25);
  LOCAL a:=gV(26);
  LOCAL r:=gV(27);
  LOCAL p:=gV(28);
  LOCAL n:=(v>0)+(a>0)+(r>0)+(p>0);
  LOCAL md:=0;
  LOCAL fm:={"R = E/I  P = E x I","I = E/R  P = E²/R","I = P/E  R = E²/P","E = I x R  P = I² x R","E = P/I  R = P/I²","E = √(PR)  I = √(P/R)"};
  IF n<2 THEN
    addRow("Enter","any two values","",cWarn);
    RETURN 0;
  END;
  IF v>0 AND a>0 THEN md:=1; END;
  IF md==0 AND v>0 AND r>0 THEN md:=2; END;
  IF md==0 AND v>0 AND p>0 THEN md:=3; END;
  IF md==0 AND a>0 AND r>0 THEN md:=4; END;
  IF md==0 AND a>0 AND p>0 THEN md:=5; END;
  IF md==0 THEN md:=6; END;
  IF md==1 THEN
    r:=v/a;
    p:=v*a;
  END;
  IF md==2 THEN
    a:=v/r;
    p:=v*v/r;
  END;
  IF md==3 THEN
    a:=p/v;
    r:=v*v/p;
  END;
  IF md==4 THEN
    v:=a*r;
    p:=a*a*r;
  END;
  IF md==5 THEN
    v:=p/a;
    r:=p/(a*a);
  END;
  IF md==6 THEN
    v:=sqroot(p*r);
    a:=sqroot(p/r);
  END;
  addRow("Volts E",fmtE(v)+" V","",cSelc);
  addRow("Amps I",fmtE(a)+" A","",cSelc);
  addRow("Ohms R",fmtE(r)+" Ω","",cSelc);
  addRow("Watts P",fmtE(p)+" W","",cSelc);
  addRow("",fm(md),"",cDim);
  IF n>2 THEN addRow("","Used 2 values; zero the rest","",cWarn); END;
  RETURN 1;
END;

goHome()
BEGIN
  gHome:=gTool;
  gEnt:=0;
  gScr:=1;
END;

toolsMenu()
BEGIN
  LOCAL t:=gTool;
  IF CHOOSE(t,"Tools",gTName) THEN
    IF t>=1 THEN openTool(t); END;
  END;
  gDown:=0;
END;

// RESET: the tool's example values
resetTool()
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL k;
  FOR k FROM 1 TO SIZE(sl) DO
    gV(sl(k)):=gDef(sl(k));
  END;
  gEnt:=0;
END;

// open tile k: a tool, or the Learn pages
openTile(k)
BEGIN
  gHome:=k;
  IF k>gNT THEN
    learnMenu(k-gNT);
  ELSE
    openTool(k);
  END;
END;

// ================= input handling =================

// returns 0 to quit
onHome(ev)
BEGIN
  LOCAL mx, my, k;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    IF my<21 THEN
      IF mx>=272 THEN RETURN 0; END;
      IF mx>=224 THEN showHelp(); END;
      RETURN 1;
    END;
    k:=IP((my-23)/36)*4+MIN(3,IP(mx/80))+1;
    IF my>=23 AND k>=1 AND k<=gNT+4 THEN openTile(k); END;
    RETURN 1;
  END;
  IF ev==7 THEN gHome:=IFTE(gHome==1,gNT+4,gHome-1); END;
  IF ev==8 THEN gHome:=IFTE(gHome==gNT+4,1,gHome+1); END;
  IF ev==2 THEN
    gHome:=gHome-4;
    WHILE gHome<1 DO gHome:=gHome+4*IP((gNT+4)/4); END;
    WHILE gHome>gNT+4 DO gHome:=gHome-4; END;
  END;
  IF ev==12 THEN
    gHome:=gHome+4;
    IF gHome>gNT+4 THEN gHome:=((gHome-1) MOD 4)+1; END;
  END;
  IF ev==30 THEN openTile(gHome); END;
  IF ev==3 THEN showHelp(); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN RETURN 0; END;
  RETURN 1;
END;

onTool(ev)
BEGIN
  LOCAL sl:=toolIns(gTool);
  LOCAL n:=SIZE(sl);
  LOCAL k, mx, my, rw, idx, dg;
  IF gTSel>n THEN gTSel:=1; END;
  IF TYPE(ev)==6 THEN
    IF ev(1)==0 THEN gDown:=1; END;
    IF ev(1)<>3 OR gDown==0 OR SIZE(ev)<3 THEN RETURN 1; END;
    gDown:=0;
    mx:=ev(2);
    my:=ev(3);
    commitEnt();
    IF my>=220 THEN
      k:=IP(mx/64)+1;
      IF k==1 THEN editSlot(sl(gTSel)); END;
      IF k==2 THEN resetTool(); END;
      IF k==3 THEN openTool(IFTE(gTool==1,gNT,gTool-1)); END;
      IF k==4 THEN openTool(IFTE(gTool==gNT,1,gTool+1)); END;
      IF k==5 THEN goHome(); END;
      RETURN 1;
    END;
    IF my<21 THEN
      IF mx>=272 THEN toolHelp(gTool); END;
      RETURN 1;
    END;
    // tap an input box to change it
    rw:=IP((my-23)/19);
    idx:=rw+1;
    IF my>=23 AND mx<160 AND idx>=1 AND idx<=n THEN
      gTSel:=idx;
      editSlot(sl(idx));
    END;
    RETURN 1;
  END;
  // digits and the decimal point type straight into a number box
  dg:=keyDigit(ev);
  IF dg>=0 THEN
    IF gSC(sl(gTSel))==0 THEN typeKey(dg); END;
    RETURN 1;
  END;
  IF gEnt THEN
    IF ev==45 OR ev==27 THEN
      gENeg:=1-gENeg;
      RETURN 1;
    END;
    IF ev==19 THEN
      backKey();
      RETURN 1;
    END;
    IF ev==4 THEN
      gEnt:=0;
      RETURN 1;
    END;
    // any other key keeps the typed number first
    commitEnt();
    IF ev==30 THEN RETURN 1; END;
  ELSE
    IF (ev==45 OR ev==27) AND gSC(sl(gTSel))==0 THEN
      gEnt:=1;
      gEK:={};
      gENeg:=1;
      RETURN 1;
    END;
  END;
  IF ev==7 THEN gTSel:=IFTE(gTSel==1,n,gTSel-1); END;
  IF ev==8 THEN gTSel:=IFTE(gTSel==n,1,gTSel+1); END;
  IF ev==2 THEN bumpSlot(sl(gTSel),1); END;
  IF ev==12 THEN bumpSlot(sl(gTSel),-1); END;
  IF ev==30 THEN editSlot(sl(gTSel)); END;
  IF ev==3 THEN toolHelp(gTool); END;
  IF ev==13 THEN toolsMenu(); END;
  IF ev==4 THEN goHome(); END;
  RETURN 1;
END;

// ================= main =================

EXPORT ELECREV()
BEGIN
  LOCAL run:=1;
  LOCAL ev;
  DIMGROB_P(G1,320,240);
  DIMGROB_P(G9,320,30);
  gEnt:=0;
  WHILE run DO
    IF gScr==1 THEN drawHome(); ELSE drawTool(); END;
    ev:=WAIT(-1);   // key code, touch event list, or -1 after a 1 minute timeout
    IF gScr==1 THEN run:=onHome(ev); ELSE run:=onTool(ev); END;
  END;
END;
